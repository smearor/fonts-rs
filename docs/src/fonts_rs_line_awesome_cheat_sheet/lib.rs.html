1//! Line Awesome Cheat Sheet application.
2//!
3//! Browse all available Line Awesome icons with a search filter.
4//! Displays each icon with its Unicode codepoint and icon name.
5//! Click an icon to open a detail sidebar.
6//!
7//! The UI lives in `fonts-rs-cheat-sheet`; this binary only supplies
8//! the Line Awesome hooks via [`CheatSheetFamily`]. With the `metadata`
9//! feature (default), search also covers filename-derived keywords
10//! (e.g. "address book" finds `la-address-book-symbolic`), the style
11//! categories (`regular`/`solid`/`brands`) and the vendored Font
12//! Awesome aliases (e.g. "address-book-o" finds
13//! `la-address-book-symbolic`).
14//!
15//! Line Awesome is a hybrid icon set: grid cells render via the
16//! `GtkIconTheme` (`glyph_icon_name`) - the bundled SVGs are
17//! authoritative, while the three bundled `la-{regular-400,solid-900,
18//! brands-400}.ttf` webfonts back `la-icon` text labels so PUA
19//! codepoints never fall back to a random system font.
20//!
21//! Run with:
22//! ```sh
23//! cargo run -p fonts-rs-line-awesome-cheat-sheet
24//! ```
25

27use fonts_rs_cheat_sheet::FamilyKind;
28use fonts_rs_line_awesome::LineAwesome;

32use fonts_rs_model::CodePoint;
33use fonts_rs_model::FontFamily;

35use fonts_rs_model::GlyphMetadata;

39/// Line Awesome hooks for the generic cheat sheet.
40pub struct LineAwesomeSheet;
41
42impl CheatSheetFamily for LineAwesomeSheet {
43    const TITLE: &'static str = "Line Awesome - Cheat Sheet";
44    const NOUN: &'static str = "icons";
45    const APP_ID: &'static str = "io.smearor.fonts_rs.line_awesome_cheat_sheet";
46    const SEARCH_PLACEHOLDER: &'static str = "Search by name, keyword, alias, style, or codepoint (e.g. 'home', 'address-book-o', 'F0001')...";
47    const GLYPH_CSS_CLASS: Option<&'static str> = Some("la-icon");
48    const NAME_FIELD_LABEL: &'static str = "Icon Name";
49    const FAMILY_NAME: &'static str = "Line Awesome";
50    const FAMILY_KIND: FamilyKind = FamilyKind::IconSet;
51    const LICENSE: &'static str = "MIT or Good Boy License";
52
53    type Name = LineAwesomeIconName;
54
55    fn init() -> Result<()> {
56        LineAwesome::install().into_diagnostic()?;
57        // Grid cells render via Image::from_icon_name - the icon theme
58        // path must be added to the display's GtkIconTheme. Runs after
59        // gtk4::init(), so the default display is already open.
60        let display = gtk4::gdk::Display::default().ok_or_else(|| miette::miette!("no default GDK display"))?;
61        LineAwesome::install_icon_theme(&display).into_diagnostic()?;
62        // Hybrid set: the three bundled TTFs drive `la-icon` text
63        // labels. Without install_pango, PUA codepoints resolve via
64        // whatever system font claims them (e.g. Nerd Fonts).
65        LineAwesome::install_pango(&pangocairo::FontMap::default()).into_diagnostic()?;
66        fonts_rs_model::gtk::load_css(&fonts_rs_model::icon_css(&[("la-icon", LineAwesome::CSS_FONT_FAMILY)]));
67        Ok(())
68    }
69
70    fn all_glyphs() -> Vec<(CodePoint, LineAwesomeIconName)> {
71        LineAwesome::all_glyphs().collect()
72    }
73
74    fn codepoint(name: &str) -> Option<CodePoint> {
75        LineAwesome::codepoint(name)
76    }
77
78    fn parse_name(name: &str) -> Option<LineAwesomeIconName> {
79        LineAwesomeIconName::parse(name)
80    }
81
82    fn display_name(name: &LineAwesomeIconName) -> String {
83        name.css_class_name().to_string()
84    }
85
86    fn subtitle(_name: &LineAwesomeIconName) -> String {
87        "Line Awesome (Font Awesome 5 compatible)".to_string()
88    }
89
90    fn search(query: &str) -> Vec<String> {
91        #[cfg(feature = "metadata")]
92        {
93            LineAwesomeMetadata.search(query).iter().map(|s| s.to_string()).collect()
94        }
95        #[cfg(not(feature = "metadata"))]
96        {
97            let _ = query;
98            Vec::new()
99        }
100    }
101
102    fn keywords(name: &LineAwesomeIconName) -> Vec<String> {
103        #[cfg(feature = "metadata")]
104        {
105            LineAwesomeMetadata.keywords(name).iter().map(|k| k.as_str().to_string()).collect()
106        }
107        #[cfg(not(feature = "metadata"))]
108        {
109            let _ = name;
110            Vec::new()
111        }
112    }
113
114    /// Style categories (`regular`/`solid`/`brands`) derived from the
115    /// filename-pairing rule - they pre-fill the cheat sheet's
116    /// category dropdown.
117    fn categories(name: &LineAwesomeIconName) -> Vec<String> {
118        #[cfg(feature = "metadata")]
119        {
120            LineAwesomeMetadata.categories(name).iter().map(|c| c.as_str().to_string()).collect()
121        }
122        #[cfg(not(feature = "metadata"))]
123        {
124            let _ = name;
125            Vec::new()
126        }
127    }
128
129    /// Style-group dropdown (`regular`/`solid`/`brands`) plus
130    /// regular/solid variant collapsing, backed by
131    /// `LineAwesomeMetadata`'s `GlyphOrdering` implementation.
132    fn ordering() -> Option<&'static dyn fonts_rs_model::GlyphOrdering> {
133        #[cfg(feature = "metadata")]
134        {
135            Some(&LineAwesomeMetadata)
136        }
137        #[cfg(not(feature = "metadata"))]
138        {
139            None
140        }
141    }
142
143    /// Resolves vendored Font Awesome aliases to their canonical
144    /// Line Awesome names (e.g. `la-address-book-o-symbolic` ->
145    /// `la-address-book-symbolic`).
146    fn alias_display(name: &LineAwesomeIconName) -> Option<String> {
147        #[cfg(feature = "metadata")]
148        {
149            let canonical = LineAwesomeMetadata.resolve_alias(name.as_ref())?;
150            Some(
151                LineAwesomeIconName::parse(canonical)
152                    .map(|n| n.css_class_name().to_string())
153                    .unwrap_or_else(|| canonical.to_string()),
154            )
155        }
156        #[cfg(not(feature = "metadata"))]
157        {
158            let _ = name;
159            None
160        }
161    }
162
163    /// Render cells via the GtkIconTheme: a text label would render the
164    /// PUA char via font fallback - the bundled SVGs are authoritative.
165    /// The `la-icon` CSS class remains as the font path for detail
166    /// previews.
167    fn glyph_icon_name(name: &LineAwesomeIconName) -> Option<String> {
168        Some(name.as_ref().to_string())
169    }
170
171    fn glyph_svg(name: &LineAwesomeIconName) -> Option<String> {
172        LineAwesome::glyph_svg(name)
173    }
174}
