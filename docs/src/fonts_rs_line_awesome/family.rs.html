1//! Runtime font family API for Line Awesome.
2//!
3//! Implements the family facade via `impl_font_family!` with the
4//! generated glyph maps, the typed icon name, and the capabilities
5//! Line Awesome supports: `icon_theme` (GResource SVGs) and `html`
6//! (TTF via font.gresource / @font-face - the `html` capability
7//! already includes `HtmlSvgExport` for inline-SVG export).
8//!
9//! `pango` is deliberately *not* a declared capability: the macro emits
10//! the single-TTF `install_pango` default, but Line Awesome ships three
11//! faces - a manual [`PangoFont`](fonts_rs_model::PangoFont) impl below
12//! registers all of them. `SoftwareRender` (per-style `ab_glyph` faces)
13//! is implemented here as well.
14
15use crate::codepoint_map;
16use crate::definition::GLYPH_PREFIX;
17use crate::definition::LineAwesomeConfig;
18use crate::definition::TTF_FAMILY_NAME;
19use crate::naming::LineAwesome;
20use crate::naming::LineAwesomeIconName;
21use fonts_rs_model::FontVariant;

23/// Shipped TTF faces: one Font Awesome 5 style per file.
24const FONT_VARIANTS: &[FontVariant] = &[
25    FontVariant::file("regular", "la-regular-400"),
26    FontVariant::file("solid", "la-solid-900"),
27    FontVariant::file("brands", "la-brands-400"),
28];
29
30fonts_rs_generator::impl_font_family!(
31    LineAwesome,
32    LineAwesomeConfig,
33    glyphs = codepoint_map::GLYPHS,
34    reverse = codepoint_map::REVERSE_GLYPHS,
35    name = LineAwesomeIconName,
36    css_font_family = TTF_FAMILY_NAME,
37    glyph_prefix = GLYPH_PREFIX,
38    gresource_prefix = LineAwesomeConfig::GRESOURCE_PREFIX,
39    variants = FONT_VARIANTS,
40    capabilities = [icon_theme, html],
41);

43#[cfg(feature = "gtk")]
44impl fonts_rs_model::PangoFont for LineAwesome {
45    fn install_pango(font_map: &fonts_rs_model::pango::FontMap) -> Result<(), fonts_rs_model::FontError> {
46        Self::install_pango_resources(font_map, crate::gtk::FONT_RESOURCE_PATHS, env!("CARGO_PKG_NAME"))
47    }
48}
49
50#[cfg(feature = "gtk")]
51impl LineAwesome {
52    /// Loads all three TTFs (regular, solid, brands) into `font_map`.
53    ///
54    /// Forwards to [`PangoFont::install_pango`](fonts_rs_model::PangoFont::install_pango).
55    pub fn install_pango(font_map: &fonts_rs_model::pango::FontMap) -> Result<(), fonts_rs_model::FontError> {
56        <Self as fonts_rs_model::PangoFont>::install_pango(font_map)
57    }
58}

60#[cfg(feature = "render")]
61impl fonts_rs_model::SoftwareRender for LineAwesome {
62    type Variant = crate::style::LineAwesomeStyle;
63
64    fn ab_glyph(style: Self::Variant) -> Option<&'static ab_glyph::FontVec> {
65        crate::fonts::font(style)
66    }
67}

69#[cfg(feature = "render")]
70impl LineAwesome {
71    /// Returns the cached `ab_glyph` font for the given style.
72    ///
73    /// Forwards to [`SoftwareRender::ab_glyph`](fonts_rs_model::SoftwareRender::ab_glyph).
74    pub fn ab_glyph(style: crate::style::LineAwesomeStyle) -> Option<&'static ab_glyph::FontVec> {
75        <Self as fonts_rs_model::SoftwareRender>::ab_glyph(style)
76    }
77
78    /// Returns the cached `ab_glyph` font for the default (solid) style.
79    ///
80    /// Forwards to [`SoftwareRender::font`](fonts_rs_model::SoftwareRender::font).
81    pub fn font() -> Option<&'static ab_glyph::FontVec> {
82        <Self as fonts_rs_model::SoftwareRender>::font()
83    }
84}
