1//! Icon metadata for Line Awesome.
2//!
3//! Keywords are filename-derived (the kebab stem splits into word
4//! tokens), categories carry the Font Awesome 5 style (`solid`,
5//! `regular`, `brands`), and aliases merge the vendored FA4-compat
6//! `*-o` and `fa-*` rename table - all generated as `phf::Map`s by
7//! `build.rs` when the `metadata` feature is enabled.
8//!
9//! [`GlyphOrdering`] groups glyphs by their style category (the cheat
10//! sheet's group dropdown) and treats the solid face of a regular/
11//! solid codepoint pair as a variant of the regular base glyph.
12//!
13//! This module is only available when the `metadata` feature is enabled.
14

55/// Style groups in canonical order: regular, then solid, then brands.
56static GROUP_ORDER: [GroupLabel; 3] = [GroupLabel::new("regular"), GroupLabel::new("solid"), GroupLabel::new("brands")];
57
58/// Ordering tables derived once from the generated maps.
59///
60/// Line Awesome has no upstream ordering JSON, so the tables are
61/// computed lazily from `REVERSE_GLYPHS` (all exported names) and
62/// `CATEGORIES` (style per name) instead of being codegen'd.
63struct OrderingTables {
64    /// Alphabetical index per base glyph (paired solid variants are
65    /// excluded - they sort through their base via `base_of`).
66    order: HashMap<&'static str, u32>,
67    /// `la-{stem}-solid-symbolic` -> `la-{stem}-symbolic` for the
68    /// regular/solid pairs sharing one codepoint.
69    base_of: HashMap<&'static str, &'static str>,
70    /// `la-{stem}-symbolic` -> `[la-{stem}-solid-symbolic]` inverse.
71    alternates: HashMap<&'static str, &'static [&'static str]>,
72    /// Style groups with alphabetically sorted glyph names.
73    groups: [GlyphGroup; 3],
74}
75
76/// Returns the process-cached [`OrderingTables`].
77fn tables() -> &'static OrderingTables {
78    static TABLES: OnceLock<OrderingTables> = OnceLock::new();
79    TABLES.get_or_init(build_tables)
80}
81
82/// Builds the ordering tables from the generated glyph maps.
83///
84/// The small slices are intentionally leaked: [`GlyphOrdering`]
85/// returns `&'static` references, so the tables live for the process
86/// lifetime.
87fn build_tables() -> OrderingTables {
88    let names: Vec<&'static str> = REVERSE_GLYPHS.entries().map(|(name, _)| *name).collect();
89
90    // Strip the `la-` prefix and `-symbolic` suffix:
91    // `la-heart-solid-symbolic` -> `heart-solid`.
92    let stem_of = |name: &'static str| -> &'static str { name.strip_prefix("la-").and_then(|s| s.strip_suffix("-symbolic")).unwrap_or(name) };
93
94    // Regular/solid pairs: `la-{stem}-solid-symbolic` is a variant of
95    // `la-{stem}-symbolic` when the plain spelling exists.
96    let mut base_of: HashMap<&'static str, &'static str> = HashMap::new();
97    for name in &names {
98        if let Some(stem) = stem_of(name).strip_suffix("-solid") {
99            let plain = format!("la-{stem}-symbolic");
100            if let Some(&base) = names.iter().find(|candidate| **candidate == plain) {
101                base_of.insert(*name, base);
102            }
103        }
104    }
105
106    let mut alternates: HashMap<&'static str, Vec<&'static str>> = HashMap::new();
107    for (variant, base) in &base_of {
108        alternates.entry(*base).or_default().push(*variant);
109    }
110    let alternates: HashMap<&'static str, &'static [&'static str]> = alternates
111        .into_iter()
112        .map(|(base, mut v)| {
113            v.sort();
114            (base, &*Box::leak(v.into_boxed_slice()))
115        })
116        .collect();
117
118    // Alphabetical index over base glyphs only.
119    let mut bases: Vec<&'static str> = names.iter().copied().filter(|name| !base_of.contains_key(*name)).collect();
120    bases.sort();
121    let order: HashMap<&'static str, u32> = bases.iter().enumerate().map(|(index, name)| (*name, index as u32)).collect();
122
123    // Style groups: every glyph is filed under its style category,
124    // sorted alphabetically within the group.
125    let metadata = LineAwesomeMetadata;
126    let groups = GROUP_ORDER.map(|label| {
127        let mut glyphs: Vec<&'static str> = names
128            .iter()
129            .copied()
130            .filter(|name| metadata.categories(*name).iter().any(|c| c.as_str() == label.as_str()))
131            .collect();
132        glyphs.sort();
133        GlyphGroup::new(label, Box::leak(glyphs.into_boxed_slice()))
134    });
135
136    OrderingTables {
137        order,
138        base_of,
139        alternates,
140        groups,
141    }
142}

144impl GlyphOrdering for LineAwesomeMetadata {
145    fn order_index(&self, name: &str) -> Option<u32> {
146        tables().order.get(name).copied()
147    }
148
149    /// Style group of a glyph: `regular`, `solid` or `brands`.
150    fn group(&self, name: &str) -> Option<GroupLabel> {
151        self.categories(name).first().map(|c| GroupLabel::new(c.as_str()))
152    }
153
154    fn group_labels(&self) -> &'static [GroupLabel] {
155        &GROUP_ORDER
156    }
157
158    fn groups(&self) -> &'static [GlyphGroup] {
159        &tables().groups
160    }
161
162    /// The solid face of a regular/solid codepoint pair.
163    fn alternates(&self, name: &str) -> &'static [&'static str] {
164        tables().alternates.get(name).copied().unwrap_or_default()
165    }
166
167    /// `la-{stem}-solid-symbolic` -> `la-{stem}-symbolic` for paired
168    /// glyphs; solid-only and brand glyphs return `None`.
169    fn base_of(&self, name: &str) -> Option<&'static str> {
170        tables().base_of.get(name).copied()
171    }
172
173    /// All styles are first-class browse content - a whole style is
174    /// not a hideable variant (unlike emoji skin tones). The pair
175    /// link still powers the detail sidebar's Variants section and
176    /// keeps solid glyphs sorted next to their regular base.
177    fn collapse_variants(&self) -> bool {
178        false
179    }
