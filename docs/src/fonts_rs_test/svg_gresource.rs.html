1//! SVG GResource example.
2//!
3//! Provides [`run_svg_gresource`] for loading individual SVG glyphs from
4//! GResource and displaying them as GDK textures in a GTK grid.

13use gtk4::Orientation;

15use gtk4::glib;

17use miette::IntoDiagnostic;
18use miette::Result;

22/// Runs a GTK app displaying SVG glyphs from GResource as textures.
23///
24/// Registers the font's GResource bundles via `F::ensure_registered`
25/// and displays the first few glyphs as `Picture` widgets loaded via
26/// `F::glyph_svg`.
27///
28/// # Type Parameters
29///
30/// - `F`: Font family implementing [`IconThemeFont`].
31///
32/// # Parameters
33///
34/// - `app_id`: Unique GTK application ID.
35/// - `window_title`: Window title text.
36/// - `glyphs`: Slice of `(`[`CodePoint`]`, glyph_name)` pairs to display.
37///   The name accepts `&str`, `String`, and the typed `F::Name`
38///   (`GlyphName<F>` / `IconName`) via `AsRef<str>`.
39///
40/// # Errors
41///
42/// Returns an error if GTK init or GResource registration fails.
43pub fn run_svg_gresource<F: IconThemeFont>(app_id: &str, window_title: &str, glyphs: &[(CodePoint, impl AsRef<str>)]) -> Result<glib::ExitCode> {
44    F::ensure_registered().into_diagnostic()?;
45
46    let glyphs: Vec<(CodePoint, String)> = glyphs.iter().map(|(ch, name)| (*ch, name.as_ref().to_string())).collect();
47
48    let window_title = window_title.to_string();
49
50    run_gtk_app(app_id, move |app| {
51        build_ui::<F>(app, &window_title, &glyphs);
52    })
53}
54
55fn build_ui<F: IconThemeFont>(app: &Application, window_title: &str, glyphs: &[(CodePoint, String)]) {
56    let window = ApplicationWindow::builder()
57        .application(app)
58        .title(window_title)
59        .default_width(600)
60        .default_height(400)
61        .build();
62
63    let main_box = Box::builder()
64        .orientation(Orientation::Vertical)
65        .spacing(12)
66        .margin_top(24)
67        .margin_bottom(24)
68        .margin_start(24)
69        .margin_end(24)
70        .build();
71
72    let title = Label::builder()
73        .label("SVG Glyphs from GResource")
74        .css_classes(["title-2"])
75        .halign(Align::Center)
76        .build();
77    main_box.append(&title);
78
79    let hint = Label::builder()
80        .label("Each glyph is loaded as SVG from the GResource bundle and rendered as a texture.")
81        .halign(Align::Center)
82        .wrap(true)
83        .css_classes(["dim-label"])
84        .build();
85    main_box.append(&hint);
86
87    let grid = Box::builder().orientation(Orientation::Horizontal).spacing(12).halign(Align::Center).build();
88
89    for (_codepoint, name) in glyphs.iter().take(4) {
90        match F::glyph_svg(name) {
91            Some(svg) => match gtk4::gdk::Texture::from_bytes(&glib::Bytes::from_owned(svg.into_bytes())) {
92                Ok(texture) => {
93                    let picture = Picture::for_paintable(&texture);
94                    picture.set_halign(Align::Center);
95                    picture.set_valign(Align::Center);
96                    picture.set_size_request(64, 64);
97
98                    let vbox = Box::builder().orientation(Orientation::Vertical).spacing(4).halign(Align::Center).build();
99                    vbox.append(&picture);
100
101                    let caption = Label::builder().label(name).css_classes(["caption", "dim-label"]).halign(Align::Center).build();
102                    vbox.append(&caption);
103
104                    grid.append(&vbox);
105                }
106                Err(e) => {
107                    let label = Label::builder()
108                        .label(format!("{name}\nTexture error: {e}"))
109                        .css_classes(["caption", "error"])
110                        .halign(Align::Center)
111                        .build();
112                    grid.append(&label);
113                }
114            },
115            None => {
116                let label = Label::builder()
117                    .label(format!("{name}\nGlyph not found"))
118                    .css_classes(["caption", "error"])
119                    .halign(Align::Center)
120                    .build();
121                grid.append(&label);
122            }
123        }
124    }
125
126    main_box.append(&grid);
127    window.set_child(Some(&main_box));
128    window.present();
129}
