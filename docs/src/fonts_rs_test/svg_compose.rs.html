1//! Composed SVG text example.
2//!
3//! Provides [`run_composed_svg_text`] for composing multiple SVG glyph
4//! outlines into a single SVG document and displaying it as a GTK texture.
5
6use fonts_rs_model::IconThemeFont;

13use gtk4::Picture;
14use gtk4::glib;
15use gtk4::prelude::*;
16use miette::IntoDiagnostic;

21/// Runs a GTK app displaying composed SVG text from GResource glyphs.
22///
23/// Registers the font's GResource bundles via `F::ensure_registered`,
24/// composes the given text into a single SVG document via
25/// `F::compose_svg`, and displays the result as a `Picture` widget.
26///
27/// # Type Parameters
28///
29/// - `F`: Font family implementing [`IconThemeFont`].
30///
31/// # Parameters
32///
33/// - `app_id`: Unique GTK application ID.
34/// - `window_title`: Window title text.
35/// - `text`: Text to compose (each char is resolved via `F::glyph_name`).
36/// - `content_label`: Content type label (e.g. `"Emoji"`, `"EAN-Barcode"`).
37/// - `glyph_height`: Target height for each glyph in SVG user units.
38///
39/// # Errors
40///
41/// Returns an error if GTK init or GResource registration fails.
42pub fn run_composed_svg_text<F: IconThemeFont>(app_id: &str, window_title: &str, text: &str, content_label: &str, glyph_height: f64) -> Result<glib::ExitCode> {
43    F::ensure_registered().into_diagnostic()?;
44
45    let text = text.to_string();
46    let content_label = content_label.to_string();
47    let window_title = window_title.to_string();
48
49    run_gtk_app(app_id, move |app| {
50        build_ui::<F>(app, &window_title, &text, &content_label, glyph_height);
51    })
52}
53
54fn build_ui<F: IconThemeFont>(app: &Application, window_title: &str, text: &str, content_label: &str, glyph_height: f64) {
55    let window = ApplicationWindow::builder()
56        .application(app)
57        .title(window_title)
58        .default_width(800)
59        .default_height(300)
60        .build();
61
62    let main_box = Box::builder()
63        .orientation(Orientation::Vertical)
64        .spacing(12)
65        .margin_top(24)
66        .margin_bottom(24)
67        .margin_start(24)
68        .margin_end(24)
69        .build();
70
71    let title = Label::builder()
72        .label(format!("Composed SVG - glyphs assembled into one {content_label}"))
73        .css_classes(["title-2"])
74        .halign(Align::Center)
75        .build();
76    main_box.append(&title);
77
78    let svg = F::compose_svg(text, glyph_height);
79
80    let bytes = glib::Bytes::from_owned(svg.into_bytes());
81    match gtk4::gdk::Texture::from_bytes(&bytes) {
82        Ok(texture) => {
83            let picture = Picture::for_paintable(&texture);
84            picture.set_halign(Align::Center);
85            picture.set_valign(Align::Center);
86            main_box.append(&picture);
87        }
88        Err(e) => {
89            let label = Label::builder()
90                .label(format!("Texture error: {e}"))
91                .css_classes(["error"])
92                .halign(Align::Center)
93                .build();
94            main_box.append(&label);
95        }
96    }
97
98    window.set_child(Some(&main_box));
99    window.present();
100}
