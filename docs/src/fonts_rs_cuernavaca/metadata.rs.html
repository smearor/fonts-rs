1//! Glyph metadata: chess piece categories and style ordering.
2//!
3//! Provides category lookup for Cuernavaca glyphs: every piece glyph
4//! belongs to `king`, `queen`, `rook`, `bishop`, `knight`, or `pawn`.
5//! The upside-down pieces keep their piece category - `cuernavaca-a`
6//! (white upside-down bishop) is a `bishop`, `cuernavaca-d`/`f`
7//! (rook+knight fairy, used as upside-down rook) is a `rook`. Square
8//! and diagram-border glyphs carry no category.
9//!
10//! [`GlyphOrdering`] groups the pieces into four styles and links the
11//! non-base styles onto the `light normal` base:
12//!
13//! - `light normal` - white pieces, upright (`cuernavaca-k` .. `p`)
14//! - `dark normal` - black pieces, upright (`cuernavaca-l` .. `o`)
15//! - `light upside-down` - white pieces, rotated (`cuernavaca-a`/`d`/`g`)
16//! - `dark upside-down` - black pieces, rotated (`cuernavaca-f`/`h`/`s`)
17//!
18//! Semantically the rotated glyphs are fairy pieces in chess-problem
19//! convention - a bishop+knight (archbishop), a rook+knight
20//! (chancellor), and a generic fairy knight (nightrider) - which is
21//! why only bishop, knight, and rook exist upside-down: king, queen,
22//! and pawn carry no canonical fairy role.
23//!
24//! The link powers the detail sidebar's Variants section, sorts each
25//! variant next to its base, and collapses the variants behind the
26//! base in the browse grid.
27//!
28//! Categories are generated at build time from the glyph names -
29//! Cuernavaca ships no upstream keyword, category, or alias source,
30//! so the keyword and alias maps stay empty.
31//!
32//! This module is only available when the `metadata` feature is enabled.
33

51/// Metadata registry for the Chess Cuernavaca font family.
52///
53/// Implements [`GlyphMetadata`] using build-time generated static maps
54/// derived from the glyph names.
55pub struct CuernavacaMetadata;
56
57/// Piece table: glyph name stem -> style group, base stem, piece
58/// rank. Light-normal pieces are the bases (no `Some` base); dark and
59/// upside-down pieces link onto their light-normal counterpart.
60///
61/// Piece rank follows the canonical order king, queen, rook, bishop,
62/// knight, pawn.
63const PIECES: [(&str, &str, Option<&str>, u32); 18] = [
64    // light normal (white, upright) - the bases
65    ("k", "light normal", None, 0),
66    ("q", "light normal", None, 1),
67    ("r", "light normal", None, 2),
68    ("b", "light normal", None, 3),
69    ("n", "light normal", None, 4),
70    ("p", "light normal", None, 5),
71    // dark normal (black, upright)
72    ("l", "dark normal", Some("k"), 0),
73    ("w", "dark normal", Some("q"), 1),
74    ("t", "dark normal", Some("r"), 2),
75    ("v", "dark normal", Some("b"), 3),
76    ("m", "dark normal", Some("n"), 4),
77    ("o", "dark normal", Some("p"), 5),
78    // light upside-down (white, rotated)
79    ("a", "light upside-down", Some("b"), 3),
80    ("d", "light upside-down", Some("r"), 2),
81    ("g", "light upside-down", Some("n"), 4),
82    // dark upside-down (black, rotated)
83    ("f", "dark upside-down", Some("r"), 2),
84    ("h", "dark upside-down", Some("n"), 4),
85    ("s", "dark upside-down", Some("b"), 3),
86];
87
88/// Style group labels in canonical order.
89const STYLE_LABELS: [&str; 4] = ["light normal", "dark normal", "light upside-down", "dark upside-down"];
90
91/// Ordering tables derived once from the generated codemap.
92///
93/// Cuernavaca has no upstream ordering source, so the tables are
94/// computed lazily from `REVERSE_GLYPHS` (all exported names) plus
95/// the [`PIECES`] table instead of being codegen'd.
96struct OrderingTables {
97    /// Canonical index over base/standalone glyphs - variants sort
98    /// next to their base via `base_of`.
99    order: HashMap<&'static str, u32>,
100    /// Style label per glyph name, variants included.
101    group_of: HashMap<&'static str, GroupLabel>,
102    /// `light normal` base name for dark and upside-down pieces.
103    base_of: HashMap<&'static str, &'static str>,
104    /// Base name -> the other styles of that piece, in canonical
105    /// order (dark normal, light upside-down, dark upside-down).
106    alternates_of: HashMap<&'static str, &'static [&'static str]>,
107    /// One group per style with pieces in canonical order.
108    groups: &'static [GlyphGroup],
109}
110
111/// Returns the [`PIECES`] entry for a glyph name, or `None` for
112/// square and border glyphs.
113fn piece_info(name: &str) -> Option<&'static (&'static str, &'static str, Option<&'static str>, u32)> {
114    let stem = name.strip_prefix("cuernavaca-")?;
115    PIECES.iter().find(|(piece_stem, ..)| *piece_stem == stem)
116}
117
118/// Returns the process-cached [`OrderingTables`].
119fn tables() -> &'static OrderingTables {
120    static TABLES: OnceLock<OrderingTables> = OnceLock::new();
121    TABLES.get_or_init(build_tables)
122}
123
124/// Builds the ordering tables from the generated codemap.
125fn build_tables() -> OrderingTables {
126    let mut names: Vec<&'static str> = REVERSE_GLYPHS.entries().map(|(&name, _)| name).collect();
127    names.sort_unstable();
128
129    // Style variants: every dark/upside-down piece links onto its
130    // light-normal base (dropped if that base is not exported).
131    let mut base_of = HashMap::new();
132    let mut alternates_of: HashMap<&'static str, Vec<&'static str>> = HashMap::new();
133    let mut group_of = HashMap::with_capacity(names.len());
134    let mut info_of: HashMap<&'static str, (usize, u32)> = HashMap::new();
135    for &name in &names {
136        let Some(&(_, group, base, piece_rank)) = piece_info(name) else { continue };
137        let style_rank = STYLE_LABELS.iter().position(|label| *label == group).unwrap_or(usize::MAX);
138        info_of.insert(name, (style_rank, piece_rank));
139        group_of.insert(name, GroupLabel::new(group));
140        let Some(base_stem) = base else { continue };
141        let Some(&base_name) = names.iter().find(|n| n.strip_prefix("cuernavaca-") == Some(base_stem)) else {
142            continue;
143        };
144        base_of.insert(name, base_name);
145        alternates_of.entry(base_name).or_default().push(name);
146    }
147    // Canonical variant order follows STYLE_LABELS (dark normal,
148    // light upside-down, dark upside-down), not name order.
149    for alternates in alternates_of.values_mut() {
150        alternates.sort_by_key(|name| info_of.get(name).map(|(style_rank, _)| *style_rank).unwrap_or(usize::MAX));
151    }
152
153    // Canonical order over bases and non-piece glyphs: pieces first
154    // (king, queen, rook, bishop, knight, pawn), then the remaining
155    // square and border glyphs by name.
156    let mut bases: Vec<&'static str> = names.iter().copied().filter(|name| !base_of.contains_key(name)).collect();
157    bases.sort_by(|a, b| {
158        let rank = |name: &&'static str| info_of.get(name).map(|(_, piece_rank)| *piece_rank).unwrap_or(u32::MAX);
159        rank(a).cmp(&rank(b)).then(a.cmp(b))
160    });
161    let order: HashMap<&'static str, u32> = bases.iter().enumerate().map(|(index, &name)| (name, index as u32)).collect();
162
163    // Picker layout per style group, pieces in canonical order.
164    let mut groups_vec = Vec::with_capacity(STYLE_LABELS.len());
165    for label in STYLE_LABELS {
166        let mut members: Vec<&'static str> = group_of.iter().filter(|(_, group)| group.as_str() == label).map(|(&name, _)| name).collect();
167        members.sort_by_key(|name| info_of.get(name).map(|(_, piece_rank)| *piece_rank).unwrap_or(u32::MAX));
168        if !members.is_empty() {
169            groups_vec.push(GlyphGroup::new(GroupLabel::new(label), Box::leak(members.into_boxed_slice())));
170        }
171    }
172
173    let alternates_of: HashMap<&'static str, &'static [&'static str]> = alternates_of
174        .into_iter()
175        .map(|(base, alts)| (base, Box::leak(alts.into_boxed_slice()) as &'static [&'static str]))
176        .collect();
177
178    OrderingTables {
179        order,
180        group_of,
181        base_of,
182        alternates_of,
183        groups: Box::leak(groups_vec.into_boxed_slice()),
184    }
185}

187impl GlyphMetadata for CuernavacaMetadata {
188    fn keyword_map(&self) -> &GlyphKeywordMap {
189        &KEYWORDS
190    }
191
192    fn category_map(&self) -> &GlyphCategoryMap {
193        &CATEGORIES
194    }
195}
196
197impl GlyphOrdering for CuernavacaMetadata {
198    fn order_index(&self, name: &str) -> Option<u32> {
199        tables().order.get(name).copied()
200    }
201
202    /// Style of a piece glyph: `cuernavaca-l` -> "dark normal",
203    /// `cuernavaca-a` -> "light upside-down".
204    fn group(&self, name: &str) -> Option<GroupLabel> {
205        tables().group_of.get(name).copied()
206    }
207
208    fn group_labels(&self) -> &'static [GroupLabel] {
209        static LABELS: &[GroupLabel] = &[
210            GroupLabel::new("light normal"),
211            GroupLabel::new("dark normal"),
212            GroupLabel::new("light upside-down"),
213            GroupLabel::new("dark upside-down"),
214        ];
215        LABELS
216    }
217
218    fn groups(&self) -> &'static [GlyphGroup] {
219        tables().groups
220    }
221
222    /// The `light normal` piece a dark or upside-down style collapses
223    /// onto: `cuernavaca-l` -> `cuernavaca-k`, `cuernavaca-a` ->
224    /// `cuernavaca-b`.
225    fn base_of(&self, name: &str) -> Option<&'static str> {
226        tables().base_of.get(name).copied()
227    }
228
229    /// The other styles of a `light normal` base piece: `cuernavaca-r`
230    /// -> `cuernavaca-t`, `cuernavaca-d`, `cuernavaca-f`.
231    fn alternates(&self, name: &str) -> &'static [&'static str] {
232        tables().alternates_of.get(name).copied().unwrap_or(&[])
233    }
234}
