1//! Dicefont dice face font integration for GTK 4.
2//!
3//! Provides glyph name resolution, GResource registration, and codepoint
4//! lookup for the Dicefont font family — an icon font with die face
5//! glyphs for all standard dice (D2, D4, D6, D8, D10, D12, D20) plus
6//! dot-d6 variants.
7//!
8//! ## Quick Start
9//!
10//! ```no_run
11//! # #[cfg(feature = "gtk")]
12//! use fonts_rs_dicefont::register_glyphs;
13//!
14//! # #[cfg(feature = "gtk")]
15//! // Call once at startup before using any glyphs
16//! if let Err(e) = fonts_rs_dicefont::register_glyphs() {
17//!     eprintln!("Failed to register glyphs: {e}");
18//! }
19//! ```
20//!
21//! Then resolve glyph names to Unicode codepoints:
22//!
23//! ```no_run
24//! use fonts_rs_dicefont::naming::DicefontName;
25//! use fonts_rs_dicefont::GlyphNameExt;
26//!
27//! let name = DicefontName::new("d6-1".to_string());
28//! let codepoint = name.codepoint();
29//! ```

44/// Type alias for this font family's glyph name type.
45///
46/// Prefer [`DicefontName`] from the [`naming`] module — this alias

48pub type FamilyName = DicefontName;

50/// Extension trait adding codepoint lookup to `GlyphName<Dicefont>`.
51///
52/// Accepts either a base glyph name (e.g. `"d6-1"`) or a fully-qualified
53/// name (e.g. `"dicefont-d6-1"`). When a base name is given, the
54/// [`GLYPH_PREFIX`](variant::GLYPH_PREFIX) is prepended automatically.
55pub trait GlyphNameExt {
56    /// Resolves this glyph name to its Unicode [`CodePoint`].
57    ///
58    /// Returns `None` if the name is not found in the codepoint map.
59    fn codepoint(&self) -> Option<CodePoint>;
60}
61
62impl GlyphNameExt for DicefontName {
63    fn codepoint(&self) -> Option<CodePoint> {
64        let key = self.as_ref();
65        if let Some(ch) = codepoint_map::REVERSE_GLYPHS.get(key).copied() {
66            return Some(CodePoint::from(ch));
67        }
68        let full = format!("{}-{}", variant::GLYPH_PREFIX, key);
69        codepoint_map::REVERSE_GLYPHS.get(&full).copied().map(CodePoint::from)
70    }
71}

73/// Registers embedded Dicefont glyphs as a GResource.
74///
75/// Must be called once before using any glyphs through GTK/GIO APIs.
76/// After registration, glyphs are loadable via `Gtk::Image::from_resource`
77/// using the resource path `{GRESOURCE_PREFIX}/scalable/glyphs/{name}.svg`.
78///
79/// # Errors
80///
81/// Returns a [`gio::glib::Error`] if resource registration fails.
82#[cfg(feature = "gtk")]
83pub fn register_glyphs() -> Result<(), gio::glib::Error> {
84    gio::resources_register_include!("icons.gresource")?;
85    Ok(())
86}
87
88/// Returns an iterator over all known glyphs, yielding
89/// `(char, &'static str)` pairs of codepoint and glyph name.
90pub fn all_glyphs() -> impl Iterator<Item = (char, &'static str)> {
91    codepoint_map::GLYPHS.entries().map(|(c, name)| (*c, *name))
92}

94#[cfg(test)]
95mod tests {
96    use super::*;
97
98    #[test]
99    fn all_glyphs_is_not_empty() {
100        let count = all_glyphs().count();
101        assert!(count > 0, "Dicefont should export at least one glyph");
102    }
103
104    #[test]
105    fn all_glyphs_names_start_with_dicefont() {
106        for (_, name) in all_glyphs() {
107            assert!(name.starts_with("dicefont-"), "glyph name '{name}' should start with 'dicefont-'");
108        }
109    }
110
111    #[test]
112    fn all_glyphs_names_are_lowercase() {
113        for (_, name) in all_glyphs() {
114            assert_eq!(*name, name.to_lowercase(), "glyph name '{name}' should be lowercase");
115        }
116    }
117
118    #[test]
119    fn all_glyphs_names_contain_no_underscores() {
120        for (_, name) in all_glyphs() {
121            assert!(!name.contains('_'), "glyph name '{name}' should not contain underscores");
122        }
123    }
124
125    #[test]
126    fn all_glyphs_codepoints_are_unique() {
127        let codepoints: Vec<char> = all_glyphs().map(|(c, _)| c).collect();
128        let unique: std::collections::HashSet<char> = codepoints.iter().copied().collect();
129        assert_eq!(codepoints.len(), unique.len(), "all codepoints should be unique");
130    }
131
132    #[test]
133    fn all_glyphs_codepoints_in_pua() {
134        for (c, _) in all_glyphs() {
135            let cp = c as u32;
136            assert!((0xF000..=0xFFFF).contains(&cp), "codepoint U+{cp:04X} should be in the Private Use Area");
137        }
138    }
