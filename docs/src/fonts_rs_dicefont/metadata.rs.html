1//! Glyph metadata: die-type categories and d6 style ordering.
2//!
3//! Provides category lookup for Dicefont glyphs, grouping each die
4//! face by its die type: `dicefont-d20-4` -> "d20",
5//! `dicefont-dot-d6-2` -> "d6". The dotted d6 variants keep their
6//! "d6" category - they are six-sided dice. The Fudge dice
7//! (`dicefont-f`, `dicefont-f-*`) land in "fudge".
8//!
9//! Categories are generated at build time from the glyph names -
10//! Dicefont ships no upstream keyword, category, or alias source, so
11//! the keyword and alias maps stay empty.
12//!
13//! [`GlyphOrdering`] treats `dicefont-solid-small-dot-d6-{face}` as
14//! the base d6 style and links the other three d6 styles
15//! (`d6-{face}`, `dot-d6-{face}`, `small-dot-d6-{face}`) onto it -
16//! the link powers the detail sidebar's Variants section, sorts each
17//! variant next to its base, and collapses the variants behind the
18//! base in the browse grid.
19//!
20//! This module is only available when the `metadata` feature is enabled.
21

55/// Name-stem prefixes of the three non-base d6 styles, in canonical
56/// variant order for the sidebar's Variants section.
57const D6_VARIANT_STEMS: [&str; 3] = ["d6-", "dot-d6-", "small-dot-d6-"];
58
59/// Name-stem prefix of the base d6 style.
60const D6_BASE_STEM: &str = "solid-small-dot-d6-";
61
62/// Die-type name segments (mirrors `DICE_CATEGORIES` in `build.rs`;
63/// `f` renders as the "fudge" label).
64const DIE_SEGMENTS: [&str; 8] = ["d2", "d4", "d6", "d8", "d10", "d12", "d20", "f"];
65
66/// Group labels in canonical dice order.
67const DIE_LABELS: [&str; 8] = ["d2", "d4", "d6", "d8", "d10", "d12", "d20", "fudge"];
68
69/// Ordering tables derived once from the generated codemap.
70///
71/// Dicefont has no upstream ordering source, so the tables are
72/// computed lazily from `REVERSE_GLYPHS` (all exported names) plus
73/// the d6 style stems instead of being codegen'd.
74struct OrderingTables {
75    /// Canonical index over base/standalone glyphs - variants sort
76    /// next to their base via `base_of`.
77    order: HashMap<&'static str, u32>,
78    /// Die-type label per glyph name, variants included.
79    group_of: HashMap<&'static str, GroupLabel>,
80    /// `solid-small-dot` base name for the other d6 styles.
81    base_of: HashMap<&'static str, &'static str>,
82    /// Base name -> the other three d6 styles, in canonical order.
83    alternates_of: HashMap<&'static str, &'static [&'static str]>,
84    /// One group per die type with variants inlined after their base.
85    groups: &'static [GlyphGroup],
86}
87
88/// Returns the die-type label for a glyph name, or `None`.
89fn die_label(name: &'static str) -> Option<&'static str> {
90    let stem = name.strip_prefix("dicefont-").unwrap_or(name);
91    let segment = stem.split('-').find(|s| DIE_SEGMENTS.contains(s))?;
92    Some(if segment == "f" { "fudge" } else { segment })
93}
94
95/// Returns the canonical rank of a glyph's die type, `usize::MAX`
96/// for unlabeled names.
97fn die_rank(name: &'static str) -> usize {
98    die_label(name)
99        .and_then(|label| DIE_LABELS.iter().position(|l| *l == label))
100        .unwrap_or(usize::MAX)
101}
102
103/// Returns the face number suffix of a glyph name, `0` for names
104/// without one (bare die glyphs like `dicefont-d6`).
105fn face_number(name: &str) -> u32 {
106    name.rsplit('-').next().and_then(|s| s.parse().ok()).unwrap_or(0)
107}
108
109/// Returns the process-cached [`OrderingTables`].
110fn tables() -> &'static OrderingTables {
111    static TABLES: OnceLock<OrderingTables> = OnceLock::new();
112    TABLES.get_or_init(build_tables)
113}
114
115/// Builds the ordering tables from the generated codemap.
116fn build_tables() -> OrderingTables {
117    let mut names: Vec<&'static str> = REVERSE_GLYPHS.entries().map(|(&name, _)| name).collect();
118    names.sort_unstable();
119
120    // d6 style variants: `dicefont-{style}-d6-{face}` links onto the
121    // `dicefont-solid-small-dot-d6-{face}` base. Bare names without a
122    // face suffix (dicefont-d6, dicefont-dot-d6) have no base glyph -
123    // they stay standalone.
124    let mut base_of = HashMap::new();
125    let mut alternates_of: HashMap<&'static str, Vec<&'static str>> = HashMap::new();
126    for &name in &names {
127        let Some(stem) = name.strip_prefix("dicefont-") else { continue };
128        let Some(face) = D6_VARIANT_STEMS.iter().find_map(|prefix| stem.strip_prefix(prefix)) else {
129            continue;
130        };
131        let base: &'static str = Box::leak(format!("dicefont-{D6_BASE_STEM}{face}").into_boxed_str());
132        if REVERSE_GLYPHS.get(base).is_some() {
133            base_of.insert(name, base);
134            alternates_of.entry(base).or_default().push(name);
135        }
136    }
137    // Canonical variant order follows D6_VARIANT_STEMS, not the
138    // alphabetical name order produced above.
139    for alternates in alternates_of.values_mut() {
140        alternates.sort_by_key(|name| {
141            D6_VARIANT_STEMS
142                .iter()
143                .position(|prefix| name.strip_prefix("dicefont-").is_some_and(|stem| stem.starts_with(prefix)))
144                .unwrap_or(D6_VARIANT_STEMS.len())
145        });
146    }
147
148    // Die-type label per glyph name.
149    let mut group_of = HashMap::with_capacity(names.len());
150    for &name in &names {
151        if let Some(label) = die_label(name) {
152            group_of.insert(name, GroupLabel::new(label));
153        }
154    }
155
156    // Canonical order over bases and standalone glyphs: dice in
157    // pip-count order (d2..d20, fudge last), bare die names first,
158    // then faces ascending.
159    let mut bases: Vec<&'static str> = names.iter().copied().filter(|name| !base_of.contains_key(name)).collect();
160    bases.sort_by(|a, b| die_rank(a).cmp(&die_rank(b)).then(face_number(a).cmp(&face_number(b))).then(a.cmp(b)));
161    let order: HashMap<&'static str, u32> = bases.iter().enumerate().map(|(index, &name)| (name, index as u32)).collect();
162
163    // Picker layout per die group, variants inlined after their base.
164    let mut groups_vec = Vec::with_capacity(DIE_LABELS.len());
165    for label in DIE_LABELS {
166        let mut group_names: Vec<&'static str> = Vec::new();
167        for &base in &bases {
168            if group_of.get(&base) != Some(&GroupLabel::new(label)) {
169                continue;
170            }
171            group_names.push(base);
172            if let Some(alternates) = alternates_of.get(&base) {
173                group_names.extend(alternates.iter().copied());
174            }
175        }
176        if !group_names.is_empty() {
177            groups_vec.push(GlyphGroup::new(GroupLabel::new(label), Box::leak(group_names.into_boxed_slice())));
178        }
179    }
180
181    let alternates_of: HashMap<&'static str, &'static [&'static str]> = alternates_of
182        .into_iter()
183        .map(|(base, alts)| (base, Box::leak(alts.into_boxed_slice()) as &'static [&'static str]))
184        .collect();
185
186    OrderingTables {
187        order,
188        group_of,
189        base_of,
190        alternates_of,
191        groups: Box::leak(groups_vec.into_boxed_slice()),
192    }
193}

195impl GlyphOrdering for DicefontMetadata {
196    fn order_index(&self, name: &str) -> Option<u32> {
197        tables().order.get(name).copied()
198    }
199
200    /// Die type of a glyph: `dicefont-dot-d6-3` -> "d6",
201    /// `dicefont-f-minus` -> "fudge".
202    fn group(&self, name: &str) -> Option<GroupLabel> {
203        tables().group_of.get(name).copied()
204    }
205
206    fn group_labels(&self) -> &'static [GroupLabel] {
207        static LABELS: &[GroupLabel] = &[
208            GroupLabel::new("d2"),
209            GroupLabel::new("d4"),
210            GroupLabel::new("d6"),
211            GroupLabel::new("d8"),
212            GroupLabel::new("d10"),
213            GroupLabel::new("d12"),
214            GroupLabel::new("d20"),
215            GroupLabel::new("fudge"),
216        ];
217        LABELS
218    }
219
220    fn groups(&self) -> &'static [GlyphGroup] {
221        tables().groups
222    }
223
224    /// The `solid-small-dot` face a d6 style variant collapses onto:
225    /// `dicefont-d6-1` -> `dicefont-solid-small-dot-d6-1`.
226    fn base_of(&self, name: &str) -> Option<&'static str> {
227        tables().base_of.get(name).copied()
228    }
229
230    /// The three other d6 styles of a `solid-small-dot` base face.
231    fn alternates(&self, name: &str) -> &'static [&'static str] {
232        tables().alternates_of.get(name).copied().unwrap_or(&[])
233    }
234}
