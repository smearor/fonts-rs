1//! Heroicons Cheat Sheet application.
2//!
3//! Browse all available Heroicons icons with a search filter.
4//! Displays each icon with its Unicode codepoint and icon name.
5//! Click an icon to open a detail sidebar.
6//!
7//! The UI lives in `fonts-rs-cheat-sheet`; this binary only supplies
8//! the Heroicons hooks via [`CheatSheetFamily`]. With the `metadata`
9//! feature (default), search also covers the heroicons.com keywords -
10//! e.g. "stars" finds `heroicons-sparkles-symbolic` - and Heroicons v1
11//! aliases resolve to their v2 names.
12//!
13//! Heroicons is a fontless icon set: grid cells render via the
14//! `GtkIconTheme` (`glyph_icon_name`), not as font text.
15//!
16//! Run with:
17//! ```sh
18//! cargo run -p fonts-rs-heroicons-cheat-sheet
19//! ```
20
21use fonts_rs_cheat_sheet::CheatSheetFamily;
22use fonts_rs_cheat_sheet::FamilyKind;
23use fonts_rs_heroicons::Heroicons;
24use fonts_rs_heroicons::HeroiconsIconName;
25#[cfg(feature = "metadata")]
26use fonts_rs_heroicons::metadata::HeroiconsMetadata;
27use fonts_rs_model::CodePoint;
28#[cfg(feature = "metadata")]
29use fonts_rs_model::GlyphMetadata;
30use miette::IntoDiagnostic;
31use miette::Result;
32
33/// Heroicons hooks for the generic cheat sheet.
34pub struct HeroiconsSheet;
35
36impl CheatSheetFamily for HeroiconsSheet {
37    const TITLE: &'static str = "Heroicons - Cheat Sheet";
38    const NOUN: &'static str = "icons";
39    const APP_ID: &'static str = "io.smearor.fonts_rs.heroicons_cheat_sheet";
40    const SEARCH_PLACEHOLDER: &'static str = "Search by name, keyword, or codepoint (e.g. 'star', 'glitter', 'EA01')...";
41    const NAME_FIELD_LABEL: &'static str = "Icon Name";
42    const FAMILY_NAME: &'static str = "Heroicons";
43    const FAMILY_KIND: FamilyKind = FamilyKind::IconSet;
44    const LICENSE: &'static str = "MIT";
45
46    type Name = HeroiconsIconName;
47
48    fn init() -> Result<()> {
49        Heroicons::install().into_diagnostic()?;
50        // Grid cells render via Image::from_icon_name - the icon theme
51        // path must be added to the display's GtkIconTheme. Runs after
52        // gtk4::init(), so the default display is already open. No
53        // install_pango/icon_css: Heroicons ships no TTF.
54        let display = gtk4::gdk::Display::default().ok_or_else(|| miette::miette!("no default GDK display"))?;
55        Heroicons::install_icon_theme(&display).into_diagnostic()
56    }
57
58    fn all_glyphs() -> Vec<(CodePoint, HeroiconsIconName)> {
59        Heroicons::all_glyphs().collect()
60    }
61
62    fn codepoint(name: &str) -> Option<CodePoint> {
63        Heroicons::codepoint(name)
64    }
65
66    fn parse_name(name: &str) -> Option<HeroiconsIconName> {
67        HeroiconsIconName::parse(name)
68    }
69
70    fn display_name(name: &HeroiconsIconName) -> String {
71        name.css_class_name().to_string()
72    }
73
74    fn subtitle(_name: &HeroiconsIconName) -> String {
75        "Heroicons".to_string()
76    }
77
78    fn search(query: &str) -> Vec<String> {
79        #[cfg(feature = "metadata")]
80        {
81            HeroiconsMetadata.search(query).iter().map(|s| s.to_string()).collect()
82        }
83        #[cfg(not(feature = "metadata"))]
84        {
85            let _ = query;
86            Vec::new()
87        }
88    }
89
90    fn keywords(name: &HeroiconsIconName) -> Vec<String> {
91        #[cfg(feature = "metadata")]
92        {
93            HeroiconsMetadata.keywords(name).iter().map(|k| k.as_str().to_string()).collect()
94        }
95        #[cfg(not(feature = "metadata"))]
96        {
97            let _ = name;
98            Vec::new()
99        }
100    }
101
102    fn categories(name: &HeroiconsIconName) -> Vec<String> {
103        #[cfg(feature = "metadata")]
104        {
105            HeroiconsMetadata.categories(name).iter().map(|c| c.as_str().to_string()).collect()
106        }
107        #[cfg(not(feature = "metadata"))]
108        {
109            let _ = name;
110            Vec::new()
111        }
112    }
113
114    /// Style groups (outline/solid/mini/micro) and outline-base
115    /// variant collapsing.
116    fn ordering() -> Option<&'static dyn fonts_rs_model::GlyphOrdering> {
117        #[cfg(feature = "metadata")]
118        {
119            Some(&HeroiconsMetadata)
120        }
121        #[cfg(not(feature = "metadata"))]
122        {
123            None
124        }
125    }
126
127    /// Resolves Heroicons v1 alias names to their v2 canonical names
128    /// (e.g. `heroicons-exclaimation-circle-symbolic` →
129    /// `heroicons-exclamation-circle-symbolic`).
130    fn alias_display(name: &HeroiconsIconName) -> Option<String> {
131        #[cfg(feature = "metadata")]
132        {
133            let canonical = HeroiconsMetadata.resolve_alias(name.as_ref())?;
134            Some(
135                HeroiconsIconName::parse(canonical)
136                    .map(|n| n.css_class_name().to_string())
137                    .unwrap_or_else(|| canonical.to_string()),
138            )
139        }
140        #[cfg(not(feature = "metadata"))]
141        {
142            let _ = name;
143            None
144        }
145    }
146
147    /// Render cells via the GtkIconTheme: Heroicons is fontless, so a
148    /// text label would show a PUA glyph only if some installed font
149    /// covers the codepoint - the bundled SVGs are authoritative.
150    fn glyph_icon_name(name: &HeroiconsIconName) -> Option<String> {
151        Some(name.as_ref().to_string())
152    }
153
154    fn glyph_svg(name: &HeroiconsIconName) -> Option<String> {
155        Heroicons::glyph_svg(name)
156    }
157}
