1//! Icon metadata: keywords and categories for search functionality.
2//!
3//! Provides keyword and category lookup for Lucide icons, enabling
4//! search beyond simple name matching - e.g. "font size" finds
5//! `lucide-a-arrow-down-symbolic`.
6//!
7//! Data is sourced from upstream per-icon JSON files (`icons/{name}.json`
8//! `tags`/`categories` fields), vendored merged as
9//! `resources/metadata/icons.json`. Lucide ships no aliases, so the
10//! alias map is empty.
11//!
12//! This module is only available when the `metadata` feature is enabled.
13
14use fonts_rs_model::GlyphCategory;
15use fonts_rs_model::GlyphCategoryMap;
16use fonts_rs_model::GlyphKeyword;
17use fonts_rs_model::GlyphKeywordMap;
18use fonts_rs_model::GlyphMetadata;
19
20// Include the build-time generated phf::Map constants.
21include!(concat!(env!("OUT_DIR"), "/keywords.rs"));
22include!(concat!(env!("OUT_DIR"), "/categories.rs"));
23
24/// Metadata registry for the Lucide Icons font family.
25///
26/// Implements [`GlyphMetadata`] using build-time generated static maps
27/// sourced from upstream `icons.json`.
28pub struct LucideMetadata;
29
30impl GlyphMetadata for LucideMetadata {
31    fn keyword_map(&self) -> &GlyphKeywordMap {
32        &KEYWORDS
33    }
34
35    fn category_map(&self) -> &GlyphCategoryMap {
36        &CATEGORIES
37    }
38}
