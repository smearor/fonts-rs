1//! DSEG7 Seven-Segment Cheat Sheet application.
2//!
3//! Browse all available DSEG7 seven-segment display glyphs with a
4//! search filter. Displays each glyph with its Unicode codepoint and
5//! glyph name. Search covers glyph names, hex codepoints, and
6//! characters. Click a glyph to open a detail sidebar.
7//!
8//! The UI lives in `fonts-rs-cheat-sheet`; this binary only supplies
9//! the SevenSegment hooks via [`CheatSheetFamily`].
10//!
11//! Run with:
12//! ```sh
13//! cargo run -p fonts-rs-seven-segment-cheat-sheet
14//! ```
15
16use fonts_rs_cheat_sheet::CheatSheetFamily;
17use fonts_rs_cheat_sheet::FamilyKind;
18use fonts_rs_model::CodePoint;
19use fonts_rs_seven_segment::SevenSegment;
20use fonts_rs_seven_segment::SevenSegmentName;
21use miette::IntoDiagnostic;
22use miette::Result;
23
24/// DSEG7 hooks for the generic cheat sheet.
25pub struct SevenSegmentSheet;
26
27impl CheatSheetFamily for SevenSegmentSheet {
28    const TITLE: &'static str = "DSEG7 Seven-Segment - Cheat Sheet";
29    const NOUN: &'static str = "glyphs";
30    const APP_ID: &'static str = "io.smearor.fonts_rs.seven_segment_cheat_sheet";
31    const SEARCH_PLACEHOLDER: &'static str = "Search by glyph name (e.g. 'dseg7-zero', 'dseg7-colon')...";
32    const GLYPH_CSS_CLASS: Option<&'static str> = Some("dseg7-icon");
33    const FAMILY_NAME: &'static str = "DSEG7";
34    const FAMILY_KIND: FamilyKind = FamilyKind::SpecialFont;
35    const LICENSE: &'static str = "OFL-1.1";
36
37    type Name = SevenSegmentName;
38
39    fn init() -> Result<()> {
40        SevenSegment::install().into_diagnostic()?;
41        // Register the bundled TTF with the global Pango font map -
42        // without this, `font-family: 'DSEG7 Classic'` falls back to the
43        // system font covering the ASCII codepoints.
44        SevenSegment::install_pango(&pangocairo::FontMap::default()).into_diagnostic()?;
45        fonts_rs_model::gtk::load_css(&fonts_rs_model::icon_css(&[("dseg7-icon", "DSEG7 Classic")]));
46        Ok(())
47    }
48
49    fn all_glyphs() -> Vec<(CodePoint, SevenSegmentName)> {
50        SevenSegment::all_glyphs().collect()
51    }
52
53    fn codepoint(name: &str) -> Option<CodePoint> {
54        SevenSegment::codepoint(name)
55    }
56
57    fn parse_name(name: &str) -> Option<SevenSegmentName> {
58        // Round-trip via the codepoint resolves shorthand input to the
59        // canonical glyph name and rejects unknown names.
60        SevenSegment::codepoint(name).and_then(SevenSegment::glyph_name)
61    }
62
63    fn display_name(name: &SevenSegmentName) -> String {
64        name.to_string()
65    }
66
67    fn subtitle(_name: &SevenSegmentName) -> String {
68        "DSEG7".to_string()
69    }
70
71    /// No metadata module - the generic supplement still matches
72    /// names, hex codepoints, and characters.
73    fn search(_query: &str) -> Vec<String> {
74        Vec::new()
75    }
76
77    fn keywords(_name: &SevenSegmentName) -> Vec<String> {
78        Vec::new()
79    }
80
81    fn categories(_name: &SevenSegmentName) -> Vec<String> {
82        Vec::new()
83    }
84
85    fn glyph_svg(name: &SevenSegmentName) -> Option<String> {
86        SevenSegment::glyph_svg(name)
87    }
88}
