1//! Nerd Font icon collection enumeration.
2//!
3//! Provides [`IconSet`] for prefix-based grouping of Nerd Font icons
4//! (Font Awesome, Material Design, Octicons, etc.).
5
6/// Nerd Font icon collections.
7///
8/// Each variant corresponds to a prefix convention used in Nerd Font
9/// icon names (e.g. `nf-fa-*` for Font Awesome, `nf-md-*` for Material
10/// Design). Use `IconName::icon_set` to determine the
11/// collection from a normalized icon name.
12#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
13pub enum IconSet {
14    /// Material Design icons (`nf-md-*`).
15    MaterialDesign,
16    /// Font Awesome icons (`nf-fa-*`).
17    FontAwesome,
18    /// Devicons (`nf-dev-*`).
19    Devicons,
20    /// Codicons (`nf-cod-*`).
21    Codicons,
22    /// GitHub Octicons (`nf-oct-*`).
23    Octicons,
24    /// Weather icons (`nf-weather-*`).
25    Weather,
26    /// Font Awesome Extended (`nf-fae-*`).
27    FontAwesomeExtended,
28    /// Seti-UI icons (`nf-seti-*`).
29    Seti,
30    /// Linux icons (`nf-linux-*`).
31    Linux,
32    /// Powerline Extra symbols (`nf-ple-*`).
33    PowerlineExtra,
34    /// Powerline symbols (`nf-pl-*`).
35    Powerline,
36    /// Custom icons (`nf-custom-*`).
37    Custom,
38    /// Extra icons (`nf-extra-*`).
39    Extra,
40    /// Pomicons (`nf-pom-*`).
41    Pomicons,
42    /// Alpha icons (`nf-alpha-*`).
43    Alpha,
44    /// IEC symbols (`nf-iec-*`).
45    Iec,
46    /// Checkbox icons (`nf-checkbox-*`).
47    Checkbox,
48    /// File icons (`nf-file-*`).
49    File,
50    /// Heart icons (`nf-heart-*`).
51    Heart,
52    /// Image icons (`nf-image-*`).
53    Image,
54    /// Indentation icons (`nf-indentation-*`).
55    Indentation,
56    /// Music icons (`nf-music-*`).
57    Music,
58    /// Near icons (`nf-near-*`).
59    Near,
60    /// Non-marking return symbols (`nf-nonmarkingreturn-*`).
61    NonMarkingReturn,
62    /// Exit icons (`nf-exit-*`).
63    Exit,
64    /// Icons that do not match any known prefix.
65    Other,
66}

68impl IconSet {
69    /// Detects the icon set from a Nerd Font icon name prefix.
70    ///
71    /// Performs a simple prefix match — zero allocation, O(1).
72    /// The compiler can optimize the match into a jump table.
73    pub fn detect(icon_name: &str) -> Self {
74        match icon_name {
75            n if n.starts_with("nf-md-") => IconSet::MaterialDesign,
76            n if n.starts_with("nf-fa-") => IconSet::FontAwesome,
77            n if n.starts_with("nf-dev-") => IconSet::Devicons,
78            n if n.starts_with("nf-cod-") => IconSet::Codicons,
79            n if n.starts_with("nf-oct-") => IconSet::Octicons,
80            n if n.starts_with("nf-weather-") => IconSet::Weather,
81            n if n.starts_with("nf-fae-") => IconSet::FontAwesomeExtended,
82            n if n.starts_with("nf-seti-") => IconSet::Seti,
83            n if n.starts_with("nf-linux-") => IconSet::Linux,
84            n if n.starts_with("nf-ple-") => IconSet::PowerlineExtra,
85            n if n.starts_with("nf-pl-") => IconSet::Powerline,
86            n if n.starts_with("nf-custom-") => IconSet::Custom,
87            n if n.starts_with("nf-extra-") => IconSet::Extra,
88            n if n.starts_with("nf-pom-") => IconSet::Pomicons,
89            n if n.starts_with("nf-alpha-") => IconSet::Alpha,
90            n if n.starts_with("nf-iec-") => IconSet::Iec,
91            n if n.starts_with("nf-checkbox-") => IconSet::Checkbox,
92            n if n.starts_with("nf-file-") => IconSet::File,
93            n if n.starts_with("nf-heart-") => IconSet::Heart,
94            n if n.starts_with("nf-image-") => IconSet::Image,
95            n if n.starts_with("nf-indentation-") => IconSet::Indentation,
96            n if n.starts_with("nf-music-") => IconSet::Music,
97            n if n.starts_with("nf-near-") => IconSet::Near,
98            n if n.starts_with("nf-nonmarkingreturn-") => IconSet::NonMarkingReturn,
99            n if n.starts_with("nf-exit-") => IconSet::Exit,
100            _ => IconSet::Other,
101        }
102    }
