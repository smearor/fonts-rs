1//! Noto Emoji font integration for GTK 4 and pixel-drawing.
2//!
3//! Provides glyph name resolution, GResource registration, and codepoint
4//! lookup for the Noto Emoji font family - Google's monochrome emoji font
5//! covering all Unicode emoji codepoints with semantic names from CLDR
6//! annotation data.
7//!
8//! ## Quick Start
9//!
10//! ```no_run
11//! # #[cfg(feature = "gtk")]
12//! use fonts_rs_noto_emoji::NotoEmoji;
13//!
14//! # #[cfg(feature = "gtk")]
15//! // Call once at startup before using any glyphs
16//! if let Err(e) = NotoEmoji::install() {
17//!     eprintln!("Failed to install font: {e}");
18//! }
19//! ```
20//!
21//! Then resolve glyph names to Unicode codepoints:
22//!
23//! ```no_run
24//! use fonts_rs_noto_emoji::NotoEmoji;
25//!
26//! let codepoint = NotoEmoji::codepoint("grinning-face");
27//! ```

61/// Returns all known Noto Emoji glyphs as `(CodePoint, glyph_name)` pairs,
62/// sorted alphabetically by glyph name.
63pub fn all_emoji() -> Vec<(CodePoint, String)> {
64    let mut emojis: Vec<(CodePoint, String)> = codepoint_map::GLYPHS
65        .entries()
66        .map(|(cp, name)| (CodePoint::from(*cp), name.to_string()))
67        .collect();
68    emojis.sort_by(|a, b| a.1.cmp(&b.1));
69    emojis
70}

72#[cfg(test)]
73mod tests {
74    use super::*;

76    /// `glyph_name` must resolve emoji codepoints to variant-prefixed
77    /// icon names, and `glyph_svg` must return the colored vector
78    /// artwork from the GResource - the chain behind `gtk_icon_theme`
79    /// and `svg_gresource`.
80    #[cfg(feature = "gtk")]
81    #[test]
82    fn glyph_svg_returns_colored_artwork() {
83        let name = NotoEmoji::glyph_name('\u{1F600}').expect("grinning face should resolve");
84        let svg = NotoEmoji::glyph_svg(&name).expect("glyph SVG should be in the GResource");
85        assert!(svg.contains("<svg"), "expected an SVG document, got: {}", &svg[..svg.len().min(200)]);
86        assert!(
87            svg.contains("stop-color") || svg.contains("fill:") || svg.contains("fill=\""),
88            "expected colored artwork in {name}"
89        );
90    }
91
92    /// `compose_svg` must produce a parseable document for glyphs whose
93    /// source SVGs carried Illustrator DOCTYPE entity namespaces
94    /// (`emoji_u1f680.svg` declares `xmlns:i="&ns_ai;"`). The export
95    /// strips them via `StripEntityNamespaces`; a regression would
96    /// surface as "namespace prefix" parse errors in GTK.
97    #[cfg(feature = "gtk")]
98    #[test]
99    fn compose_svg_has_no_unbound_prefixes() {
100        let svg = NotoEmoji::compose_svg("\u{1F680}", 200.0);
101        assert!(svg.contains("<svg"), "expected a composed SVG document");
102        assert!(!svg.contains("&ns_"), "entity references must not leak into the composed SVG");
103        assert!(!svg.contains("i:aipgf"), "Illustrator metadata must be stripped");
104    }
105
106    /// `render_text` must produce a non-empty, non-black buffer for the
107    /// bitmap variants: CBDT strikes are PNG blobs positioned by their
108    /// bottom bearing. Under `color-*` the COLRv1 font has empty `glyf`
109    /// outlines (ab_glyph renders nothing); under `mono` the coverage
110    /// is white-on-black by design - so this test only runs when the
111    /// active variant ships CBDT strikes (`noto_emoji_bitmap` cfg is
112    /// emitted by build.rs, which also covers the default variant).
113    #[cfg(all(feature = "gtk", feature = "render", noto_emoji_bitmap))]
114    #[test]
115    fn render_text_blits_bitmap_strikes() {
116        let font = NotoEmoji::font().expect("NOTO_EMOJI_FONT_PATH TTF should load via ab_glyph");
117        let (width, height, pixels) = fonts_rs_test::render_text(font, "\u{1F600}", 64.0);
118        assert!(width > 1 && height > 1, "rendered buffer should have real dimensions, got {width}x{height}");
119        let non_black = pixels.chunks_exact(4).filter(|px| px[0] > 0 || px[1] > 0 || px[2] > 0).count();
120        assert!(non_black > 0, "rendered buffer should contain non-black pixels");
121    }
122
123    /// `render_text` must produce white silhouettes for the `mono`
124    /// variant: `NotoEmoji-Regular` carries real `glyf` outlines, so
125    /// the outline path must light up pixels. Gated on the
126    /// build-time `noto_emoji_mono` cfg like `noto_emoji_bitmap` -
127    /// under `--all-features` the `mono` feature is set while the
128    /// active variant falls back to the COLRv1 default, which has
129    /// empty outlines.
130    #[cfg(all(feature = "gtk", feature = "render", noto_emoji_mono))]
131    #[test]
132    fn render_text_draws_mono_outlines() {
133        let font = NotoEmoji::font().expect("NOTO_EMOJI_FONT_PATH TTF should load via ab_glyph");
134        let (width, height, pixels) = fonts_rs_test::render_text(font, "\u{1F600}", 64.0);
135        let non_black = pixels.chunks_exact(4).filter(|px| px[0] > 0 || px[1] > 0 || px[2] > 0).count();
136        let max_v = pixels.chunks_exact(4).map(|px| px[0].max(px[1]).max(px[2])).max().unwrap_or(0);
137        eprintln!("mono render: {width}x{height}, non_black={non_black}, max_value={max_v}");
138        assert!(width > 1 && height > 1, "rendered buffer should have real dimensions, got {width}x{height}");
139        assert!(non_black > 0, "mono outlines should produce non-black pixels");
140    }
