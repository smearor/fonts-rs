1//! Emoji metadata: keywords and categories for search functionality.
2//!
3//! Provides keyword and category lookup for Noto Emoji glyphs, enabling
4//! search and filtering beyond simple name matching.
5//!
6//! Keywords are sourced from CLDR annotation data (`default` field).
7//! Categories are sourced from Unicode `emoji-test.txt` (`group` field).
8//!
9//! This module is only available when the `metadata` feature is enabled.
10
11use fonts_rs_model::GlyphCategory;
12use fonts_rs_model::GlyphCategoryMap;
13use fonts_rs_model::GlyphKeyword;
14use fonts_rs_model::GlyphKeywordMap;
15use fonts_rs_model::GlyphMetadata;
16
17// Include the build-time generated phf::Map constants.
18include!(concat!(env!("OUT_DIR"), "/keywords.rs"));
19include!(concat!(env!("OUT_DIR"), "/categories.rs"));
20
21/// Metadata registry for the Noto Emoji font family.
22///
23/// Implements [`GlyphMetadata`] using build-time generated static maps
24/// sourced from CLDR keyword annotations and Unicode emoji-test categories.
25pub struct NotoEmojiMetadata;
26
27impl GlyphMetadata for NotoEmojiMetadata {
28    fn keyword_map(&self) -> &GlyphKeywordMap {
29        &KEYWORDS
30    }
31
32    fn category_map(&self) -> &GlyphCategoryMap {
33        &CATEGORIES
34    }
35}

41    #[test]
