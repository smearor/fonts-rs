1//! Icon metadata: keywords, style categories, aliases and ordering.
2//!
3//! Provides keyword and alias lookup for Fluent UI icons, enabling
4//! search beyond simple name matching - e.g. "wand" finds
5//! `fluent-wand-symbolic` directly and via the iconify alias
6//! `magic-wand`.
7//!
8//! Keywords are filename-derived (kebab stems split into word tokens;
9//! adjacent single-char tokens merge so `i-o-t` yields `iot`).
10//! Aliases come from the vendored `resources/metadata/aliases.json`
11//! iconify rename table and only resolve to exported glyph names.
12//! Categories carry the style ("regular" / "filled"); Fluent ships
13//! no categories upstream.
14//!
15//! [`GlyphOrdering`] groups glyphs by style and pairs each
16//! `fluent-filled-{name}-symbolic` onto its regular base
17//! `fluent-{name}-symbolic` - the link powers the detail sidebar's
18//! Variants section and sorts each filled glyph next to its regular
19//! base (styles never collapse).
20//!
21//! This module is only available when the `metadata` feature is
22//! enabled.
23

68/// Ordering tables derived once from the generated codemap.
69///
70/// Fluent has no upstream ordering JSON, so the tables are computed
71/// lazily from `REVERSE_GLYPHS` (all exported names) plus the
72/// `fluent-filled-` style prefix instead of being codegen'd.
73struct OrderingTables {
74    /// Alphabetical index over the *base* glyphs (regular style) -
75    /// variants sort next to their base via `base_of`.
76    order: HashMap<&'static str, u32>,
77    /// Style label per glyph name.
78    group_of: HashMap<&'static str, GroupLabel>,
79    /// `Some(regular_name)` for filled glyphs whose pair exists.
80    base_of: HashMap<&'static str, &'static str>,
81    /// `regular_name -> [filled_name]` for every pair.
82    alternates_of: HashMap<&'static str, &'static [&'static str]>,
83    /// One group per style with alphabetically sorted glyph names;
84    /// `regular` first - it is the upstream default style.
85    groups: &'static [GlyphGroup],
86}
87
88/// Returns the process-cached [`OrderingTables`].
89fn tables() -> &'static OrderingTables {
90    static TABLES: OnceLock<OrderingTables> = OnceLock::new();
91    TABLES.get_or_init(build_tables)
92}
93
94/// Builds the ordering tables from the generated codemap.
95fn build_tables() -> OrderingTables {
96    let mut regular: Vec<&'static str> = Vec::new();
97    let mut filled: Vec<&'static str> = Vec::new();
98    for (&name, _) in REVERSE_GLYPHS.entries() {
99        if name.starts_with(FILLED_PREFIX) {
100            filled.push(name)
101        } else {
102            regular.push(name)
103        }
104    }
105    regular.sort_unstable();
106    filled.sort_unstable();
107
108    let order: HashMap<&'static str, u32> = regular.iter().enumerate().map(|(i, &name)| (name, i as u32)).collect();
109    let mut group_of = HashMap::with_capacity(regular.len() + filled.len());
110    for &name in &regular {
111        group_of.insert(name, GroupLabel::new("regular"));
112    }
113    for &name in &filled {
114        group_of.insert(name, GroupLabel::new("filled"));
115    }
116
117    let mut base_of = HashMap::with_capacity(filled.len());
118    let mut alternates_of = HashMap::with_capacity(filled.len());
119    for &name in &filled {
120        let stem = name.strip_prefix(FILLED_PREFIX).unwrap_or(name);
121        let base_name: &'static str = Box::leak(format!("fluent-{stem}").into_boxed_str());
122        if regular.contains(&base_name) {
123            base_of.insert(name, base_name);
124            alternates_of.insert(base_name, Box::leak(Box::new([name])) as &'static [&'static str]);
125        }
126    }
127
128    let groups: &'static [GlyphGroup] = Box::leak(
129        vec![
130            GlyphGroup::new(GroupLabel::new("regular"), Box::leak(regular.into_boxed_slice())),
131            GlyphGroup::new(GroupLabel::new("filled"), Box::leak(filled.into_boxed_slice())),
132        ]
133        .into_boxed_slice(),
134    );
135
136    OrderingTables {
137        order,
138        group_of,
139        base_of,
140        alternates_of,
141        groups,
142    }
143}

145impl GlyphOrdering for FluentMetadata {
146    fn order_index(&self, name: &str) -> Option<u32> {
147        tables().order.get(name).copied()
148    }
149
150    /// Style of a glyph: `fluent-filled-{name}-symbolic` -> "filled",
151    /// `fluent-{name}-symbolic` -> "regular".
152    fn group(&self, name: &str) -> Option<GroupLabel> {
153        tables().group_of.get(name).copied()
154    }
155
156    fn group_labels(&self) -> &'static [GroupLabel] {
157        static LABELS: &[GroupLabel] = &[GroupLabel::new("regular"), GroupLabel::new("filled")];
158        LABELS
159    }
160
161    fn groups(&self) -> &'static [GlyphGroup] {
162        tables().groups
163    }
164
165    /// The regular glyph a filled glyph variants:
166    /// `fluent-filled-{name}-symbolic` -> `fluent-{name}-symbolic`.
167    fn base_of(&self, name: &str) -> Option<&'static str> {
168        tables().base_of.get(name).copied()
169    }
170
171    /// The filled variant of a regular glyph, if it exists.
172    fn alternates(&self, name: &str) -> &'static [&'static str] {
173        tables().alternates_of.get(name).copied().unwrap_or(&[])
174    }
175
176    /// Both styles are first-class browse content - a whole style is
177    /// not a hideable variant (unlike emoji skin tones). The pair
178    /// link still powers the detail sidebar's Variants section and
179    /// keeps filled glyphs sorted next to their regular base.
180    fn collapse_variants(&self) -> bool {
181        false
182    }
183}

185#[cfg(test)]
186mod tests {
187    use super::*;
188
189    #[test]
190    fn ordering_group_regular() {
191        let group = FluentMetadata.group("fluent-add-symbolic");
192        assert_eq!(group.map(|g| g.as_str()), Some("regular"));
193    }
194
195    #[test]
196    fn ordering_group_filled() {
197        let group = FluentMetadata.group("fluent-filled-add-symbolic");
198        assert_eq!(group.map(|g| g.as_str()), Some("filled"));
199    }
200
201    #[test]
202    fn ordering_base_of_pairs_filled() {
203        assert_eq!(FluentMetadata.base_of("fluent-filled-add-symbolic"), Some("fluent-add-symbolic"));
204    }
205
206    #[test]
207    fn ordering_alternates_contains_filled() {
208        let alternates = FluentMetadata.alternates("fluent-add-symbolic");
209        assert!(alternates.contains(&"fluent-filled-add-symbolic"));
210    }
211
212    #[test]
213    fn ordering_group_labels_cover_styles() {
214        let labels = FluentMetadata.group_labels();
215        assert_eq!(labels.len(), 2);
216        assert_eq!(labels[0].as_str(), "regular");
217        assert_eq!(labels[1].as_str(), "filled");
218    }
219
220    #[test]
221    fn ordering_groups_partition_all_glyphs() {
222        let total: usize = FluentMetadata.groups().iter().map(|g| g.glyphs.len()).sum();
223        let all = REVERSE_GLYPHS.entries().count();
224        assert_eq!(total, all, "groups should partition every glyph");
225    }
