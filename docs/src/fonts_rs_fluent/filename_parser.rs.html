1//! SVG filename parser for Fluent UI System Icons.
2//!
3//! Parses `ic_fluent_{name}_{size}_{style}.svg` filenames. The parser is
4//! stateful per slice: it validates that both the grid `size` and the
5//! `style` suffix match the slice directory it is constructed for, so a
6//! re-vendoring mistake (e.g. a 16px or `light` file landing in the
7//! `24-regular` slice) is rejected instead of silently exported.
8
9use fonts_rs_generator::ParsedGlyph;
10use fonts_rs_generator::SvgFilenameParser;
11
12/// Parses `ic_fluent_{name}_{size}_{style}.svg` filenames.
13///
14/// Yields [`ParsedGlyph::raw_name`] as the upstream snake-case icon stem
15/// (e.g. `i_o_t`); the style suffix (`regular`/`filled`) is expressed by
16/// the chosen [`ExportConfig`](fonts_rs_generator::ExportConfig)
17/// variant, not by the glyph name itself.
18pub struct FluentFilenameParser {
19    /// Expected grid size in px (v1: 24).
20    size: u16,
21    /// Expected style suffix (`"regular"` | `"filled"`).
22    style: &'static str,
23}
24
25impl FluentFilenameParser {
26    /// Creates a parser for the `{style}-{size}` slice directory.
27    pub const fn new(size: u16, style: &'static str) -> Self {
28        Self { size, style }
29    }
30}
31
32impl SvgFilenameParser for FluentFilenameParser {
33    fn parse(&self, filename: &str) -> Option<ParsedGlyph> {
34        let stem = filename.strip_prefix("ic_fluent_")?.strip_suffix(".svg")?;
35        let (rest, style) = stem.rsplit_once('_')?;
36        if style != self.style {
37            return None;
38        }
39        let (name, size) = rest.rsplit_once('_')?;
40        if name.is_empty() || size.parse::<u16>().ok()? != self.size {
41            return None;
42        }
43        Some(ParsedGlyph {
44            codepoints: None,
45            raw_name: name.to_string(),
46        })
47    }
48}

50#[cfg(test)]
51mod tests {
52    use super::FluentFilenameParser;
53    use fonts_rs_generator::SvgFilenameParser;
54
55    /// Parser for the 24px regular slice under test.
56    const PARSER: FluentFilenameParser = FluentFilenameParser::new(24, "regular");
57
58    #[test]
59    fn parses_regular_filename() {
60        let glyph = PARSER.parse("ic_fluent_add_24_regular.svg").expect("should parse");
61        assert_eq!(glyph.raw_name, "add");
62        assert!(glyph.codepoints.is_none());
63    }
64
65    #[test]
66    fn parses_snake_case_name() {
67        let glyph = PARSER.parse("ic_fluent_i_o_t_24_regular.svg").expect("should parse");
68        assert_eq!(glyph.raw_name, "i_o_t");
69    }
70
71    #[test]
72    fn parses_name_with_digits() {
73        let glyph = PARSER.parse("ic_fluent_usb_stick_24_regular.svg").expect("should parse");
74        assert_eq!(glyph.raw_name, "usb_stick");
75        let glyph = PARSER.parse("ic_fluent_window_dev_tools_24_regular.svg").expect("should parse");
76        assert_eq!(glyph.raw_name, "window_dev_tools");
77    }
78
79    #[test]
80    fn parses_filled_style_with_matching_parser() {
81        let parser = FluentFilenameParser::new(24, "filled");
82        let glyph = parser.parse("ic_fluent_add_24_filled.svg").expect("should parse");
83        assert_eq!(glyph.raw_name, "add");
84    }
85
86    /// Wrong-style files (`filled`, `light`, `color` variants) must be
87    /// rejected by a regular-slice parser.
88    #[test]
89    fn rejects_wrong_style() {
90        assert!(PARSER.parse("ic_fluent_add_24_filled.svg").is_none());
91        assert!(PARSER.parse("ic_fluent_add_24_light.svg").is_none());
92        assert!(PARSER.parse("ic_fluent_add_24_color.svg").is_none());
93    }
94
95    /// Files from other grid sizes must be rejected - the 24px slice
96    /// must not pick up 16/48px artwork.
97    #[test]
98    fn rejects_wrong_size() {
99        assert!(PARSER.parse("ic_fluent_add_16_regular.svg").is_none());
100        assert!(PARSER.parse("ic_fluent_add_20_regular.svg").is_none());
101        assert!(PARSER.parse("ic_fluent_add_48_regular.svg").is_none());
102    }
103
104    /// Malformed filenames (missing prefix, wrong extension, missing
105    /// size, empty name, non-numeric size) are rejected.
106    #[test]
107    fn rejects_malformed_filenames() {
108        assert!(PARSER.parse("add_24_regular.svg").is_none());
109        assert!(PARSER.parse("ic_fluent_add_24_regular.png").is_none());
110        assert!(PARSER.parse("ic_fluent_add_regular.svg").is_none());
111        assert!(PARSER.parse("ic_fluent__24_regular.svg").is_none());
112        assert!(PARSER.parse("ic_fluent_add_xx_regular.svg").is_none());
113    }
114}
