1//! Fluent UI System Icons integration for GTK 4.
2//!
3//! Hybrid family: pre-rendered upstream SVGs of both styles ship in
4//! the icon GResource (symbolic-recolorable via `currentColor`) and
5//! the `FluentSystemIcons-Regular.ttf` webfont ships for `pango`,
6//! `html` (@font-face) and `render` (pixel-drawing). Regular
7//! codepoints are the real upstream values; filled codepoints are
8//! hashed into Supplementary PUA-B (upstream allocates both styles
9//! into the same PUA space), stable per icon name.
10//!
11//! Both styles export in one build: `fluent-{name}-symbolic` for
12//! regular, `fluent-filled-{name}-symbolic` for filled. Filled
13//! glyphs render via the icon theme / SVG path only - the bundled
14//! TTF covers regular.
15//!
16//! Deprecated upstream icons (`hidden.json`: `flag-pride`,
17//! `text-color-accent`) are excluded from the export. The `metadata`
18//! feature exposes [`metadata::FluentMetadata`] with filename-derived
19//! keywords and the iconify alias table.
20//!
21//! ## Quick Start
22//!
23//! ```no_run
24//! # #[cfg(feature = "gtk")]
25//! use fonts_rs_fluent::Fluent;
26//!
27//! # #[cfg(feature = "gtk")]
28//! // Call once at startup before using any icons
29//! if let Err(e) = Fluent::install() {
30//!     eprintln!("Failed to install font: {e}");
31//! }
32//! ```
33//!
34//! Then resolve icon names to Unicode codepoints:
35//!
36//! ```no_run
37//! use fonts_rs_fluent::Fluent;
38//!
39//! let codepoint = Fluent::codepoint("fluent-add-symbolic");
40//! ```

75/// Returns all known Fluent icons as `(CodePoint, icon_name)` pairs,
76/// sorted alphabetically by icon name.
77pub fn all_icons() -> Vec<(CodePoint, String)> {
78    let mut icons: Vec<(CodePoint, String)> = codepoint_map::GLYPHS
79        .entries()
80        .map(|(cp, name)| (CodePoint::from(*cp), name.to_string()))
81        .collect();
82    icons.sort_by(|a, b| a.1.cmp(&b.1));
83    icons
84}

201    /// End-to-end `CurrentColor` proof: rendering a `-symbolic` icon
202    /// through [`gtk4::SymbolicPaintable::snapshot_symbolic`] with a
203    /// non-default foreground (magenta) must paint pixels of *that*
204    /// color. If `fill="#212121"` had survived, the rendered pixels
205    /// would be dark gray regardless of the symbolic color passed.
206    #[cfg(feature = "gtk")]
207    #[test]
208    fn icon_theme_renders_current_color() {
209        use gtk4::prelude::*;
210
211        if gtk4::init().is_err() {
212            eprintln!("SKIP: no display for GTK rendering test");
213            return;
214        }
215        let display = gtk4::gdk::Display::default().expect("display after gtk4::init");
216        Fluent::install_icon_theme(&display).expect("icon theme registration should succeed");
217        let theme = gtk4::IconTheme::for_display(&display);
218        let name = "fluent-checkmark-symbolic".to_string();
219        assert!(theme.has_icon(&name), "{name} should resolve in the icon theme");
220        let icon = theme.lookup_icon(&name, &[], 96, 1, gtk4::TextDirection::None, gtk4::IconLookupFlags::empty());
221        assert!(icon.is_symbolic(), "{name}: expected symbolic paintable (currentColor follows theme color)");
222
223        let magenta = gtk4::gdk::RGBA::new(1.0, 0.0, 1.0, 1.0);
224        let snapshot = gtk4::Snapshot::new();
225        icon.snapshot_symbolic(&snapshot, 96.0, 96.0, &[magenta]);
226        let node = snapshot.to_node().expect("snapshot should produce a render node");
227
228        let mut surface = gtk4::gdk::cairo::ImageSurface::create(gtk4::gdk::cairo::Format::ARgb32, 96, 96).expect("cairo surface");
229        {
230            let cr = gtk4::gdk::cairo::Context::new(&surface).expect("cairo context");
231            node.draw(&cr);
232        }
233        surface.flush();
234        let (width, height, stride) = (surface.width() as usize, surface.height() as usize, surface.stride() as usize);
235        let data = surface.data().expect("surface data");
236
237        // ARgb32 on little-endian = premultiplied B,G,R,A byte order.
238        // Magenta: R==B, G==0 - dark #212121 would give R==G==B.
239        let (mut opaque, mut magenta_pixels) = (0usize, 0usize);
240        for y in 0..height {
241            for x in 0..width {
242                let offset = y * stride + x * 4;
243                let (b, g, r, a) = (data[offset], data[offset + 1], data[offset + 2], data[offset + 3]);
244                if a == 0 {
245                    continue;
246                }
247                opaque += 1;
248                if g as u16 <= r as u16 / 2 && r as u16 + 8 >= b as u16 {
249                    magenta_pixels += 1;
250                }
251            }
252        }
253        assert!(opaque > 0, "{name}: icon rendered fully transparent");
254        let magenta_ratio = magenta_pixels as f64 / opaque as f64;
255        assert!(
256            magenta_ratio > 0.5,
257            "{name}: rendered pixels should follow the magenta symbolic color, got {magenta_ratio:.2} magenta ratio \
258             (opaque={opaque}) - CurrentColor rewrite likely missing"
259        );
260    }
