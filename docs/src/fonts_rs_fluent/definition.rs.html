1//! Font family configuration for Fluent UI System Icons.
2//!
3//! Defines the constants that make Fluent a distinct font family:
4//! family name, display name, GResource prefix, codepoint ranges, and
5//! icon name prefix.
6
7use const_format::concatcp;
8use fonts_rs_model::CodePoint;
9use fonts_rs_model::CodePointRange;
10use fonts_rs_model::FontFamilyConfig;
11use fonts_rs_model::GRESOURCE_BASE_PREFIX;
12
13/// Fluent UI System Icons codepoint ranges.
14///
15/// The combined export spans three Private Use Areas: the BMP PUA
16/// (`U+E000`–`U+F8FF`) holds the regular map's bulk, Supplementary
17/// PUA-A (`U+F0000`–`U+F0DFF`) its ~26% tail, and Supplementary PUA-B
18/// (`U+100000`–`U+10FFFD`) the hashed filled codepoints (upstream
19/// allocates both fonts into the same space, so filled cannot reuse
20/// its map). All ranges must be probed for the reverse cmap.
21const FLUENT_RANGES: &[CodePointRange] = &[
22    CodePointRange::new(CodePoint::from_char('\u{E000}'), CodePoint::from_char('\u{F8FF}')),
23    CodePointRange::new(CodePoint::from_char('\u{F0000}'), CodePoint::from_char('\u{F0DFF}')),
24    CodePointRange::new(CodePoint::from_char('\u{100000}'), CodePoint::from_char('\u{10FFFD}')),
25];
26
27/// Configuration for the Fluent UI System Icons font family.
28///
29/// Hybrid set: bundled `FluentSystemIcons-Regular.ttf` webfont for
30/// `pango`, `html` (@font-face) and `render` (pixel-drawing), plus
31/// pre-rendered SVGs of *both* styles (regular + filled) for the
32/// GResource icon theme. Regular codepoints come from the vendored
33/// upstream map; filled codepoints are hashed into PUA-B (upstream
34/// allocates both fonts into the same PUA space).
35pub struct FluentConfig;
36
37/// Family name inside the bundled TTF's name table.
38///
39/// The combined build ships only the regular webfont - the filled
40/// style renders via the icon theme / SVG path (its codepoints are
41/// shifted, not the real upstream values). Used as `CSS_FONT_FAMILY`
42/// so Pango/`font-family` resolution and `@font-face` pages reference
43/// the name the TTF actually exposes; the display name `"Fluent UI
44/// System Icons"` would never match.
45///
46/// `dead_code`: unused in the `build.rs` context (`#[path]` include).
47#[allow(dead_code)]
48pub const TTF_FAMILY_NAME: &str = "FluentSystemIcons-Regular";
49
50impl FontFamilyConfig for FluentConfig {
51    /// Cargo feature / font family identifier.
52    const FONT_FAMILY_NAME: &'static str = "fluent";
53
54    /// Human-readable display name for UI and documentation.
55    const FAMILY_DISPLAY_NAME: &'static str = "Fluent UI System Icons";
56
57    /// GResource path prefix for this family's resources.
58    const GRESOURCE_PREFIX: &'static str = concatcp!(GRESOURCE_BASE_PREFIX, "/", FluentConfig::FONT_FAMILY_NAME);
59
60    /// Codepoint ranges covered by this font family (BMP PUA + PUA-A tail).
61    const CODEPOINT_RANGES: &[CodePointRange] = FLUENT_RANGES;
62
63    /// Prefix for icon names and CSS classes (`fluent-{name}`).
64    const ICON_PREFIX: &'static str = "fluent-";
65}
