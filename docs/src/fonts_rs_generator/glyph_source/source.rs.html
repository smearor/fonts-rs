12/// A source of glyphs for the build pipeline.
13///
14/// Implementations provide glyph identities ([`GlyphDescriptor`] list)
15/// and the SVG artwork for each glyph. The shared [`export`](Self::export)
16/// runner turns both into `GlyphEntry` artifacts - `metadata.json`,
17/// `gresource.xml`, `phf` codegen consume them uniformly.
18///
19/// Two implementations exist: `FontFileSource` (TTF/OTF outline
20/// extraction via `skrifa`) and `SvgDirSource` (pre-rendered SVG
21/// directories).
22pub trait GlyphSource {
23    /// Lists all glyphs the source provides, in deterministic order.
24    fn glyphs(&self) -> Result<Vec<GlyphDescriptor>, GlyphSourceError>;
25
26    /// Returns the SVG artwork for the given glyph.
27    fn svg_for(&self, glyph: &GlyphDescriptor) -> Result<String, GlyphSourceError>;
28
29    /// Runs the shared export pipeline: writes each glyph's SVG into
30    /// `resources_dir`, then emits `metadata.json` and
31    /// `icons.gresource.xml`.
32    ///
33    /// Default implementation - works for every `GlyphSource`, whether
34    /// `FontFileSource` or `SvgDirSource`. Adapter over
35    /// [`export_to`](Self::export_to) building the [`GlyphExportTarget`]
36    /// from `X`'s compile-time constants and `config`.
37    ///
38    /// Returns the number of exported glyphs.
39    fn export<X: FontFamilyConfig>(&self, config: &ExportConfig<X>, resources_dir: &Path) -> Result<usize, GlyphSourceError> {
40        let target = GlyphExportTarget {
41            gresource_prefix: config.gresource_prefix.clone(),
42            icons_context: X::ICONS_CONTEXT.to_string(),
43        };
44        self.export_to(&target, resources_dir)
45    }
46
47    /// Runs the shared export pipeline against a runtime
48    /// [`GlyphExportTarget`] - the `icons_context`/`gresource_prefix`
49    /// values [`export`](Self::export) previously sourced from its
50    /// `FontFamilyConfig` type parameter.
51    ///
52    /// Entry point for builds without a compile-time config type, e.g.
53    /// TOML-driven custom icon sets.
54    ///
55    /// Returns the number of exported glyphs.
56    fn export_to(&self, target: &GlyphExportTarget, resources_dir: &Path) -> Result<usize, GlyphSourceError> {
57        let icons_dir = target.prepare_icons_dir(resources_dir)?;
58
59        let descriptors = self.glyphs()?;
60        let mut entries: Vec<GlyphEntry<String>> = Vec::with_capacity(descriptors.len());
61        for descriptor in descriptors {
62            let svg = self.svg_for(&descriptor)?;
63            let mut entry = GlyphEntry::new(descriptor.code, descriptor.name, &target.gresource_prefix, &target.icons_context);
64            entry.sequence = descriptor.sequence;
65            fs::write(icons_dir.join(format!("{}.svg", entry.name)), svg)?;
66            entries.push(entry);
67        }
68
69        entries.sort_by(|a, b| a.name.cmp(&b.name));
70
71        let json_path = resources_dir.join("metadata.json");
72        let json = serde_json::to_string_pretty(&entries)?;
73        fs::write(&json_path, json)?;
74
75        let xml_path = resources_dir.join("icons.gresource.xml");
76        target.generate_gresource_xml(&entries, &xml_path)?;
77
78        Ok(entries.len())
79    }
80}

88    /// Minimal `FontFamilyConfig` fixture for the regression test.
89    struct TestConfig;

91    impl FontFamilyConfig for TestConfig {
92        const FONT_FAMILY_NAME: &'static str = "test-icons";
93        const FAMILY_DISPLAY_NAME: &'static str = "Test Icons";
94        const GRESOURCE_PREFIX: &'static str = "/io/test/icons";
95    }

97    /// Collects all files under `dir` as `(relative path, contents)` pairs.
98    fn collect_files(dir: &Path, base: &Path, files: &mut Vec<(String, Vec<u8>)>) {
99        for entry in fs::read_dir(dir).expect("read dir") {
100            let path = entry.expect("dir entry").path();
101            if path.is_dir() {
102                collect_files(&path, base, files);
103            } else {
104                let rel = path.strip_prefix(base).expect("prefix").to_string_lossy().to_string();
105                files.push((rel, fs::read(&path).expect("read file")));
106            }
107        }
108        files.sort_by(|a, b| a.0.cmp(&b.0));
109    }
110
111    /// `export` delegates to `export_to` via a `GlyphExportTarget` built
112    /// from `X`'s constants - both paths must produce byte-identical
113    /// artifacts (regression gate for the export refactor).
114    #[test]
115    fn export_and_export_to_produce_identical_output() {
116        let base = std::env::temp_dir().join(format!("glyph_source_export_test_{}", std::process::id()));
117        let svg_dir = base.join("svgs");
118        fs::create_dir_all(&svg_dir).expect("create svg dir");
119        let svg = "<svg xmlns=\"http://www.w3.org/2000/svg\"><path d=\"M0\"/></svg>";
120        fs::write(svg_dir.join("alpha.svg"), svg).expect("write alpha");
121        fs::write(svg_dir.join("beta.svg"), svg).expect("write beta");
122
123        let source = SvgDirSource::new(&svg_dir);
124        let config = ExportConfig::<TestConfig>::new();
125
126        let via_export = base.join("via_export");
127        let via_export_to = base.join("via_export_to");
128        source.export(&config, &via_export).expect("export");
129
130        let target = GlyphExportTarget {
131            gresource_prefix: config.gresource_prefix.clone(),
132            icons_context: TestConfig::ICONS_CONTEXT.to_string(),
133        };
134        source.export_to(&target, &via_export_to).expect("export_to");
135
136        let mut a = Vec::new();
137        let mut b = Vec::new();
138        collect_files(&via_export, &via_export, &mut a);
139        collect_files(&via_export_to, &via_export_to, &mut b);
140        assert_eq!(a, b);
141
142        fs::remove_dir_all(&base).expect("remove temp dir");
143    }
