12/// A source of glyphs for the build pipeline.
13///
14/// Implementations provide glyph identities ([`GlyphDescriptor`] list)
15/// and the SVG artwork for each glyph. The shared [`export`](Self::export)
16/// runner turns both into `GlyphEntry` artifacts - `metadata.json`,
17/// `gresource.xml`, `phf` codegen consume them uniformly.
18///
19/// Two implementations exist: `FontFileSource` (TTF/OTF outline
20/// extraction via `skrifa`) and `SvgDirSource` (pre-rendered SVG
21/// directories).
22pub trait GlyphSource {
23    /// Lists all glyphs the source provides, in deterministic order.
24    fn glyphs(&self) -> Result<Vec<GlyphDescriptor>, GlyphSourceError>;
25
26    /// Returns the SVG artwork for the given glyph.
27    fn svg_for(&self, glyph: &GlyphDescriptor) -> Result<String, GlyphSourceError>;
28
29    /// Runs the shared export pipeline: writes each glyph's SVG into
30    /// `resources_dir`, then emits `metadata.json` and
31    /// `icons.gresource.xml`.
32    ///
33    /// Default implementation - works for every `GlyphSource`, whether
34    /// `FontFileSource` or `SvgDirSource`.
35    ///
36    /// Returns the number of exported glyphs.
37    fn export<X: FontFamilyConfig>(&self, config: &ExportConfig<X>, resources_dir: &Path) -> Result<usize, GlyphSourceError> {
38        let icons_dir = X::prepare_icons_dir(resources_dir)?;
39
40        let descriptors = self.glyphs()?;
41        let mut entries: Vec<GlyphEntry<String>> = Vec::with_capacity(descriptors.len());
42        for descriptor in descriptors {
43            let svg = self.svg_for(&descriptor)?;
44            let mut entry = GlyphEntry::new(descriptor.code, descriptor.name, &config.gresource_prefix, X::ICONS_CONTEXT);
45            entry.sequence = descriptor.sequence;
46            fs::write(icons_dir.join(format!("{}.svg", entry.name)), svg)?;
47            entries.push(entry);
48        }
49
50        entries.sort_by(|a, b| a.name.cmp(&b.name));
51
52        let json_path = resources_dir.join("metadata.json");
53        let json = serde_json::to_string_pretty(&entries)?;
54        fs::write(&json_path, json)?;
55
56        let xml_path = resources_dir.join("icons.gresource.xml");
57        config.generate_gresource_xml(&entries, &xml_path)?;
58
59        Ok(entries.len())
60    }
61}
