1//! [`GlyphSource`] that merges several sub-sources into one export.
2//!
3//! [`CombinedSource`] concatenates the glyph lists of its sub-sources
4//! and routes [`svg_for`](GlyphSource::svg_for) back to the owning
5//! source via a name marker. It exists for families whose upstream set
6//! ships several *styles* (fill/outline, regular/filled, …) that a
7//! single build should export side by side - e.g. so the cheat sheet
8//! can group and collapse them via `GlyphOrdering`.
9//!
10//! Names must disambiguate across sources (e.g. `eva-{name}-symbolic`
11//! vs `eva-{name}-outline-symbolic`); a duplicate resolved name or
12//! codepoint fails the build.
13
14use crate::glyph_source::descriptor::GlyphDescriptor;
15use crate::glyph_source::error::GlyphSourceError;
16use crate::glyph_source::source::GlyphSource;
17use std::collections::HashSet;
18
19/// Merges multiple [`GlyphSource`]s into a single export.
20///
21/// `svg_for` routes a glyph back to exactly the source that produced
22/// it: [`glyphs`](GlyphSource::glyphs) tags each descriptor's
23/// `source_key` with the owner's registration index. (The `source()`
24/// *name marker* remains as fallback routing for descriptors that did
25/// not pass through `glyphs()`.) Registration order still matters for
26/// `glyphs`: sub-sources allocate codepoints in registration order, so
27/// register the style whose codepoints must stay stable first.
28///
29/// All sources must be of the same concrete type - typically
30/// [`SvgDirSource`](crate::glyph_source::SvgDirSource) with one entry
31/// per style directory.
32pub struct CombinedSource<S: GlyphSource> {
33    /// `(name marker, source)` pairs, checked in order for `svg_for`.
34    sources: Vec<(String, S)>,
35}
36
37impl<S: GlyphSource> CombinedSource<S> {
38    /// Creates an empty combined source.
39    pub fn new() -> Self {
40        Self { sources: Vec::new() }
41    }
42
43    /// Adds a sub-source whose glyphs are routed back when `name`
44    /// contains `name_marker` (e.g. `"-outline"`). For `svg_for` the
45    /// longest matching marker wins; for `glyphs` sources are visited
46    /// in registration order.
47    pub fn source(mut self, name_marker: impl Into<String>, source: S) -> Self {
48        self.sources.push((name_marker.into(), source));
49        self
50    }
51}
52
53impl<S: GlyphSource> Default for CombinedSource<S> {
54    fn default() -> Self {
55        Self::new()
56    }
57}
58
59impl<S: GlyphSource> GlyphSource for CombinedSource<S> {
60    /// Returns the concatenation of all sub-source glyph lists.
61    ///
62    /// Each descriptor's `source_key` is tagged with the owning
63    /// source's registration index (`"{index}\u{1F}{key}"`) so
64    /// `svg_for` routes back to exactly that source - names and
65    /// filenames may coincide across sources (e.g. two crates shipping
66    /// a `check.svg`). The tag never reaches serialized output:
67    /// `GlyphEntry` derives `file`/`resource_path` from the name.
68    ///
69    /// Fails when two sources resolve to the same glyph name or
70    /// codepoint - merged exports must partition the glyph namespace.
71    fn glyphs(&self) -> Result<Vec<GlyphDescriptor>, GlyphSourceError> {
72        let mut descriptors = Vec::new();
73        for (index, (_, source)) in self.sources.iter().enumerate() {
74            for mut descriptor in source.glyphs()? {
75                descriptor.source_key = format!("{index}\u{1F}{}", descriptor.source_key);
76                descriptors.push(descriptor);
77            }
78        }
79
80        let mut names = HashSet::with_capacity(descriptors.len());
81        let mut codes = HashSet::with_capacity(descriptors.len());
82        for descriptor in &descriptors {
83            if !names.insert(&descriptor.name) {
84                return Err(GlyphSourceError::PostProcess(format!("combined sources produced duplicate glyph name '{}'", descriptor.name)));
85            }
86            if let Some(code) = descriptor.code
87                && !codes.insert(code.as_char())
88            {
89                return Err(GlyphSourceError::PostProcess(format!(
90                    "combined sources produced duplicate codepoint U+{:04X} ('{}')",
91                    code.as_char() as u32,
92                    descriptor.name
93                )));
94            }
95        }
96        Ok(descriptors)
97    }
98
99    /// Routes to the source that produced the descriptor.
100    ///
101    /// Descriptors passing through [`glyphs`](Self::glyphs) carry their
102    /// owner's registration index in `source_key`; that tag takes
103    /// precedence. Foreign descriptors fall back to the longest
104    /// matching name marker (the original routing rule).
105    fn svg_for(&self, glyph: &GlyphDescriptor) -> Result<String, GlyphSourceError> {
106        if let Some((index, key)) = glyph.source_key.split_once('\u{1F}')
107            && let Ok(index) = index.parse::<usize>()
108            && let Some((_, source)) = self.sources.get(index)
109        {
110            let mut routed = glyph.clone();
111            routed.source_key = key.to_string();
112            return source.svg_for(&routed);
113        }
114
115        self.sources
116            .iter()
117            .filter(|(marker, _)| glyph.name.contains(marker.as_str()))
118            .max_by_key(|(marker, _)| marker.len())
119            .map(|(_, source)| source.svg_for(glyph))
120            .unwrap_or_else(|| Err(GlyphSourceError::PostProcess(format!("no combined source matches glyph '{}'", glyph.name))))
121    }
122}
