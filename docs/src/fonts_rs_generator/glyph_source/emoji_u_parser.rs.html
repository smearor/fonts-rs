1//! `emoji_u*` filename parser for emoji SVG sets.
2
3use crate::glyph_source::filename_parser::SvgFilenameParser;
4use crate::glyph_source::parsed_glyph::ParsedGlyph;
5use fonts_rs_model::CodePoint;
6
7/// Parses `emoji_u{cp1}_{cp2}_….svg` filenames (blobmoji2, noto-emoji).
8///
9/// Handles multi-codepoint sequences (ZWJ emoji, skin-tone modifiers,
10/// keycaps) - each `_`-separated hex segment becomes one [`CodePoint`].
11/// The filename stem (without `emoji_u` prefix and `.svg` suffix)
12/// becomes [`ParsedGlyph::raw_name`].
13pub struct EmojiUFilenameParser;
14
15impl SvgFilenameParser for EmojiUFilenameParser {
16    fn parse(&self, filename: &str) -> Option<ParsedGlyph> {
17        let stem = filename.strip_prefix("emoji_u")?.strip_suffix(".svg")?;
18        let codepoints = stem
19            .split('_')
20            .map(|segment| u32::from_str_radix(segment, 16).ok().and_then(char::from_u32).map(CodePoint::from_char))
21            .collect::<Option<Vec<_>>>()?;
22        Some(ParsedGlyph {
23            codepoints: Some(codepoints),
24            raw_name: stem.to_string(),
25        })
26    }
27}

29#[cfg(test)]
30mod tests {
31    use super::*;
32
33    #[test]
34    fn parses_single_codepoint() {
35        let parsed = EmojiUFilenameParser.parse("emoji_u1f600.svg").expect("parsed");
36        assert_eq!(parsed.raw_name, "1f600");
37        let codepoints = parsed.codepoints.expect("codepoints");
38        assert_eq!(codepoints.len(), 1);
39        assert_eq!(codepoints[0].as_char(), '\u{1F600}');
40    }
41
42    #[test]
43    fn parses_zwj_sequence() {
44        let parsed = EmojiUFilenameParser.parse("emoji_u1f468_200d_1f469.svg").expect("parsed");
45        let codepoints = parsed.codepoints.expect("codepoints");
46        let chars: Vec<char> = codepoints.iter().map(|codepoint| codepoint.as_char()).collect();
47        assert_eq!(chars, ['\u{1F468}', '\u{200D}', '\u{1F469}']);
48    }
49
50    #[test]
51    fn rejects_non_emoji_filenames() {
52        assert!(EmojiUFilenameParser.parse("home.svg").is_none());
53        assert!(EmojiUFilenameParser.parse("emoji_u1f600.png").is_none());
54        assert!(EmojiUFilenameParser.parse("emoji_u.svg").is_none());
55    }
56
57    #[test]
58    fn rejects_invalid_hex_segments() {
59        assert!(EmojiUFilenameParser.parse("emoji_uZZZ.svg").is_none());
60        assert!(EmojiUFilenameParser.parse("emoji_u1f600_.svg").is_none());
61    }
