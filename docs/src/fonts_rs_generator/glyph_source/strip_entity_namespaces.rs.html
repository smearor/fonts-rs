12/// Strips DOCTYPE entity declarations and the namespaced content that
13/// relies on them.
14///
15/// Adobe Illustrator exports declare namespace URIs via DOCTYPE
16/// entities (`xmlns:i="&ns_ai;"`) and embed metadata such as
17/// `<i:aipgf>` PGIF blobs or `i:extraneous` attributes. librsvg rejects
18/// entity declarations outright, and composing the SVG into a larger
19/// document drops the DOCTYPE - either way the prefixes end up
20/// unbound and parsing fails.
21///
22/// The processor removes the DOCTYPE, every `xmlns:*` attribute whose
23/// value is an entity reference, and all elements and attributes using
24/// the removed prefixes. Files without a DOCTYPE pass through
25/// unchanged.
26pub struct StripEntityNamespaces;
27
28impl SvgPostProcessor for StripEntityNamespaces {
29    fn process(&self, svg: String) -> Result<String, GlyphSourceError> {
30        if !svg.contains("<!DOCTYPE") {
31            return Ok(svg);
32        }
33        let svg = strip_doctype(&svg);
34
35        let mut reader = Reader::from_str(&svg);
36        let mut writer = Writer::new(Vec::new());
37        let mut removed: HashSet<String> = HashSet::new();
38
39        loop {
40            match reader.read_event().map_err(|error| GlyphSourceError::PostProcess(error.to_string()))? {
41                Event::Eof => break,
42                Event::DocType(_) => continue,
43                Event::Start(element) => {
44                    let name = element.name().as_ref().to_owned();
45                    let Some(element) = filter_element(element, &mut removed) else {
46                        reader
47                            .read_to_end(QName(name.as_str()))
48                            .map_err(|error| GlyphSourceError::PostProcess(error.to_string()))?;
49                        continue;
50                    };
51                    writer
52                        .write_event(Event::Start(element))
53                        .map_err(|error| GlyphSourceError::PostProcess(error.to_string()))?;
54                }
55                Event::Empty(element) => {
56                    if let Some(element) = filter_element(element, &mut removed) {
57                        writer
58                            .write_event(Event::Empty(element))
59                            .map_err(|error| GlyphSourceError::PostProcess(error.to_string()))?;
60                    }
61                }
62                event => writer.write_event(event).map_err(|error| GlyphSourceError::PostProcess(error.to_string()))?,
63            }
64        }
65        String::from_utf8(writer.into_inner()).map_err(|error| GlyphSourceError::PostProcess(error.to_string()))
66    }
67}

69/// Removes the `<!DOCTYPE …>` block, including an internal entity
70/// subset (`[ <!ENTITY …> ]>`), from the document prolog.
71fn strip_doctype(svg: &str) -> String {
72    let Some(start) = svg.find("<!DOCTYPE") else {
73        return svg.to_string();
74    };
75    let rest = &svg[start..];
76    let tag_end = rest.find('>');
77    let subset_start = rest.find('[');
78    let end = match (subset_start, tag_end) {
79        (Some(subset), Some(tag)) if subset < tag => rest.find("]>").map(|i| i + 2),
80        _ => tag_end.map(|i| i + 1),
81    };
82    let Some(end) = end else {
83        return svg.to_string();
84    };
85    format!("{}{}", &svg[..start], &svg[start + end..])
86}

88/// Returns the namespace prefix of a qualified name, if any.
89fn prefix_of(name: &str) -> Option<&str> {
90    name.find(':').map(|pos| &name[..pos])
91}
92
93/// Filters one element's attributes, collecting prefixes declared via
94/// entity references into `removed` and dropping attributes that use
95/// removed prefixes.
96///
97/// Returns `None` when the element's own name uses a removed prefix -
98/// the caller then skips the element (and its subtree for
99/// [`Event::Start`]).
100fn filter_element(element: BytesStart<'_>, removed: &mut HashSet<String>) -> Option<BytesStart<'static>> {
101    let name = element.name().as_ref().to_owned();
102    let mut kept = Vec::new();
103    for attribute in element.attributes().flatten() {
104        let key = attribute.key.as_ref();
105        if let Some(prefix) = key.strip_prefix("xmlns:")
106            && attribute.value.contains('&')
107        {
108            removed.insert(prefix.to_owned());
109            continue;
110        }
111        if prefix_of(key).is_some_and(|prefix| removed.contains(prefix)) {
112            continue;
113        }
114        kept.push(attribute);
115    }
116    if prefix_of(&name).is_some_and(|prefix| removed.contains(prefix)) {
117        return None;
118    }
119    let mut filtered = BytesStart::new(&name);
120    filtered.extend_attributes(kept);
121    Some(filtered.into_owned())
122}

124#[cfg(test)]
125mod tests {
126    use super::*;
127
128    /// Illustrator-style document: DOCTYPE entities, prefixed metadata
129    /// attributes, and an `<i:aipgf>` blob.
130    #[test]
131    fn strips_entity_namespaces_and_their_users() {
132        let svg = concat!(
133            "<?xml version=\"1.0\"?>",
134            "<!DOCTYPE svg [ <!ENTITY ns_ai \"http://ns.adobe.com/AdobeIllustrator/10.0/\"> ]>",
135            "<svg xmlns=\"http://www.w3.org/2000/svg\" xmlns:i=\"&ns_ai;\" xmlns:xlink=\"http://www.w3.org/1999/xlink\" viewBox=\"0 0 128 128\">",
136            "<g i:extraneous=\"self\"><path fill=\"#fff\" d=\"M0 0\"/></g>",
137            "<i:aipgf i:pgfEncoding=\"z\"><![CDATA[binary]]></i:aipgf>",
138            "</svg>"
139        )
140        .to_string();
141        let result = StripEntityNamespaces.process(svg).expect("processed");
142        assert!(!result.contains("DOCTYPE"), "doctype removed: {result}");
143        assert!(!result.contains("ns_ai"), "entity declaration removed: {result}");
144        assert!(!result.contains("i:"), "prefixed content removed: {result}");
145        assert!(result.contains("xmlns:xlink"), "real namespace kept: {result}");
146        assert!(result.contains("<path"), "artwork kept: {result}");
147    }
148
149    /// Documents without a DOCTYPE pass through unchanged.
150    #[test]
151    fn clean_documents_pass_through() {
152        let svg = "<svg xmlns=\"http://www.w3.org/2000/svg\"><path d=\"M0 0\"/></svg>".to_string();
153        assert_eq!(StripEntityNamespaces.process(svg.clone()).expect("processed"), svg);
154    }
