1//! GTK symbolic class hints for stroke-based icon sets.
2//!
3//! GTK's symbolic icon recoloring ignores the `fill` and `stroke`
4//! presentation attributes entirely: the painted colors are derived
5//! from symbolic classes (`transparent-fill`, `foreground-stroke`,
6//! ...) or, since GTK 4.20, `gtk:`-namespace attributes. Elements
7//! without fill classes are filled with the foreground color, elements
8//! without stroke classes are not stroked.
9//!
10//! Stroke-based icon sets (Lucide, Tabler outline) therefore render as
11//! solid blobs when exported under `-symbolic` names without class
12//! hints: their `fill="none"` is ignored and the foreground fill
13//! swallows the stroked details. This processor maps each shape's
14//! effective `fill`/`stroke` (including values inherited from ancestor
15//! `<svg>`/`<g>` elements) onto the matching symbolic classes.
16
17use crate::glyph_source::error::GlyphSourceError;
18use crate::glyph_source::post_processor::SvgPostProcessor;
19use quick_xml::Reader;
20use quick_xml::Writer;
21use quick_xml::events::BytesStart;
22use quick_xml::events::Event;
23
24/// SVG shape elements that carry paintable fill/stroke geometry.
25const SHAPE_ELEMENTS: &[&str] = &["path", "circle", "ellipse", "line", "polyline", "polygon", "rect", "text"];
26
27/// Effective presentation paint seen at the current element.
28#[derive(Default, Clone)]
29struct Paint {
30    /// Effective `fill` value, `None` when unset (SVG default: black).
31    fill: Option<String>,
32    /// Effective `stroke` value, `None` when unset (not stroked).
33    stroke: Option<String>,
34}
35
36impl Paint {
37    /// The parent's paints merged with this element's own
38    /// `fill`/`stroke` attributes.
39    fn inherited(&self, element: &BytesStart) -> Result<Paint, GlyphSourceError> {
40        let mut paint = self.clone();
41        for attribute in element.attributes().with_checks(false) {
42            let attribute = attribute.map_err(|error| GlyphSourceError::PostProcess(error.to_string()))?;
43            match attribute.key.as_ref() {
44                "fill" => paint.fill = Some(attribute.value.to_string()),
45                "stroke" => paint.stroke = Some(attribute.value.to_string()),
46                _ => {}
47            }
48        }
49        Ok(paint)
50    }
51
52    /// The symbolic class tokens for this paint state.
53    fn classes(&self) -> Vec<&'static str> {
54        let mut classes = Vec::new();
55        match self.fill.as_deref() {
56            Some("none") => classes.push("transparent-fill"),
57            _ => classes.push("foreground-fill"),
58        }
59        match self.stroke.as_deref() {
60            Some(stroke) if !is_unpainted(stroke) => classes.push("foreground-stroke"),
61            _ => {}
62        }
63        classes
64    }
65}

67/// Adds GTK symbolic classes to every shape element, derived from its
68/// effective `fill`/`stroke` paint attributes.
69///
70/// - `fill="none"` (inherited or direct) → `transparent-fill`;
71///   any other fill or no fill at all → `foreground-fill`
72/// - a `stroke` other than `none`/`url(…)` → `foreground-stroke`;
73///   `stroke="none"` or no stroke → no stroke class (symbolic icons
74///   are "not stroked" by default)
75///
76/// The original attributes are kept untouched so non-symbolic
77/// consumers (HTML export, `glyph_svg`, `compose_svg`) still see the
78/// authored paints - the classes are inert outside GTK's symbolic
79/// renderer. Existing `class` values are preserved and extended.
80///
81/// Limitation: `fill:`/`stroke:` declarations inside `style` attributes
82/// are not interpreted; icon sets authored this way need a different
83/// processor.
84pub struct SymbolicClasses;
85
86impl SvgPostProcessor for SymbolicClasses {
87    fn process(&self, svg: String) -> Result<String, GlyphSourceError> {
88        let mut reader = Reader::from_str(&svg);
89        let mut writer = Writer::new(Vec::new());
90        let mut paint_stack: Vec<Paint> = Vec::new();
91        loop {
92            match reader.read_event().map_err(|error| GlyphSourceError::PostProcess(error.to_string()))? {
93                Event::Start(element) => {
94                    let inherited = paint_stack.last().cloned().unwrap_or_default().inherited(&element)?;
95                    writer.write_event(Event::Start(Self::add_classes(&element, &inherited)?))?;
96                    paint_stack.push(inherited);
97                }
98                Event::Empty(element) => {
99                    let inherited = paint_stack.last().cloned().unwrap_or_default().inherited(&element)?;
100                    writer.write_event(Event::Empty(Self::add_classes(&element, &inherited)?))?;
101                }
102                Event::End(event) => {
103                    paint_stack.pop();
104                    writer.write_event(Event::End(event))?;
105                }
106                Event::Eof => break,
107                event => writer.write_event(event)?,
108            }
109        }
110        String::from_utf8(writer.into_inner()).map_err(|error| GlyphSourceError::PostProcess(error.to_string()))
111    }
112}

114impl SymbolicClasses {
115    /// Rebuilds one element tag with symbolic classes appended for
116    /// shape elements; all other elements pass through unchanged.
117    fn add_classes(element: &BytesStart, paint: &Paint) -> Result<BytesStart<'static>, GlyphSourceError> {
118        let tag = element.name().as_ref().to_string();
119        if !SHAPE_ELEMENTS.contains(&tag.as_str()) {
120            return Ok(element.to_owned());
121        }
122
123        let classes = paint.classes();
124        let mut rewritten = element.to_owned();
125        rewritten.clear_attributes();
126        let mut wrote_class = false;
127        for attribute in element.attributes().with_checks(false) {
128            let attribute = attribute.map_err(|error| GlyphSourceError::PostProcess(error.to_string()))?;
129            let key = attribute.key.as_ref().to_string();
130            let value = attribute.value.to_string();
131            if key == "class" {
132                let merged = format!("{} {}", classes.join(" "), value);
133                rewritten.push_attribute(("class", merged.as_str()));
134                wrote_class = true;
135            } else {
136                rewritten.push_attribute((key.as_str(), value.as_str()));
137            }
138        }
139        if !wrote_class {
140            rewritten.push_attribute(("class", classes.join(" ").as_str()));
141        }
142        Ok(rewritten)
143    }
144}

146/// Returns `true` for `stroke` values that paint nothing.
147fn is_unpainted(value: &str) -> bool {
148    let value = value.trim();
149    value == "none" || value == "transparent" || value.starts_with("url(")
150}

152#[cfg(test)]
153mod tests {
154    use super::*;
155
156    #[test]
157    fn stroke_only_shape_gets_transparent_fill_and_foreground_stroke() {
158        let svg = r#"<svg fill="none" stroke="currentColor"><path d="M0 0"/></svg>"#.to_string();
159        let result = SymbolicClasses.process(svg).expect("processed");
160        assert!(result.contains(r#"class="transparent-fill foreground-stroke""#), "got: {result}");
161        assert!(result.contains(r#"stroke="currentColor""#), "authored stroke must be kept: {result}");
162    }
163
164    #[test]
165    fn fill_only_shape_gets_foreground_fill_without_stroke_class() {
166        let svg = r#"<svg fill="currentColor"><rect width="4" height="4"/></svg>"#.to_string();
167        let result = SymbolicClasses.process(svg).expect("processed");
168        assert!(result.contains(r#"class="foreground-fill""#), "got: {result}");
169        assert!(!result.contains("stroke"), "no stroke class expected: {result}");
170    }
171
172    #[test]
173    fn element_attributes_override_inherited_paints() {
174        let svg = r#"<svg fill="none" stroke="currentColor"><circle cx="1" cy="1" r="1" fill="currentColor" stroke="none"/></svg>"#.to_string();
175        let result = SymbolicClasses.process(svg).expect("processed");
176        assert!(result.contains(r#"class="foreground-fill""#), "overridden fill must win: {result}");
177        assert!(!result.contains("foreground-stroke"), "stroke=\"none\" suppresses the stroke class: {result}");
178    }
179
180    #[test]
181    fn existing_class_values_are_extended() {
182        let svg = r#"<svg fill="none" stroke="currentColor"><path class="foo" d="M0 0"/></svg>"#.to_string();
183        let result = SymbolicClasses.process(svg).expect("processed");
184        assert!(result.contains(r#"class="transparent-fill foreground-stroke foo""#), "got: {result}");
185    }
186
187    #[test]
188    fn non_shape_elements_pass_through() {
189        let svg = r#"<svg fill="none" stroke="currentColor"><defs><clipPath id="c"/></defs></svg>"#.to_string();
190        let result = SymbolicClasses.process(svg).expect("processed");
191        assert!(result.contains("<clipPath"), "non-shape element must pass through: {result}");
192        assert!(!result.contains("foreground-"), "non-shape element must not get classes: {result}");
193    }
194}
