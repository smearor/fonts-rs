1//! Deterministic Private Use Area codepoint allocation.
2
3use crate::glyph_source::error::GlyphSourceError;
4use fonts_rs_model::CodePoint;
5use fonts_rs_model::CodePointRange;
6use fonts_rs_model::PUA_B_RANGE;
7use fonts_rs_model::PUA_RANGE;
8use std::collections::HashSet;
9
10/// Allocates stable Private Use Area codepoints via deterministic
11/// hash-of-name with linear probing on collision.
12///
13/// The default range is the BMP PUA (U+E000–U+F8FF, 6,400 slots);
14/// [`with_range`](Self::with_range) or [`pua_b`](Self::pua_b) select
15/// other arenas - e.g. Supplementary PUA-B for extra styles of a
16/// [`CombinedSource`](crate::glyph_source::CombinedSource), so their
17/// codepoints stay stable even when the base style's map grows.
18///
19/// An icon keeps its codepoint across upstream additions/removals -
20/// persisted references (favorites, recents) never silently break.
21///
22/// Callers must feed names in a deterministic order (e.g. sorted
23/// filenames): on hash collision, the first-processed name wins the
24/// primary slot, so allocation order determines the result.
25#[derive(Debug)]
26pub struct PuaAllocator {
27    /// Codepoints already assigned.
28    occupied: HashSet<CodePoint>,
29    /// Allocatable range (inclusive).
30    range: CodePointRange,
31    /// Number of allocatable slots in `range`.
32    len: u32,
33}
34
35impl Default for PuaAllocator {
36    fn default() -> Self {
37        Self::new()
38    }
39}
40
41impl PuaAllocator {
42    /// Creates an empty allocator over the BMP PUA (U+E000–U+F8FF).
43    pub fn new() -> Self {
44        Self::from_range(PUA_RANGE)
45    }
46
47    /// Creates an empty allocator over Supplementary PUA-B
48    /// (U+100000–U+10FFFD).
49    ///
50    /// PUA-B is disjoint from both the BMP PUA and PUA-A (U+F0000–
51    /// U+FFFFD), so style variants hashed here can never collide with
52    /// a base style's real or hashed codepoints - the allocation stays
53    /// stable no matter how the base map grows.
54    pub fn pua_b() -> Self {
55        Self::from_range(PUA_B_RANGE)
56    }
57
58    /// Creates an empty allocator over `range` (inclusive), or `None`
59    /// when `range` is empty (start after end).
60    ///
61    /// Non-scalar codepoints (the surrogate range U+D800–U+DFFF) are
62    /// skipped during probing - they never occupy a slot.
63    pub fn with_range(range: CodePointRange) -> Option<Self> {
64        (range.end().as_char() >= range.start().as_char()).then(|| Self::from_range(range))
65    }
66
67    /// Infallible construction for callers with a known non-empty
68    /// range (the built-in arenas are compile-time constants).
69    fn from_range(range: CodePointRange) -> Self {
70        let start = range.start().as_char() as u32;
71        let end = range.end().as_char() as u32;
72        Self {
73            occupied: HashSet::new(),
74            range,
75            len: end.saturating_sub(start) + 1,
76        }
77    }
78
79    /// Returns the deterministic codepoint for `name`.
80    ///
81    /// Each name is allocated exactly once per export run - glyph names
82    /// are unique within a family.
83    ///
84    /// # Errors
85    ///
86    /// Returns [`GlyphSourceError::PuaExhausted`] when all slots are
87    /// occupied.
88    pub fn allocate(&mut self, name: &str) -> Result<CodePoint, GlyphSourceError> {
89        let hash = Self::hash(name.as_bytes());
90        let start_slot = (hash % u64::from(self.len)) as u32;
91        let start = self.range.start().as_char() as u32;
92        for offset in 0..self.len {
93            let candidate = start + (start_slot + offset) % self.len;
94            // Non-scalar candidates (surrogates) are not codepoints -
95            // skip without occupying the slot.
96            let Ok(candidate) = CodePoint::try_from(candidate) else {
97                continue;
98            };
99            if self.occupied.insert(candidate) {
100                return Ok(candidate);
101            }
102        }
103        Err(GlyphSourceError::PuaExhausted)
104    }
105
106    /// FNV-1a hash - stable across Rust versions, unlike `DefaultHasher`,
107    /// so generated codepoints are reproducible across builds.
108    fn hash(bytes: &[u8]) -> u64 {
109        bytes
110            .iter()
111            .fold(0xcbf29ce484222325, |hash, byte| (hash ^ u64::from(*byte)).wrapping_mul(0x100000001b3))
112    }
113}

115#[cfg(test)]
116mod tests {
117    use super::*;
118
119    #[test]
120    fn allocation_is_deterministic() {
121        let mut first = PuaAllocator::new();
122        let mut second = PuaAllocator::new();
123        for name in ["home", "settings", "arrow-left"] {
124            assert_eq!(first.allocate(name).ok(), second.allocate(name).ok());
125        }
126    }
127
128    #[test]
129    fn allocated_codepoints_are_in_pua_range() {
130        let mut allocator = PuaAllocator::new();
131        let codepoint = allocator.allocate("home").expect("codepoint");
132        assert!(('\u{E000}'..='\u{F8FF}').contains(&codepoint.as_char()));
133    }
134
135    #[test]
136    fn custom_range_allocation_stays_in_range() {
137        let mut allocator = PuaAllocator::with_range(CodePointRange::from_char('\u{F0000}', '\u{F00FF}')).expect("range non-empty");
138        for index in 0..100 {
139            let codepoint = allocator.allocate(&format!("icon-{index}")).expect("codepoint");
140            assert!(('\u{F0000}'..='\u{F00FF}').contains(&codepoint.as_char()));
141        }
142    }
143
144    /// A range spanning the surrogate gap still yields its valid
145    /// slots; surrogates are skipped instead of exhausting the range.
146    /// U+D7FF–U+E001 has exactly three scalar values.
147    #[test]
148    fn allocation_skips_surrogate_slots() {
149        let mut allocator = PuaAllocator::with_range(CodePointRange::from_char('\u{D7FF}', '\u{E001}')).expect("range non-empty");
150        for name in ["a", "b", "c"] {
151            let codepoint = allocator.allocate(name).expect("codepoint");
152            assert!(!(0xD800..=0xDFFF).contains(&(codepoint.as_char() as u32)));
153        }
154        // Only U+D7FF, U+E000 and U+E001 are scalar values - now
155        // exhausted.
156        assert!(matches!(allocator.allocate("d"), Err(GlyphSourceError::PuaExhausted)));
157    }
158
159    /// An inverted range yields `None` - the caller decides how to
160    /// report the misconfiguration, no panic.
161    #[test]
162    fn empty_range_returns_none() {
163        let allocator = PuaAllocator::with_range(CodePointRange::from_char('\u{E001}', '\u{E000}'));
164        assert!(allocator.is_none());
165    }
166
167    #[test]
168    fn distinct_names_get_distinct_codepoints() {
169        let mut allocator = PuaAllocator::new();
170        let mut codepoints = HashSet::new();
171        for index in 0..1000 {
172            let codepoint = allocator.allocate(&format!("icon-{index}")).expect("codepoint");
173            assert!(codepoints.insert(codepoint.as_char()));
174        }
175    }
