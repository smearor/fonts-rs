13/// Controls the static variable names in generated codemap output.
14pub trait CodemapNaming {
15    /// Name of the forward (codepoint → name) map.
16    const FORWARD: &'static str;
17    /// Name of the reverse (name → codepoint) map.
18    const REVERSE: &'static str;
19}
20
21/// Default naming: `GLYPHS` / `REVERSE_GLYPHS`.
22pub struct DefaultNaming;
23
24impl CodemapNaming for DefaultNaming {
25    const FORWARD: &'static str = "GLYPHS";
26    const REVERSE: &'static str = "REVERSE_GLYPHS";
27}
28
29/// Generates phf::Map constants for codepoint-to-name and name-to-codepoint lookups.
30pub struct CodemapGenerator<Naming: CodemapNaming = DefaultNaming>(PhantomData<Naming>);
31
32impl<N, Naming> GlyphGenerator<N> for CodemapGenerator<Naming>
33where
34    N: AsRef<str> + Clone + Ord + Serialize + for<'a> Deserialize<'a>,
35    Naming: CodemapNaming,
36{
37    fn generate(entries: &[GlyphEntry<N>]) -> Result<String, GenerateError> {
38        let mut output = String::new();
39        output.push_str("// @generated by build.rs - do not edit\n\n");
40        output.push_str("use phf::phf_map;\n\n");
41
42        // Codepoint -> name map
43        //
44        // One phf key per codepoint: sets whose style variants share a
45        // codepoint (Line Awesome regular/solid pairs) would emit
46        // duplicate keys and fail `phf_map!` compilation otherwise. On
47        // collision the canonical name wins - the shortest, with the
48        // lexically largest as tiebreak so the choice is independent
49        // of entry order. `glyph_name()` then resolves a shared
50        // codepoint to e.g. `la-adjust-symbolic`, not the `-solid`
51        // sibling; the reverse map below keeps every exported name
52        // resolvable.
53        let mut forward: BTreeMap<char, &str> = BTreeMap::new();
54        for entry in entries {
55            let Some(code) = &entry.code else { continue };
56            forward
57                .entry(code.as_char())
58                .and_modify(|slot| {
59                    let candidate = entry.name.as_ref();
60                    if candidate.len() < slot.len() || (candidate.len() == slot.len() && candidate > *slot) {
61                        *slot = candidate;
62                    }
63                })
64                .or_insert_with(|| entry.name.as_ref());
65        }
66        output.push_str(&format!("pub static {}: phf::Map<char, &'static str> = phf_map! {{\n", Naming::FORWARD));
67        for (ch, name) in &forward {
68            output.push_str(&format!("    '\\u{{{:X}}}' => \"{}\",\n", *ch as u32, name));
69        }
70        output.push_str("};\n\n");
71
72        // Name -> codepoint map (reverse lookup)
73        output.push_str(&format!("pub static {}: phf::Map<&'static str, char> = phf_map! {{\n", Naming::REVERSE));
74        for entry in entries {
75            let Some(code) = &entry.code else { continue };
76            let ch = code.as_char();
77            output.push_str(&format!("    \"{}\" => '\\u{{{:X}}}',\n", entry.name.as_ref(), ch as u32));
78        }
79        output.push_str("};\n");
80
81        Ok(output)
82    }
83
84    fn output_path() -> Result<PathBuf, GenerateError> {
85        let out_dir = std::env::var("OUT_DIR")?;
86        Ok(PathBuf::from(out_dir).join("codemap.rs"))
87    }
88}

90#[cfg(test)]
91mod tests {
92    use super::CodemapGenerator;
93    use super::DefaultNaming;
94    use super::GlyphGenerator;
95    use fonts_rs_model::CodePoint;
96    use fonts_rs_model::GlyphEntry;
97
98    /// Style-variant pairs share one codepoint (Line Awesome
99    /// `la-adjust` / `la-adjust-solid` -> U+F042): the forward map
100    /// must emit the key exactly once - `phf_map!` rejects duplicates -
101    /// while the reverse map keeps every exported name.
102    #[test]
103    fn shared_codepoints_dedupe_forward_map() {
104        let entries = vec![
105            GlyphEntry::new(Some(CodePoint::from_char('\u{F042}')), "x-adjust-symbolic".to_string(), "/pfx", "icons"),
106            GlyphEntry::new(Some(CodePoint::from_char('\u{F042}')), "x-adjust-solid-symbolic".to_string(), "/pfx", "icons"),
107        ];
108        let output = CodemapGenerator::<DefaultNaming>::generate(&entries).expect("generate");
109        assert_eq!(output.matches("'\\u{F042}' =>").count(), 1, "forward map must not contain a duplicate key");
110        assert!(output.contains("\"x-adjust-solid-symbolic\" => '\\u{F042}'"), "reverse map must keep every exported name");
111        assert!(
112            output.contains("'\\u{F042}' => \"x-adjust-symbolic\""),
113            "canonical (shorter) name should win the shared codepoint"
114        );
115    }
