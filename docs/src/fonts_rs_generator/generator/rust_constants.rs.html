1//! Generator for Rust glyph name constants.
2
3use super::error::GenerateError;
4use super::generate::GlyphGenerator;
5use fonts_rs_model::GlyphEntry;
6use serde::Deserialize;
7use serde::Serialize;
8use std::collections::HashSet;
9use std::path::PathBuf;
10
11/// Generates Rust constants for each glyph name.
12pub struct RustConstantsGenerator;
13
14impl<N> GlyphGenerator<N> for RustConstantsGenerator
15where
16    N: AsRef<str> + Clone + Ord + Serialize + for<'a> Deserialize<'a>,
17{
18    fn generate(entries: &[GlyphEntry<N>]) -> Result<String, GenerateError> {
19        let names: HashSet<String> = entries.iter().map(|entry| entry.name.as_ref().to_string()).collect();
20
21        let mut output = String::new();
22        output.push_str("// @generated by build.rs — do not edit\n\n");
23
24        for name in names {
25            let const_name = name.to_uppercase().replace('-', "_");
26            output.push_str(&format!("pub const {}: &str = \"{}\";\n", const_name, name));
27        }
28
29        Ok(output)
30    }
31
32    fn output_path() -> Result<PathBuf, GenerateError> {
33        let out_dir = std::env::var("OUT_DIR")?;
34        Ok(PathBuf::from(out_dir).join("icons.rs"))
35    }
36}
