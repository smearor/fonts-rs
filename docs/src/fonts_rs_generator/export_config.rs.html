20/// Runtime configuration for glyph export, parameterized by a [`FontFamilyConfig`].
21///
22/// Allows build scripts to export glyphs with variant-specific parameters
23/// determined at runtime (e.g. from Cargo features) without needing a
24/// separate `FontDefinition` impl per variant.
25///
26/// The `X` type parameter provides compile-time constants (`GRESOURCE_PREFIX`,
27/// `ICONS_CONTEXT`, `CODEPOINT_RANGES`) while the struct fields hold
28/// runtime-determined values (variant prefix, axis settings, etc.).
29pub struct ExportConfig<X: FontFamilyConfig> {
30    /// GResource prefix, e.g. `/io/smearor/fonts/dseg7/classic_regular`.
31    ///
32    /// This is the full prefix including any variant slug, constructed by
33    /// [`ExportConfig::new`] or [`ExportConfig::with_variant`] from
34    /// `X::GRESOURCE_PREFIX`.
35    pub gresource_prefix: String,
36    /// Glyph name prefix, e.g. `dseg7-classic-regular`.
37    pub glyph_name_prefix: String,
38    /// Variable font axis settings in user space (e.g. `wght=700.0, ROND=50.0`).
39    ///
40    /// Empty for non-variable fonts (renders at default location).
41    pub axes: AxisValues,
42    /// Custom glyph name filter, returning `true` if the glyph should be exported.
43    ///
44    /// If `None`, uses [`default_name_filter`] which skips names starting with
45    /// `.`, `uni`, or `u` (auto-generated PostScript names).
46    ///
47    /// Fonts with legitimate glyph names starting with `u` (e.g. SMuFL's
48    /// `unison`, `upBow`) should provide a custom filter.
49    pub name_filter: Option<fn(&str) -> bool>,
50    /// Human-readable font family name for diagnostics and generated code comments,
51    /// e.g. `"DSEG7"`, `"Doto"`, `"Bravura"`.
52    pub family_display_name: String,
53    /// Marker for the [`FontFamilyConfig`] type parameter.
54    pub(crate) _marker: PhantomData<X>,
55}
56
57/// Default glyph name filter: skips auto-generated PostScript names.
58///
59/// Returns `false` for names starting with `.`, `uni`, or `u`.
60pub fn default_name_filter(name: &str) -> bool {
61    !(name.starts_with('.') || name.starts_with("uni") || name.starts_with("u"))
62}
63
64impl<X: FontFamilyConfig> ExportConfig<X> {
65    /// Create a new `ExportConfig` without a variant.
66    ///
67    /// Uses `X::FONT_FAMILY_NAME` as the glyph name prefix and
68    /// `X::GRESOURCE_PREFIX` as the GResource prefix, with empty axis values.
69    pub fn new() -> Self {
70        Self {
71            gresource_prefix: X::GRESOURCE_PREFIX.to_string(),
72            glyph_name_prefix: X::FONT_FAMILY_NAME.to_string(),
73            axes: AxisValues::EMPTY,
74            name_filter: None,
75            family_display_name: X::FAMILY_DISPLAY_NAME.to_string(),
76            _marker: PhantomData,
77        }
78    }
79
80    /// Create a new `ExportConfig` with a font variant.
81    ///
82    /// Constructs `glyph_name_prefix` from `X::FONT_FAMILY_NAME` and the variant,
83    /// and `gresource_prefix` from `X::GRESOURCE_PREFIX` and the variant slug.
84    /// Axis values are taken from the variant's [`FontVariant::axis_values`].
85    pub fn with_variant(variant: FontVariant) -> Self {
86        let variant_slug = variant.slug();
87        Self {
88            gresource_prefix: format!("{}/{}", X::GRESOURCE_PREFIX, variant_slug),
89            glyph_name_prefix: format!("{}-{}", X::FONT_FAMILY_NAME, variant),
90            axes: variant.axis_values(),
91            name_filter: None,
92            family_display_name: X::FAMILY_DISPLAY_NAME.to_string(),
93            _marker: PhantomData,
94        }
95    }
96
97    /// Export all glyphs from a TTF/OTF font file.
98    ///
99    /// Generates:
100    /// - `<output_dir>/scalable/{context}/*.svg` - one SVG file per glyph
101    /// - `<output_dir>/metadata.json` - glyph metadata
102    /// - `<output_dir>/icons.gresource.xml` - GResource bundle manifest
103    ///
104    /// # Arguments
105    ///
106    /// - `font_path` - Path to the TTF/OTF font file
107    /// - `output_dir` - Root output directory (typically `resources/`)
108    ///
109    /// # Returns
110    ///
111    /// The number of exported glyphs on success, or an [`ExportError`] on failure.
112    pub fn export_glyphs(&self, font_path: &Path, output_dir: &Path) -> Result<usize, ExportError> {
113        Ok(FontFileSource::new(font_path, self)?.export(self, output_dir)?)
114    }
115
116    /// Export glyphs from a font using an external name→codepoint map.
117    ///
118    /// For fonts that lack PostScript glyph names (e.g. SMuFL fonts with
119    /// `post` table version 3.0), this method uses an externally provided
120    /// mapping of glyph names to Unicode codepoints (e.g. from SMuFL's
121    /// `glyphnames.json`) to drive the export.
122    ///
123    /// For each entry in `name_map`, the method:
124    /// 1. Looks up the glyph ID via the font's `cmap` table
125    /// 2. Renders the glyph to SVG
126    /// 3. Names the file using the kebab-cased glyph name prefixed with `glyph_name_prefix`
127    ///
128    /// Generates the same outputs as [`export_glyphs`](Self::export_glyphs):
129    /// - `<output_dir>/scalable/{context}/*.svg`
130    /// - `<output_dir>/metadata.json`
131    /// - `<output_dir>/icons.gresource.xml`
132    ///
133    /// # Arguments
134    ///
135    /// - `font_path` - Path to the TTF/OTF font file
136    /// - `output_dir` - Root output directory (typically `resources/`)
137    /// - `name_map` - Mapping of glyph names to Unicode codepoints
138    ///
139    /// # Returns
140    ///
141    /// The number of exported glyphs on success, or an [`ExportError`] on failure.
142    pub fn export_glyphs_by_name_map(&self, font_path: &Path, output_dir: &Path, name_map: &GlyphNameMap) -> Result<usize, ExportError> {
143        Ok(FontFileSource::new(font_path, self)?.name_map(name_map).export(self, output_dir)?)
144    }
145
146    /// Write `variant.rs` with `GRESOURCE_PREFIX` and `GLYPH_PREFIX` constants.
147    ///
148    /// Generates a Rust source file in `OUT_DIR/variant.rs` containing the
149    /// GResource prefix and glyph name prefix as `pub const` strings, for use
150    /// by the crate's `lib.rs` at compile time.
151    pub fn write_variant_info(&self) -> miette::Result<()> {
152        let out_dir = std::env::var("OUT_DIR").map_err(|e| miette::miette!("OUT_DIR not set: {e}"))?;
153        let variant_info = format!(
154            "// @generated by build.rs - do not edit\n\n\
155             /// GResource prefix for {}.\n\
156             pub const GRESOURCE_PREFIX: &str = \"{}\";\n\n\
157             /// Glyph name prefix for {}.\n\
158             pub const GLYPH_PREFIX: &str = \"{}\";\n",
159            self.family_display_name, self.gresource_prefix, self.family_display_name, self.glyph_name_prefix,
160        );
161        fs::write(Path::new(&out_dir).join("variant.rs"), variant_info).map_err(|e| miette::miette!("Failed to write variant.rs: {e}"))?;
162        Ok(())
163    }
164}

166impl<X: FontFamilyConfig> Default for ExportConfig<X> {
167    fn default() -> Self {
168        Self::new()
169    }
