1//! The [`IconSetBuildError`] error type for custom icon set builds.
2
3use crate::generator::error::GenerateError;
4use crate::glyph_source::error::GlyphSourceError;
5use fonts_rs_model::IconSetSpecError;
6use thiserror::Error;
7
8/// Error returned by [`IconSetBuild`](crate::icon_set::IconSetBuild)
9/// operations.
10///
11/// Implements `miette::Diagnostic`, so `?` converts directly into
12/// `miette::Result` - the `build.rs` one-liner works unchanged.
13#[derive(Debug, Error, miette::Diagnostic)]
14pub enum IconSetBuildError {
15    /// The `icons.toml` spec file does not exist.
16    #[error("icon set spec not found: {path}")]
17    SpecNotFound {
18        /// Path that was looked up.
19        path: String,
20    },
21
22    /// The `icons.toml` spec file failed to parse.
23    #[error("failed to parse icon set spec {path}: {parse_error}")]
24    SpecParse {
25        /// Path of the spec file.
26        path: String,
27        /// Underlying TOML parse error.
28        parse_error: toml::de::Error,
29    },
30
31    /// The spec failed schema validation (naming conventions, empty
32    /// sources, duplicates, name collisions).
33    #[error(transparent)]
34    Spec(#[from] IconSetSpecError),
35
36    /// A whitelisted upstream icon matched no file in its source
37    /// directory.
38    #[error("unknown icon '{name}' in {source_id} (closest matches: {suggestions})")]
39    UnknownIcon {
40        /// Source identifier (crate name or directory).
41        source_id: String,
42        /// Whitelisted upstream icon name.
43        name: String,
44        /// Closest matching file stems in the source, or `none`.
45        suggestions: String,
46    },
47
48    /// A source location could not be resolved to a directory.
49    #[error("could not resolve icon source {source_id}: {hint}")]
50    SourceNotFound {
51        /// Source identifier (crate name or directory).
52        source_id: String,
53        /// What was tried (`DEP_*` variables, `cargo metadata`,
54        /// filesystem).
55        hint: String,
56    },
57
58    /// A [`GlyphSource`] export failed.
59    #[error("glyph source error: {0}")]
60    GlyphSource(#[from] GlyphSourceError),
61
62    /// An I/O error occurred while reading sources or writing output.
63    #[error("I/O error: {0}")]
64    Io(#[from] std::io::Error),
65
66    /// JSON serialization of the glyph entries failed.
67    #[error("JSON error: {0}")]
68    Json(#[from] serde_json::Error),
69
70    /// Code generation failed.
71    #[error("code generation failed: {0}")]
72    Generate(#[from] GenerateError),
73}
