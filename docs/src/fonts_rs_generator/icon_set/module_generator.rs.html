1//! The [`IconSetModuleGenerator`] - emits `iconset.rs` into `OUT_DIR`.
2
3use crate::generator::error::GenerateError;
4use crate::generator::metadata_generator::escape_str;
5use fonts_rs_model::IconSetSpec;
6use std::fs;
7use std::path::PathBuf;
8
9/// Generates `iconset.rs` - the complete, self-contained module an
10/// application includes with one line:
11///
12/// ```ignore
13/// include!(concat!(env!("OUT_DIR"), "/iconset.rs"));
14/// ```
15///
16/// The generated module contains the marker type, `sealed::Sealed` /
17/// `FontFamilyMarker` impls, a `FontFamilyConfig` impl built from the
18/// spec values, the `impl_font_family!`/`impl_gtk_font!` invocations,
19/// and the `codemap`/`icon_names` module includes.
20pub struct IconSetModuleGenerator;
21
22impl IconSetModuleGenerator {
23    /// Writes `iconset.rs` into `OUT_DIR`.
24    ///
25    /// # Errors
26    ///
27    /// Returns [`GenerateError`] when `OUT_DIR` is unset or the write
28    /// fails.
29    pub fn run(spec: &IconSetSpec) -> Result<(), GenerateError> {
30        let out_dir = std::env::var("OUT_DIR")?;
31        fs::write(PathBuf::from(out_dir).join("iconset.rs"), Self::generate(spec))?;
32        Ok(())
33    }
34
35    /// The name alias type: `{Name}IconName`, stripping a trailing
36    /// `Icons` plural so `MyAppIcons` produces `MyAppIconName`.
37    fn name_alias(marker_name: &str) -> String {
38        let base = marker_name.strip_suffix("Icons").unwrap_or(marker_name);
39        format!("{base}IconName")
40    }
41
42    /// Generates the module source for the spec.
43    ///
44    /// `display_name` is free text and is string-escaped; the remaining
45    /// values are validation-gated formats (PascalCase marker, kebab
46    /// prefix, `/…` path).
47    pub fn generate(spec: &IconSetSpec) -> String {
48        let name = &spec.name;
49        let alias = Self::name_alias(name);
50        let display_name = escape_str(&spec.display_name);
51        format!(
52            "// @generated by fonts-rs-generator IconSetBuild - do not edit\n\
53             // Regenerate via `IconSetBuild::from_file(\"icons.toml\")` in build.rs.\n\n\
54             /// Build-time configuration for the `{icon_prefix}` custom icon set.\n\
55             pub struct {name}Config;\n\n\
56             impl ::fonts_rs_model::FontFamilyConfig for {name}Config {{\n\
57             \x20   /// Cargo feature / icon set identifier.\n\
58             \x20   const FONT_FAMILY_NAME: &'static str = \"{icon_prefix}\";\n\
59             \x20   /// Human-readable display name.\n\
60             \x20   const FAMILY_DISPLAY_NAME: &'static str = \"{display_name_escaped}\";\n\
61             \x20   /// GResource path prefix for this icon set's resources.\n\
62             \x20   const GRESOURCE_PREFIX: &'static str = \"{gresource_prefix}\";\n\
63             \x20   /// Freedesktop icon context for the exported glyphs.\n\
64             \x20   const ICONS_CONTEXT: &'static str = \"{icons_context}\";\n\
65             \x20   /// Prefix for icon names and CSS classes.\n\
66             \x20   const ICON_PREFIX: &'static str = \"{icon_prefix}-\";\n\
67             }}\n\n\
68             /// Marker type identifying the `{display_name}` icon set in\n\
69             /// `GlyphName<{name}>`.\n\
70             #[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, PartialOrd, Ord)]\n\
71             pub enum {name} {{}}\n\n\
72             impl ::fonts_rs_model::sealed::Sealed for {name} {{}}\n\
73             impl ::fonts_rs_model::FontFamilyMarker for {name} {{}}\n\n\
74             /// Typed glyph name for {display_name} icons.\n\
75             pub type {alias} = ::fonts_rs_model::GlyphName<{name}>;\n\n\
76             mod codepoint_map {{\n\
77             \x20   include!(concat!(env!(\"OUT_DIR\"), \"/codemap.rs\"));\n\
78             }}\n\n\
79             /// Icon name constants for `Image::from_icon_name`.\n\
80             pub mod icon_names {{\n\
81             \x20   include!(concat!(env!(\"OUT_DIR\"), \"/icons.rs\"));\n\
82             }}\n\n\
83             ::fonts_rs_generator::impl_font_family!(\n\
84             \x20   {name},\n\
85             \x20   {name}Config,\n\
86             \x20   glyphs = codepoint_map::GLYPHS,\n\
87             \x20   reverse = codepoint_map::REVERSE_GLYPHS,\n\
88             \x20   name = {alias},\n\
89             \x20   glyph_prefix = \"{icon_prefix}\",\n\
90             \x20   capabilities = [icon_theme, html_svg],\n\
91             );\n\n\
92             #[cfg(feature = \"gtk\")]\n\
93             ::fonts_rs_generator::impl_gtk_font!({name});\n",
94            icon_prefix = spec.icon_prefix,
95            display_name = spec.display_name,
96            display_name_escaped = display_name,
97            gresource_prefix = spec.gresource_prefix,
98            icons_context = spec.icons_context,
99        )
100    }
