1//! The [`IconSetSourceResolver`] - maps `icons.toml` source locations
2//! to absolute SVG directories.
3
4use crate::icon_set::error::IconSetBuildError;
5use fonts_rs_model::IconSetSourceLocation;
6use fonts_rs_model::Package;
7use serde::Deserialize;
8use std::env;
9use std::path::PathBuf;
10use std::process::Command;
11
12/// Resolves [`IconSetSourceLocation`]s to absolute SVG directories.
13///
14/// Resolution order for `crate = "…"` locations:
15///
16/// 1. `DEP_*_SVG_DIR` links metadata from the source crate's own
17///    `build.rs` (primary; requires the crate as a *normal*
18///    dependency - `links` metadata does not propagate through
19///    `[build-dependencies]`).
20/// 2. `cargo metadata` fallback: find the package by name and resolve
21///    `<manifest_dir>/resources/svg`. Emits a `cargo:warning`.
22///
23/// `dir = "…"` locations resolve against `CARGO_MANIFEST_DIR`.
24pub struct IconSetSourceResolver {
25    /// `CARGO_MANIFEST_DIR` of the building crate (current dir as
26    /// fallback for non-build-script contexts, e.g. preview tooling).
27    manifest_dir: PathBuf,
28}
29
30/// Minimal `cargo metadata` output - only the fields we read.
31#[derive(Deserialize)]
32struct CargoMetadata {
33    /// Workspace/registry packages in the dependency graph.
34    packages: Vec<MetadataPackage>,
35}
36
37/// One package entry of `cargo metadata` output.
38#[derive(Deserialize)]
39struct MetadataPackage {
40    /// Package name.
41    name: String,
42    /// Absolute path of the package's `Cargo.toml`.
43    manifest_path: PathBuf,
44}
45
46impl IconSetSourceResolver {
47    /// Creates a resolver rooted at `CARGO_MANIFEST_DIR`.
48    pub fn new() -> Self {
49        Self {
50            manifest_dir: env::var_os("CARGO_MANIFEST_DIR").map_or_else(|| PathBuf::from("."), PathBuf::from),
51        }
52    }
53
54    /// Resolves one source location to an existing directory.
55    ///
56    /// # Errors
57    ///
58    /// Returns [`IconSetBuildError::SourceNotFound`] when no resolution
59    /// path yields an existing directory.
60    pub fn resolve(&self, location: &IconSetSourceLocation) -> Result<PathBuf, IconSetBuildError> {
61        match location {
62            IconSetSourceLocation::Dir { dir } => {
63                let resolved = self.manifest_dir.join(dir);
64                if resolved.is_dir() {
65                    Ok(resolved)
66                } else {
67                    Err(IconSetBuildError::SourceNotFound {
68                        source_id: location.label(),
69                        hint: format!("resolved to {}, which is not a directory", resolved.display()),
70                    })
71                }
72            }
73            IconSetSourceLocation::Crate { name, subdirectory } => self.resolve_crate(location, name, subdirectory.as_deref()),
74        }
75    }
76
77    /// Resolves a `crate = "…"` source: `DEP_*` links metadata first,
78    /// `cargo metadata` as fallback.
79    fn resolve_crate(&self, location: &IconSetSourceLocation, name: &Package, subdirectory: Option<&str>) -> Result<PathBuf, IconSetBuildError> {
80        let mut tried = Vec::new();
81        for var in name.dep_env_vars("SVG_DIR") {
82            tried.push(var.clone());
83            if let Some(dir) = env::var_os(&var) {
84                return self.check_dir(PathBuf::from(dir), subdirectory, location);
85            }
86        }
87
88        if let Some(svg_root) = self.resolve_via_cargo_metadata(name.as_str()) {
89            println!("cargo:warning=icons.toml: '{name}' resolved via `cargo metadata` (no DEP_* links metadata)");
90            return self.check_dir(svg_root, subdirectory, location);
91        }
92        tried.push("cargo metadata".to_string());
93
94        Err(IconSetBuildError::SourceNotFound {
95            source_id: location.label(),
96            hint: format!("tried {}; is '{name}' a dependency of this crate?", tried.join(", ")),
97        })
98    }
99
100    /// Joins the optional `subdirectory` onto a resolved SVG root and
101    /// verifies the result exists.
102    fn check_dir(&self, svg_root: PathBuf, subdirectory: Option<&str>, location: &IconSetSourceLocation) -> Result<PathBuf, IconSetBuildError> {
103        let dir = match subdirectory {
104            Some(sub) => svg_root.join(sub),
105            None => svg_root,
106        };
107        if dir.is_dir() {
108            Ok(dir)
109        } else {
110            Err(IconSetBuildError::SourceNotFound {
111                source_id: location.label(),
112                hint: format!("resolved to {}, which is not a directory", dir.display()),
113            })
114        }
115    }
116
117    /// `cargo metadata` fallback: finds the package by name and returns
118    /// its `resources/svg` root.
119    fn resolve_via_cargo_metadata(&self, name: &str) -> Option<PathBuf> {
120        let cargo = env::var_os("CARGO").unwrap_or_else(|| "cargo".into());
121        let output = Command::new(cargo).args(["metadata", "--format-version", "1"]).output().ok()?;
122        if !output.status.success() {
123            return None;
124        }
125        let metadata: CargoMetadata = serde_json::from_slice(&output.stdout).ok()?;
126        metadata
127            .packages
128            .iter()
129            .find(|package| package.name == name)
130            .and_then(|package| package.manifest_path.parent().map(|dir| dir.join("resources/svg")))
131    }
132}
133
134impl Default for IconSetSourceResolver {
135    fn default() -> Self {
136        Self::new()
137    }
138}
