27/// Computes a fast FNV-1a-like hash of a font file.
28///
29/// Used by `build.rs` to detect whether the font has changed and
30/// glyph export needs to re-run.
31pub fn hash_font_file(path: impl AsRef<std::path::Path>) -> String {
32    use std::fs;
33    use std::io::Read;
34
35    let mut file = fs::File::open(path).expect("Failed to open font file");
36    let mut buffer = [0u8; 8192];
37    let mut hash: u64 = 0xcbf29ce484222325;
38    while let Ok(n) = file.read(&mut buffer) {
39        if n == 0 {
40            break;
41        }
42        for &byte in &buffer[..n] {
43            hash ^= byte as u64;
44            hash = hash.wrapping_mul(0x100000001b3);
45        }
46    }
47    format!("{hash:016x}")
48}
