1//! Build-time extension trait for font families.
2//!
3//! Provides [`FontFamilyConfigExt`] as an extension trait for any type
4//! implementing [`FontFamilyConfig`](fonts_rs_model::FontFamilyConfig).
5//! The base trait lives in `fonts-rs-model`; this module adds build-time
6//! filesystem operations (directory creation, cleanup).
7
8use std::fs;
9use std::path::Path;
10use std::path::PathBuf;
11
12use fonts_rs_model::FontFamilyConfig;
13use fonts_rs_model::SCALABLE_DIR;
14
15/// Build-time extension trait for [`FontFamilyConfig`].
16///
17/// Adds filesystem operations for preparing icon output directories during
18/// the build pipeline. Implemented blanketly for all `T: FontFamilyConfig`.
19pub trait FontFamilyConfigExt: FontFamilyConfig {
20    /// Build the icons output directory: `{output_dir}/{SCALABLE_DIR}/{ICONS_CONTEXT}`.
21    ///
22    /// Follows the Freedesktop Icon Theme Specification used by GTK 4's
23    /// `GtkIconTheme`.
24    fn icons_dir(output_dir: &Path) -> PathBuf {
25        output_dir.join(SCALABLE_DIR).join(Self::ICONS_CONTEXT)
26    }
27
28    /// Prepare the icons output directory for a fresh glyph export.
29    ///
30    /// Removes any existing icons directory (including stale SVGs from
31    /// previous builds) and recreates it. Returns the cleaned directory path.
32    fn prepare_icons_dir(output_dir: &Path) -> std::io::Result<PathBuf> {
33        let icons_dir = Self::icons_dir(output_dir);
34        if icons_dir.exists() {
35            fs::remove_dir_all(&icons_dir)?;
36        }
37        fs::create_dir_all(&icons_dir)?;
38        Ok(icons_dir)
39    }
40}
41
42impl<T: FontFamilyConfig> FontFamilyConfigExt for T {}
