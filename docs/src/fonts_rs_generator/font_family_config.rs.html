1//! Base configuration trait for font families.
2//!
3//! Provides compile-time constants shared by both [`FontDefinition`](crate::FontDefinition)
4//! (type-safe glyph name pipeline) and [`ExportConfig`](crate::ExportConfig)
5//! (runtime variant pipeline).
6
7use std::fs;
8use std::path::Path;
9use std::path::PathBuf;
10
11use fonts_rs_model::BMP_RANGE;
12use fonts_rs_model::CodePointRange;
13use fonts_rs_model::ICONS_CONTEXT_GLYPHS;
14use fonts_rs_model::SCALABLE_DIR;
15
16/// Base configuration for font families.
17///
18/// Implemented by all font family definition types to provide
19/// compile-time constants for GResource prefixes, icon context
20/// directories, and codepoint ranges.
21///
22/// [`FontDefinition`](crate::FontDefinition) extends this trait with
23/// type-safe glyph name handling. [`ExportConfig`](crate::ExportConfig)
24/// is generic over any type implementing this trait.
25pub trait FontFamilyConfig {
26    /// Font family slug used in the GResource prefix and glyph names.
27    ///
28    /// e.g. `"dseg7"` or `"doto"`. Combined with
29    /// [`GRESOURCE_BASE_PREFIX`](fonts_rs_model::GRESOURCE_BASE_PREFIX) to
30    /// produce the full `GRESOURCE_PREFIX`, and used directly as the
31    /// `glyph_name_prefix` base in [`ExportConfig::new`](crate::ExportConfig::new).
32    const FONT_FAMILY_NAME: &'static str;
33
34    /// Human-readable display name for the font family.
35    ///
36    /// e.g. `"DSEG7"` or `"Doto"`. Used in build log messages and
37    /// [`ExportConfig::new`](crate::ExportConfig::new) for the `family_display_name` field.
38    const FAMILY_DISPLAY_NAME: &'static str;
39
40    /// GResource prefix for this font family.
41    ///
42    /// e.g. `/io/smearor/fonts/dseg7` or `/io/smearor/fonts/nerd_fonts`.
43    const GRESOURCE_PREFIX: &'static str;
44
45    /// Icon context subdirectory within the GResource prefix.
46    ///
47    /// Follows the Freedesktop Icon Theme Specification used by GTK 4's
48    /// `GtkIconTheme`: `{prefix}/scalable/{context}/{name}.svg`.
49    ///
50    /// The `scalable` directory signals that icons are vector (SVG) and
51    /// can be rendered at any size. The `context` subdirectory groups
52    /// icons by semantic category (e.g. `glyphs`, `status`, `actions`).
53    ///
54    /// For font glyph icons, [`ICONS_CONTEXT_GLYPHS`] is the default context.
55    /// Font families can override this to use a different context if
56    /// needed (e.g. [`ICONS_CONTEXT_EMOJI`](fonts_rs_model::ICONS_CONTEXT_EMOJI) for Noto Emoji).
57    const ICONS_CONTEXT: &'static str = ICONS_CONTEXT_GLYPHS;
58
59    /// Unicode codepoint ranges to probe when building the reverse cmap.
60    ///
61    /// Each tuple is `(start, end)` inclusive. The generic pipeline probes
62    /// these ranges to map `GlyphId` -> `CodePoint` for each glyph in the
63    /// font.
64    ///
65    /// Defaults to the BMP (`U+0000`–`U+FFFF`), which covers most fonts.
66    /// Font families with glyphs in supplementary planes (e.g. Nerd Fonts
67    /// PUA at `U+F0001`–`U+10FFFF`) should override this.
68    const CODEPOINT_RANGES: &[CodePointRange] = &[BMP_RANGE];
69
70    /// Build the icons output directory: `{output_dir}/{SCALABLE_DIR}/{ICONS_CONTEXT}`.
71    ///
72    /// Follows the Freedesktop Icon Theme Specification used by GTK 4's
73    /// `GtkIconTheme`.
74    fn icons_dir(output_dir: &Path) -> PathBuf {
75        output_dir.join(SCALABLE_DIR).join(Self::ICONS_CONTEXT)
76    }
77
78    /// Prepare the icons output directory for a fresh glyph export.
79    ///
80    /// Removes any existing icons directory (including stale SVGs from
81    /// previous builds) and recreates it. Returns the cleaned directory path.
82    fn prepare_icons_dir(output_dir: &Path) -> std::io::Result<PathBuf> {
83        let icons_dir = Self::icons_dir(output_dir);
84        if icons_dir.exists() {
85            fs::remove_dir_all(&icons_dir)?;
86        }
87        fs::create_dir_all(&icons_dir)?;
88        Ok(icons_dir)
89    }
90
91    /// Build the GResource sub-prefix for icons: `{GRESOURCE_PREFIX}/{SCALABLE_DIR}/{ICONS_CONTEXT}`.
92    ///
93    /// Used in `icons.gresource.xml` and resource paths.
94    fn icons_resource_prefix() -> String {
95        format!("{}/{}/{}", Self::GRESOURCE_PREFIX, SCALABLE_DIR, Self::ICONS_CONTEXT)
96    }
97}
