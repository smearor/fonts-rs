1//! Font family runtime API macro.
2//!
3//! Provides the [`impl_font_family!`] macro for generating the runtime
4//! font family API on a marker type: the
5//! [`FontFamilyConfig`](::fonts_rs_model::FontFamilyConfig) delegation,
6//! the [`FontFamily`](::fonts_rs_model::FontFamily) core impl wiring the
7//! generated phf maps, the declared capability impls, and inherent
8//! forwarding methods so call sites need no trait imports.
9//!
10//! Each font crate calls `impl_font_family!` once in a `src/family.rs`
11//! file. The other macros keep their roles and stay compositional:
12//! [`impl_gtk_font!`](crate::impl_gtk_font) provides the `GtkFont`
13//! registration primitives, [`impl_font_loader!`](crate::impl_font_loader)
14//! provides `SoftwareRender`.
15
16/// Generates the runtime font family API for a marker type.
17///
18/// # Invocation
19///
20/// ```ignore
21/// fonts_rs_generator::impl_font_family!(
22///     Code39,
23///     Code39Definition,
24///     glyphs = codepoint_map::GLYPHS,
25///     reverse = codepoint_map::REVERSE_GLYPHS,
26///     name = Code39Name,
27///     css_font_family = "LibreBarcode39", // optional
28///     capabilities = [icon_theme, pango, html],
29/// );
30/// ```
31///
32/// `css_font_family` is optional: it overrides
33/// [`FontFamily::CSS_FONT_FAMILY`](::fonts_rs_model::FontFamily::CSS_FONT_FAMILY)
34/// when the TTF's family name differs from `FAMILY_DISPLAY_NAME`.
35///
36/// # Emitted items
37///
38/// - `impl FontFamilyConfig for $marker` - delegates all constants to
39///   `$definition`.
40/// - `impl FontFamily for $marker` - `type Name = $name`, glyph lookups
41///   via the `$glyphs` (`phf::Map<char, &'static str>`) and `$reverse`
42///   (`phf::Map<&'static str, char>`) maps.
43/// - One empty impl per declared capability - all capability methods
44///   are trait defaults. Valid capabilities: `icon_theme`
45///   ([`IconThemeFont`](::fonts_rs_model::IconThemeFont)), `pango`
46///   ([`PangoFont`](::fonts_rs_model::PangoFont)), `html`
47///   ([`HtmlExport`](::fonts_rs_model::HtmlExport)).
48/// - Inherent forwarding methods for every declared capability plus
49///   `install()`/`ensure_registered()`, so concrete call sites need no
50///   trait import.
51///
52/// Capability impls and forwarders are `#[cfg(feature = "gtk")]`-gated;
53/// the core impl is feature-free. `SoftwareRender` is not a capability
54/// here - it is emitted by [`impl_font_loader!`](crate::impl_font_loader).
55#[macro_export]
56macro_rules! impl_font_family {
57    // Internal: resolves `GRESOURCE_PREFIX` on the marker - the
58    // variant-aware override when `gresource_prefix = ...` was passed
59    // (families exporting under a variant-suffixed prefix via
60    // `ExportConfig::with_variant`), else the definition's static prefix.
61    (@gresource_prefix $definition:ty) => {
62        <$definition as ::fonts_rs_model::FontFamilyConfig>::GRESOURCE_PREFIX
63    };
64    (@gresource_prefix $definition:ty, $prefix:expr) => {
65        $prefix
66    };
67    ($marker:ident, $definition:ty,
68     glyphs = $glyphs:expr,
69     reverse = $reverse:expr,
70     name = $name:ty,
71     $(css_font_family = $css_font_family:expr,)*
72     $(glyph_prefix = $glyph_prefix:expr,)*
73     $(gresource_prefix = $gresource_prefix:expr,)?
74     $(variants = $variants:expr,)?
75     capabilities = [$($capability:ident),* $(,)?]
76     $(,)?) => {
77        impl ::fonts_rs_model::FontFamilyConfig for $marker {
78            const FONT_FAMILY_NAME: &'static str = <$definition as ::fonts_rs_model::FontFamilyConfig>::FONT_FAMILY_NAME;
79            const FAMILY_DISPLAY_NAME: &'static str = <$definition as ::fonts_rs_model::FontFamilyConfig>::FAMILY_DISPLAY_NAME;
80            const GRESOURCE_PREFIX: &'static str =
81                $crate::impl_font_family!(@gresource_prefix $definition $(, $gresource_prefix)?);
82            const ICONS_CONTEXT: &'static str = <$definition as ::fonts_rs_model::FontFamilyConfig>::ICONS_CONTEXT;
83            const ICON_PREFIX: &'static str = <$definition as ::fonts_rs_model::FontFamilyConfig>::ICON_PREFIX;
84            const CODEPOINT_RANGES: &[::fonts_rs_model::CodePointRange] =
85                <$definition as ::fonts_rs_model::FontFamilyConfig>::CODEPOINT_RANGES;
86        }
87
88        impl ::fonts_rs_model::FontFamily for $marker {
89            type Name = $name;
90
91            $(const CSS_FONT_FAMILY: &'static str = $css_font_family;)*
92            $(const VARIANTS: &'static [::fonts_rs_model::FontVariant] = $variants;)?
93
94            fn glyph_name(ch: impl Into<::fonts_rs_model::CodePoint>) -> Option<Self::Name> {
95                let ch = ch.into().as_char();
96                $glyphs.get(&ch).and_then(|name| <Self::Name as ::std::str::FromStr>::from_str(name).ok())
97            }
98
99            fn codepoint(name: impl AsRef<str>) -> Option<::fonts_rs_model::CodePoint> {
100                let name = name.as_ref();
101                if let Some(ch) = $reverse.get(name).copied() {
102                    return Some(::fonts_rs_model::CodePoint::from_char(ch));
103                }
104                $(let full = ::std::format!("{}-{}", $glyph_prefix, name);
105                  if let Some(ch) = $reverse.get(&full).copied() {
106                      return Some(::fonts_rs_model::CodePoint::from_char(ch));
107                  })*
108                None
109            }
110
111            fn all_glyphs() -> impl Iterator<Item = (::fonts_rs_model::CodePoint, Self::Name)> {
112                $glyphs.entries().filter_map(|(ch, name)| {
113                    <Self::Name as ::std::str::FromStr>::from_str(name)
114                        .ok()
115                        .map(|name| (::fonts_rs_model::CodePoint::from_char(*ch), name))
116                })
117            }
118
119            fn glyph_count() -> usize {
120                $glyphs.len()
121            }
122        }
123
124        $($crate::impl_font_family!(@capability $marker $capability);)*
125
126        impl $marker {
127            /// Resolves a glyph name to its Unicode codepoint.
128            ///
129            /// Accepts `&str`, `String`, and the typed glyph name via
130            /// `AsRef<str>`. Forwards to [`FontFamily::codepoint`](::fonts_rs_model::FontFamily::codepoint).
131            pub fn codepoint(name: impl AsRef<str>) -> Option<::fonts_rs_model::CodePoint> {
132                <Self as ::fonts_rs_model::FontFamily>::codepoint(name)
133            }
134
135            /// Resolves a Unicode codepoint to its typed glyph name.
136            ///
137            /// Accepts `char` and `CodePoint` via `Into<CodePoint>`.
138            /// Forwards to [`FontFamily::glyph_name`](::fonts_rs_model::FontFamily::glyph_name).
139            pub fn glyph_name(ch: impl Into<::fonts_rs_model::CodePoint>) -> Option<<Self as ::fonts_rs_model::FontFamily>::Name> {
140                <Self as ::fonts_rs_model::FontFamily>::glyph_name(ch)
141            }
142
143            /// Returns an iterator over all known glyphs, yielding
144            /// `(CodePoint, Self::Name)` pairs of codepoint and typed glyph name.
145            ///
146            /// Forwards to [`FontFamily::all_glyphs`](::fonts_rs_model::FontFamily::all_glyphs).
147            pub fn all_glyphs() -> impl Iterator<Item = (::fonts_rs_model::CodePoint, <Self as ::fonts_rs_model::FontFamily>::Name)> {
148                <Self as ::fonts_rs_model::FontFamily>::all_glyphs()
149            }
150
151            /// Returns the number of glyphs in this family - O(1).
152            ///
153            /// Forwards to [`FontFamily::glyph_count`](::fonts_rs_model::FontFamily::glyph_count).
154            pub fn glyph_count() -> usize {
155                <Self as ::fonts_rs_model::FontFamily>::glyph_count()
156            }
157        }
158
159        #[cfg(feature = "gtk")]
160        impl $marker {
161            /// Explicit fail-fast setup for application startup.
162            ///
163            /// Forwards to [`GtkFont::install`](::fonts_rs_model::GtkFont::install).
164            pub fn install() -> Result<(), ::fonts_rs_model::FontError> {
165                <Self as ::fonts_rs_model::GtkFont>::install()
166            }
167
168            /// Idempotent registration of all GResource bundles.
169            ///
170            /// Forwards to [`GtkFont::ensure_registered`](::fonts_rs_model::GtkFont::ensure_registered).
171            pub fn ensure_registered() -> Result<(), ::fonts_rs_model::FontError> {
172                <Self as ::fonts_rs_model::GtkFont>::ensure_registered()
173            }
174        }
175
176        $($crate::impl_font_family!(@forward $marker $capability);)*
177    };
178
179    (@capability $marker:ident icon_theme) => {
180        #[cfg(feature = "gtk")]
181        impl ::fonts_rs_model::IconThemeFont for $marker {}
182    };
183    (@capability $marker:ident pango) => {
184        #[cfg(feature = "gtk")]
185        impl ::fonts_rs_model::PangoFont for $marker {}
186    };
187    (@capability $marker:ident html) => {
188        #[cfg(feature = "gtk")]
189        impl ::fonts_rs_model::HtmlSvgExport for $marker {}
190        #[cfg(feature = "gtk")]
191        impl ::fonts_rs_model::HtmlExport for $marker {}
192    };
193    (@capability $marker:ident html_svg) => {
194        #[cfg(feature = "gtk")]
195        impl ::fonts_rs_model::HtmlSvgExport for $marker {}
196    };
197    (@capability $marker:ident $unknown:ident) => {
198        ::core::compile_error!(concat!(
199            "impl_font_family!: unknown capability `",
200            stringify!($unknown),
201            "` - expected one of: icon_theme, pango, html, html_svg"
202        ));
203    };
204
205    (@forward $marker:ident icon_theme) => {
206        #[cfg(feature = "gtk")]
207        impl $marker {
208            /// Registers the glyph GResource and adds the family's icon
209            /// resource path to the `GtkIconTheme` of `display`.
210            ///
211            /// Forwards to [`IconThemeFont::install_icon_theme`](::fonts_rs_model::IconThemeFont::install_icon_theme).
212            pub fn install_icon_theme(
213                display: &::fonts_rs_model::gtk4::gdk::Display,
214            ) -> Result<(), ::fonts_rs_model::FontError> {
215                <Self as ::fonts_rs_model::IconThemeFont>::install_icon_theme(display)
216            }
217
218            /// The icon theme resource path of this family.
219            ///
220            /// Forwards to [`IconThemeFont::icon_theme_path`](::fonts_rs_model::IconThemeFont::icon_theme_path).
221            pub fn icon_theme_path() -> String {
222                <Self as ::fonts_rs_model::IconThemeFont>::icon_theme_path()
223            }
224
225            /// Returns the SVG document for one glyph.
226            ///
227            /// Forwards to [`IconThemeFont::glyph_svg`](::fonts_rs_model::IconThemeFont::glyph_svg).
228            pub fn glyph_svg(name: impl AsRef<str>) -> Option<String> {
229                <Self as ::fonts_rs_model::IconThemeFont>::glyph_svg(name)
230            }
231
232            /// Composes `text` into a single SVG document.
233            ///
234            /// Forwards to [`IconThemeFont::compose_svg`](::fonts_rs_model::IconThemeFont::compose_svg).
235            pub fn compose_svg(text: &str, glyph_height: f64) -> String {
236                <Self as ::fonts_rs_model::IconThemeFont>::compose_svg(text, glyph_height)
237            }
238        }
239    };
240    (@forward $marker:ident pango) => {
241        #[cfg(feature = "gtk")]
242        impl $marker {
243            /// Loads all TTFs of the family into `font_map`.
244            ///
245            /// Forwards to [`PangoFont::install_pango`](::fonts_rs_model::PangoFont::install_pango).
246            pub fn install_pango(font_map: &::fonts_rs_model::pango::FontMap) -> Result<(), ::fonts_rs_model::FontError> {
247                <Self as ::fonts_rs_model::PangoFont>::install_pango(font_map)
248            }
249        }
250    };
251    (@forward $marker:ident html) => {
252        #[cfg(feature = "gtk")]
253        impl $marker {
254            /// Returns a `@font-face` CSS rule embedding the
255            /// default-variant TTF as a base64 data URI.
256            ///
257            /// Forwards to [`HtmlExport::css_font_face`](::fonts_rs_model::HtmlExport::css_font_face).
258            pub fn css_font_face() -> Result<String, ::fonts_rs_model::FontError> {
259                <Self as ::fonts_rs_model::HtmlExport>::css_font_face()
260            }
261
262            /// Starts an `HtmlBuilder` for exporting a self-contained
263            /// HTML page.
264            ///
265            /// Forwards to [`HtmlSvgExport::html`](::fonts_rs_model::HtmlSvgExport::html).
266            pub fn html() -> ::fonts_rs_model::HtmlBuilder<Self> {
267                <Self as ::fonts_rs_model::HtmlSvgExport>::html()
268            }
269        }
270    };
271    (@forward $marker:ident html_svg) => {
272        #[cfg(feature = "gtk")]
273        impl $marker {
274            /// Starts an `HtmlBuilder` for exporting a self-contained
275            /// HTML page with inline SVG glyphs.
276            ///
277            /// Forwards to [`HtmlSvgExport::html`](::fonts_rs_model::HtmlSvgExport::html).
278            pub fn html() -> ::fonts_rs_model::HtmlBuilder<Self> {
279                <Self as ::fonts_rs_model::HtmlSvgExport>::html()
280            }
281        }
282    };
283    (@forward $marker:ident $unknown:ident) => {};
