1//! Builder-pattern runner for `build.rs` scripts.
2//!
3//! Encapsulates the common boilerplate shared across all font crate build scripts:
4//! hash-based change detection, conditional glyph export, GResource compilation,
5//! and codepoint map / Rust constants generation.

28/// Builder for the common `build.rs` pipeline.
29///
30/// Handles hash-based change detection, conditional glyph export,
31/// GResource compilation, and code generator invocation.
32///
33/// # Example
34///
35/// ```ignore
36/// // In build.rs:
37/// FontBuild::new("resources/MyFont-Regular.ttf")
38///     .compile_font_gresource()
39///     .run(|font_path, resources_dir| {
40///         MyDefinition::export_glyphs(font_path, resources_dir)
41///     })?;
42/// ```
43pub struct FontBuild {
44    /// Path to the TTF/OTF font file (relative to crate root).
45    ///
46    /// `None` for fontless SVG-only icon sets created via
47    /// [`svg_only`](Self::svg_only) - change detection then relies
48    /// entirely on `extra_hash` components.
49    font_path: Option<PathBuf>,
50    /// Whether to also compile `font.gresource` in addition to `icons.gresource`.
51    compile_font_gresource: bool,
52    /// GResource prefix for the font family (e.g. `/io/smearor/fonts/barcode_ean13`).
53    ///
54    /// When set and `compile_font_gresource` is true, a `font_resource.rs` file
55    /// is generated in `OUT_DIR` containing the full GResource path constant.
56    gresource_prefix: Option<String>,
57    /// Extra hash components for change detection.
58    ///
59    /// When non-empty, the stored hash is `"{font_hash}-{c1}-{c2}-…"` instead
60    /// of just `"{font_hash}"`. This forces a re-export when any component
61    /// changes (e.g. a variant name), even if the font file itself is
62    /// unchanged.
63    extra_hash: Vec<String>,
64    /// Extra `cargo:rerun-if-changed` paths beyond the font file.
65    extra_rerun_if_changed: Vec<PathBuf>,
66    /// Extra GResource bundles to compile beyond `icons.gresource` (and optionally `font.gresource`).
67    additional_gresources: Vec<GResourceSpec>,
68}
69
70impl FontBuild {
71    /// Create a new builder with the given font path.
72    pub fn new(font_path: impl Into<PathBuf>) -> Self {
73        Self {
74            font_path: Some(font_path.into()),
75            compile_font_gresource: false,
76            gresource_prefix: None,
77            extra_hash: Vec::new(),
78            extra_rerun_if_changed: Vec::new(),
79            additional_gresources: Vec::new(),
80        }
81    }
82
83    /// Create a builder for fontless icon sets (SVG-only sources).
84    ///
85    /// There is no font file to hash or re-run on, so change detection
86    /// relies entirely on [`extra_hash`](Self::extra_hash) components -
87    /// callers must hash every input that affects the export (the SVG
88    /// directory via `hash_svg_dir`, metadata files via `hash_str`).
89    /// Run the pipeline with [`run_svg`](Self::run_svg), whose export
90    /// closure receives only the resources directory.
91    pub fn svg_only() -> Self {
92        Self {
93            font_path: None,
94            compile_font_gresource: false,
95            gresource_prefix: None,
96            extra_hash: Vec::new(),
97            extra_rerun_if_changed: Vec::new(),
98            additional_gresources: Vec::new(),
99        }
100    }
101
102    /// Add an extra hash component to change detection.
103    ///
104    /// May be called multiple times; each component is mixed into the stored
105    /// hash. When any component changes (e.g. a variant name), glyphs are
106    /// re-exported even if the font file itself hasn't changed. Useful for
107    /// variable fonts where the same TTF is rendered at different axis
108    /// locations, or for non-font inputs like SVG directories.
109    pub fn extra_hash(mut self, hash: impl Into<String>) -> Self {
110        self.extra_hash.push(hash.into());
111        self
112    }
113
114    /// Enable compilation of `font.gresource` in addition to `icons.gresource`.
115    pub fn compile_font_gresource(mut self) -> Self {
116        self.compile_font_gresource = true;
117        self
118    }
119
120    /// Set the GResource prefix for the font family.
121    ///
122    /// When `compile_font_gresource` is enabled, this triggers generation of
123    /// `OUT_DIR/font_resource.rs` containing:
124    ///
125    /// ```ignore
126    /// pub const FONT_RESOURCE_PATH: &str = "{prefix}/{font_filename}";
127    /// ```
128    ///
129    /// This constant is used by the `impl_gtk_font!` macro's `with_font_gresource`
130    /// variant to look up the font at runtime.
131    pub fn gresource_prefix(mut self, prefix: impl Into<String>) -> Self {
132        self.gresource_prefix = Some(prefix.into());
133        self
134    }
135
136    /// Add an extra `cargo:rerun-if-changed` path.
137    ///
138    /// Can be called multiple times to register multiple paths.
139    pub fn rerun_if_changed(mut self, path: impl Into<PathBuf>) -> Self {
140        self.extra_rerun_if_changed.push(path.into());
141        self
142    }
143
144    /// Add an extra GResource bundle to compile.
145    ///
146    /// Can be called multiple times to register multiple bundles.
147    /// Each bundle is compiled as
148    /// `glib_build_tools::compile_resources(&["resources"], xml, output)`.
149    pub fn additional_gresource(mut self, xml: impl Into<PathBuf>, output: impl Into<PathBuf>) -> Self {
150        self.additional_gresources.push(GResourceSpec::new(xml, output));
151        self
152    }
153
154    /// Run the build pipeline with default code generation.
155    ///
156    /// The `export` closure is called only when the font file has changed
157    /// (detected via FNV-1a hash comparison). It receives the font path
158    /// and resources directory as `&Path` arguments.
159    ///
160    /// After export (if needed), GResources are compiled and codepoint
161    /// maps / Rust constants are generated from `metadata.json`.
162    ///
163    /// # Errors
164    ///
165    /// Returns an error if any file operation or code generation fails,
166    /// or if the builder was created via [`svg_only`](Self::svg_only) -
167    /// fontless builds must use [`run_svg`](Self::run_svg).
168    pub fn run<F, E>(self, export: F) -> Result<(), E>
169    where
170        F: FnOnce(&Path, &Path) -> Result<usize, E>,
171        E: From<std::io::Error>,
172    {
173        self.run_with(export, |json| {
174            let entries: Vec<GlyphEntry<String>> = serde_json::from_str(json).map_err(std::io::Error::other)?;
175            CodemapGenerator::<DefaultNaming>::run(&entries).map_err(|e| std::io::Error::other(e.to_string()))?;
176            RustConstantsGenerator::run(&entries).map_err(|e| std::io::Error::other(e.to_string()))?;
177            Ok(())
178        })
179    }
180
181    /// Run the fontless build pipeline with default code generation.
182    ///
183    /// Like [`run`](Self::run), but for builds created via
184    /// [`svg_only`](Self::svg_only): the `export` closure receives only
185    /// the resources directory since no font path exists.
186    ///
187    /// # Errors
188    ///
189    /// Returns an error if any file operation or code generation fails,
190    /// or if the builder was created via [`new`](Self::new) - fonted
191    /// builds must use [`run`](Self::run).
192    pub fn run_svg<F, E>(self, export: F) -> Result<(), E>
193    where
194        F: FnOnce(&Path) -> Result<usize, E>,
195        E: From<std::io::Error>,
196    {
197        self.run_with(
198            |_font_path, resources_dir| export(resources_dir),
199            |json| {
200                let entries: Vec<GlyphEntry<String>> = serde_json::from_str(json).map_err(std::io::Error::other)?;
201                CodemapGenerator::<DefaultNaming>::run(&entries).map_err(|e| std::io::Error::other(e.to_string()))?;
202                RustConstantsGenerator::run(&entries).map_err(|e| std::io::Error::other(e.to_string()))?;
203                Ok(())
204            },
205        )
206    }
207
208    /// Run the build pipeline with a custom code generation closure.
209    ///
210    /// Like [`run`](Self::run), but the `generate` closure replaces the default
211    /// `CodemapGenerator` + `RustConstantsGenerator` code generation. The closure
212    /// receives the raw `metadata.json` content as a string slice and can
213    /// deserialize it as any type, run arbitrary generators, etc.
214    ///
215    /// # Errors
216    ///
217    /// Returns an error if any file operation or code generation fails.
218    pub fn run_with<F, G, E>(self, export: F, generate: G) -> Result<(), E>
219    where
220        F: FnOnce(&Path, &Path) -> Result<usize, E>,
221        G: FnOnce(&str) -> Result<(), E>,
222        E: From<std::io::Error>,
223    {
224        println!("cargo:rustc-cfg=is_lib");
225        if let Some(font_path) = &self.font_path {
226            println!("cargo:rerun-if-changed={}", font_path.display());
227        }
228
229        for path in &self.extra_rerun_if_changed {
230            println!("cargo:rerun-if-changed={}", path.display());
231        }
232
233        let out = out_dir();
234        let metadata_path = metadata_path();
235        let hash_path = hash_path();
236        let icons_xml = icons_gresource_xml();
237        // Fontless builds have no base hash - the extra_hash components
238        // carry the entire change detection (callers must supply them).
239        let base_hash = self.font_path.as_deref().map(hash_font_file).unwrap_or_default();
240        let current_hash = if self.extra_hash.is_empty() {
241            base_hash
242        } else {
243            format!("{base_hash}-{}", self.extra_hash.join("-"))
244        };
245
246        let needs_export = !metadata_path.exists() || fs::read_to_string(&hash_path).ok().as_deref() != Some(current_hash.as_str());
247
248        if needs_export {
249            eprintln!(
250                "build.rs: exporting glyphs from {}...",
251                self.font_path.as_deref().map_or("(svg-only)", |p| p.to_str().unwrap_or_default())
252            );
253            let count = export(self.font_path.as_deref().unwrap_or_else(|| Path::new("")), &out)?;
254            eprintln!("build.rs: exported {count} glyphs");
255            fs::write(&hash_path, &current_hash)?;
256        }
257
258        let out_str = out.to_str().ok_or_else(|| std::io::Error::other("invalid UTF-8 in OUT_DIR"))?;
259        let icons_xml_str = icons_xml
260            .to_str()
261            .ok_or_else(|| std::io::Error::other("invalid UTF-8 in icons gresource xml path"))?;
262        glib_build_tools::compile_resources(&[out_str], icons_xml_str, ICONS_GRESOURCE);
263
264        if self.compile_font_gresource {
265            let (font_xml_path, font_source_dir) = if let (Some(prefix), Some(file_name)) =
266                (&self.gresource_prefix, self.font_path.as_deref().and_then(|p| p.file_name()).and_then(|n| n.to_str()))
267            {
268                let xml_content = format!(
269                    "<?xml version=\"1.0\" encoding=\"UTF-8\"?>\n<gresources>\n    <gresource prefix=\"{prefix}\">\n        <file alias=\"{file_name}\">{file_name}</file>\n    </gresource>\n</gresources>\n"
270                );
271                let xml_path = out.join("font_gresource.xml");
272                fs::write(&xml_path, xml_content)?;
273
274                let resource_path = format!("{prefix}/{file_name}");
275                let content = format!("// @generated by build.rs - do not edit\n\npub const FONT_RESOURCE_PATH: &str = \"{resource_path}\";\n");
276                fs::write(out.join("font_resource.rs"), content)?;
277
278                // The generated manifest references the bare file name -
279                // its source dir is the font file's parent directory,
280                // which may be a subdirectory of `resources/` (e.g.
281                // `resources/fonts/`). Fall back to the crate root.
282                let source_dir = self.font_path.as_deref().and_then(|p| p.parent()).map(Path::to_path_buf).unwrap_or_default();
283                (xml_path, source_dir)
284            } else {
285                // The static manifest lives in `resources/` and its
286                // <file> paths are relative to it.
287                (PathBuf::from(FONT_GRESOURCE_XML), PathBuf::from(RESOURCES_DIR))
288            };
289
290            let font_xml_str = font_xml_path
291                .to_str()
292                .ok_or_else(|| std::io::Error::other("invalid UTF-8 in font gresource xml path"))?;
293            let font_source_str = font_source_dir
294                .to_str()
295                .ok_or_else(|| std::io::Error::other("invalid UTF-8 in font source dir path"))?;
296            glib_build_tools::compile_resources(&[font_source_str], font_xml_str, FONT_GRESOURCE);
297        }
298
299        for gresource in &self.additional_gresources {
300            let xml = gresource
301                .xml
302                .to_str()
303                .ok_or_else(|| std::io::Error::other("invalid UTF-8 in gresource xml path"))?;
304            let output = gresource
305                .output
306                .to_str()
307                .ok_or_else(|| std::io::Error::other("invalid UTF-8 in gresource output path"))?;
308            glib_build_tools::compile_resources(&[RESOURCES_DIR], xml, output);
309        }
310
311        let json = fs::read_to_string(&metadata_path)?;
312        generate(&json)?;
313
314        Ok(())
315    }
