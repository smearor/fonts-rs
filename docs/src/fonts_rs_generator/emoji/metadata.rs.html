1//! Generator for emoji metadata phf::Map tables (keywords and categories).
2//!
3//! Keywords are sourced from CLDR annotation data (`default` field).
4//! Categories are sourced from Unicode `emoji-test.txt` (`group` field).
5//!
6//! Implements [`MetadataGenerator`] for emoji families - keyword and
7//! category maps are keyed by Unicode codepoint, so the generator works
8//! for any font covering the emoji ranges (Noto Emoji, Blobmoji, ...).
9
10use crate::MetadataGenerator;
11use fonts_rs_model::CodePointCategoryMap;
12use fonts_rs_model::CodePointKeywordMap;
13use fonts_rs_model::GlyphEntry;
14
15/// Generates phf::Map constants for emoji keywords and categories.
16///
17/// Keywords come from CLDR annotation data. Categories come from
18/// Unicode emoji-test.txt group headers.
19pub struct EmojiMetadataGenerator {
20    keyword_map: CodePointKeywordMap,
21    category_map: CodePointCategoryMap,
22    alias_pairs: Vec<(String, String)>,
23}
24
25impl EmojiMetadataGenerator {
26    /// Creates a new metadata generator with the given keyword and category maps.
27    pub fn new(keyword_map: CodePointKeywordMap, category_map: CodePointCategoryMap) -> Self {
28        Self {
29            keyword_map,
30            category_map,
31            alias_pairs: Vec::new(),
32        }
33    }
34
35    /// Adds `(alias, canonical_name)` pairs to the generated `aliases.rs`
36    /// map - e.g. shortcodes/emoticons from the emoji ordering file.
37    pub fn with_aliases(mut self, alias_pairs: Vec<(String, String)>) -> Self {
38        self.alias_pairs = alias_pairs;
39        self
40    }
41}
42
43impl MetadataGenerator for EmojiMetadataGenerator {
44    type Name = String;
45
46    fn keywords_for(&self, entry: &GlyphEntry<String>) -> Vec<String> {
47        let Some(code) = &entry.code else { return Vec::new() };
48        self.keyword_map.get(code).cloned().unwrap_or_default()
49    }
50
51    fn categories_for(&self, entry: &GlyphEntry<String>) -> Vec<String> {
52        let Some(code) = &entry.code else { return Vec::new() };
53        self.category_map.get(code).map(|c| vec![c.clone()]).unwrap_or_default()
54    }
55
56    fn aliases(&self) -> Vec<(String, String)> {
57        self.alias_pairs.clone()
58    }
59}
