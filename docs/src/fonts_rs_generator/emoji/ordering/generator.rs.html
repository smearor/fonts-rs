1//! Codegen for emoji picker ordering data (`emoji_*_ordering.json`).
2//!
3//! [`EmojiOrderingGenerator`] takes a parsed [`EmojiOrdering`] plus a
4//! family-provided sequence→glyph-name resolver and emits the static
5//! tables the runtime ordering API needs:
6//!
7//! - `ordering.rs` - `ORDERING: phf::Map<&str, u32>` (base name → flat
8//!   picker index over all ordered base entries)
9//! - `groups.rs` - `GROUP_ORDER: &[GroupLabel]`, `GROUPS: &[GlyphGroup]`
10//!   (ordered groups with resolved names, alternates inlined after
11//!   their base) and `GROUP_OF: phf::Map<&str, &str>` (base and
12//!   alternate names → group label)
13//! - `alternates.rs` - `ALTERNATES: phf::Map<&str, &[&str]>` (base →
14//!   variant names) and `BASE_OF: phf::Map<&str, &str>` (variant → base)
15//! - `shortcodes.rs` - `SHORTCODES: phf::Map<&str, &[&str]>` (name →
16//!   shortcodes + emoticons)
17//! - `tone_groups.rs` - `TONE_GROUP: phf::Map<&str, u32>` (base name →
18//!   opaque variant-shape group id; `0`/no tones is omitted)
19//!
20//! [`EmojiOrderingGenerator::run`] additionally returns the
21//! shortcode/emoticon alias pairs so callers can feed them into a
22//! [`MetadataGenerator`](crate::MetadataGenerator) alias map - search
23//! hits `:smile:` without dedicated search code.
24//!
25//! Unresolvable sequences (upstream knows emoji the family does not
26//! ship, or vice versa) are skipped: they get no ordering, group, or
27//! alternate entries.
28
29use std::path::PathBuf;
30
31use fonts_rs_model::CodePoint;
32
33use crate::emoji::ordering::EmojiOrdering;
34use crate::escape_str;
35
36/// Generates the emoji picker ordering tables.
37///
38/// The resolver closure joins a Unicode codepoint sequence onto the
39/// family's exported glyph names - for blobmoji the same CLDR
40/// sequence→name map `SvgDirSource` uses, then prefix + kebab +
41/// membership check against the exported set.
42pub struct EmojiOrderingGenerator<'a> {
43    ordering: &'a EmojiOrdering,
44}
45
46impl<'a> EmojiOrderingGenerator<'a> {
47    /// Creates a generator over a parsed ordering document.
48    pub fn new(ordering: &'a EmojiOrdering) -> Self {
49        Self { ordering }
50    }
51
52    /// Emits `ordering.rs`, `groups.rs`, `alternates.rs`,
53    /// `shortcodes.rs`, and `tone_groups.rs` into `OUT_DIR`.
54    ///
55    /// Returns the `(alias, canonical_name)` pairs collected from
56    /// shortcodes and emoticons - pass them to
57    /// [`EmojiMetadataGenerator::with_aliases`](crate::emoji::EmojiMetadataGenerator::with_aliases)
58    /// (or another `MetadataGenerator`) so `resolve_alias`/`search` see them.
59    ///
60    /// # Errors
61    ///
62    /// Returns an I/O error when `OUT_DIR` is unset or a file cannot be
63    /// written.
64    pub fn run(&self, resolve: impl Fn(&[CodePoint]) -> Option<String>) -> Result<Vec<(String, String)>, std::io::Error> {
65        let out_dir = std::env::var("OUT_DIR").map_err(|e| std::io::Error::other(format!("OUT_DIR not set: {e}")))?;
66        let out_dir = PathBuf::from(out_dir);
67        std::fs::create_dir_all(&out_dir)?;
68
69        // Resolve every sequence once: (flat base index, resolved names).
70        let mut position: u32 = 0;
71        let mut ordering_src = String::from("// @generated by build.rs - do not edit\n\npub static ORDERING: phf::Map<&'static str, u32> = phf::phf_map! {\n");
72        let mut groups_src = String::from("// @generated by build.rs - do not edit\n\npub static GROUP_ORDER: &[GroupLabel] = &[\n");
73        let mut group_of_src = String::from("\npub static GROUP_OF: phf::Map<&'static str, &'static str> = phf::phf_map! {\n");
74        let mut groups_slice_src = String::from("\npub static GROUPS: &[GlyphGroup] = &[\n");
75        let mut alternates_src = String::from(
76            "// @generated by build.rs - do not edit\n\npub static ALTERNATES: phf::Map<&'static str, &'static [&'static str]> = phf::phf_map! {\n",
77        );
78        let mut base_of_src = String::from("\npub static BASE_OF: phf::Map<&'static str, &'static str> = phf::phf_map! {\n");
79        let mut shortcodes_src = String::from(
80            "// @generated by build.rs - do not edit\n\npub static SHORTCODES: phf::Map<&'static str, &'static [&'static str]> = phf::phf_map! {\n",
81        );
82        let mut tone_groups_src =
83            String::from("// @generated by build.rs - do not edit\n\npub static TONE_GROUP: phf::Map<&'static str, u32> = phf::phf_map! {\n");
84        let mut aliases: Vec<(String, String)> = Vec::new();
85        let mut seen_aliases: std::collections::HashSet<&str> = std::collections::HashSet::new();
86        // Distinct upstream sequences can resolve to the same glyph name
87        // (e.g. U+2603/U+26C4 both map to the CLDR name "snowman") -
88        // emit each glyph name once, first occurrence in picker order wins.
89        let mut emitted: std::collections::HashSet<String> = std::collections::HashSet::new();
90        let mut misses: usize = 0;
91
92        for group in &self.ordering.0 {
93            groups_src.push_str(&format!("    GroupLabel::new(\"{}\"),\n", escape_str(&group.group)));
94            let mut group_names: Vec<String> = Vec::new();
95            for emoji in &group.emoji {
96                let base_sequence: Vec<CodePoint> = emoji.base.iter().filter_map(|v| char::from_u32(*v)).map(CodePoint::from).collect();
97                let Some(base_name) = resolve(&base_sequence) else {
98                    misses += 1;
99                    continue;
100                };
101                if !emitted.insert(base_name.clone()) {
102                    continue;
103                }
104
105                ordering_src.push_str(&format!("    \"{}\" => {}u32,\n", escape_str(&base_name), position));
106                position += 1;
107                group_of_src.push_str(&format!("    \"{}\" => \"{}\",\n", escape_str(&base_name), escape_str(&group.group)));
108                // Tone group: opaque shape key for default-tone picking;
109                // 0 means "no tone variants" and stays absent so
110                // `tone_group` can return `None` for plain glyphs.
111                if emoji.tone_group != 0 {
112                    tone_groups_src.push_str(&format!("    \"{}\" => {}u32,\n", escape_str(&base_name), emoji.tone_group));
113                }
114
115                // Variant names: alternates[1..] (alternates[0] is the base
116                // itself in upstream data; the emitted set already filters
117                // it together with any other duplicate resolution).
118                let mut variant_names: Vec<String> = Vec::new();
119                for alternate in &emoji.alternates {
120                    let sequence: Vec<CodePoint> = alternate.iter().filter_map(|v| char::from_u32(*v)).map(CodePoint::from).collect();
121                    let Some(name) = resolve(&sequence) else { continue };
122                    if !emitted.insert(name.clone()) {
123                        continue;
124                    }
125                    group_of_src.push_str(&format!("    \"{}\" => \"{}\",\n", escape_str(&name), escape_str(&group.group)));
126                    base_of_src.push_str(&format!("    \"{}\" => \"{}\",\n", escape_str(&name), escape_str(&base_name)));
127                    variant_names.push(name);
128                }
129
130                group_names.push(base_name.clone());
131                group_names.extend(variant_names.iter().cloned());
132
133                if !variant_names.is_empty() {
134                    let refs: Vec<String> = variant_names.iter().map(|name| format!("\"{}\"", escape_str(name))).collect();
135                    alternates_src.push_str(&format!("    \"{}\" => &[{}],\n", escape_str(&base_name), refs.join(", ")));
136                }
137
138                let mut search_terms: Vec<&str> = emoji.shortcodes.iter().chain(emoji.emoticons.iter()).map(String::as_str).collect();
139                if !search_terms.is_empty() {
140                    search_terms.sort_unstable();
141                    search_terms.dedup();
142                    let refs: Vec<String> = search_terms.iter().map(|term| format!("\"{}\"", escape_str(term))).collect();
143                    shortcodes_src.push_str(&format!("    \"{}\" => &[{}],\n", escape_str(&base_name), refs.join(", ")));
144                    for term in &search_terms {
145                        // The same shortcode can map to several emoji
146                        // (e.g. ':snowman-with-snow:') - first base in
147                        // picker order wins the alias.
148                        if seen_aliases.insert(*term) {
149                            aliases.push(((*term).to_string(), base_name.clone()));
150                        }
151                    }
152                }
153            }
154            if !group_names.is_empty() {
155                let refs: Vec<String> = group_names.iter().map(|name| format!("\"{}\"", escape_str(name))).collect();
156                groups_slice_src.push_str(&format!("    GlyphGroup::new(GroupLabel::new(\"{}\"), &[{}]),\n", escape_str(&group.group), refs.join(", ")));
157            }
158        }
159
160        groups_src.push_str("];\n");
161        groups_src.push_str(&group_of_src);
162        groups_src.push_str("};\n");
163        groups_src.push_str(&groups_slice_src);
164        groups_src.push_str("];\n");
165
166        ordering_src.push_str("};\n");
167        alternates_src.push_str("};\n");
168        alternates_src.push_str(&base_of_src);
169        alternates_src.push_str("};\n");
170        shortcodes_src.push_str("};\n");
171        tone_groups_src.push_str("};\n");
172
173        std::fs::write(out_dir.join("ordering.rs"), ordering_src)?;
174        std::fs::write(out_dir.join("groups.rs"), groups_src)?;
175        std::fs::write(out_dir.join("alternates.rs"), alternates_src)?;
176        std::fs::write(out_dir.join("shortcodes.rs"), shortcodes_src)?;
177        std::fs::write(out_dir.join("tone_groups.rs"), tone_groups_src)?;
178
179        if misses > 0 {
180            eprintln!("ordering generator: {misses} unresolved base sequences skipped");
181        }
182        Ok(aliases)
183    }
184}
