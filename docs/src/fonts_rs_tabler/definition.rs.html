1//! `FontFamilyConfig` for the Tabler Icons font family.
2
3use const_format::concatcp;
4use fonts_rs_model::CodePoint;
5use fonts_rs_model::CodePointRange;
6use fonts_rs_model::FontFamilyConfig;
7use fonts_rs_model::GRESOURCE_BASE_PREFIX;
8
9/// Tabler codepoint ranges: the Unicode Private Use Area plus the
10/// supplementary range upstream allocates for newer icons beyond U+FFFF.
11const TABLER_RANGES: &[CodePointRange] = &[
12    CodePointRange::new(CodePoint::from_char('\u{E000}'), CodePoint::from_char('\u{F8FF}')),
13    CodePointRange::new(CodePoint::from_char('\u{10000}'), CodePoint::from_char('\u{103FF}')),
14];
15
16/// Build-time configuration for the Tabler Icons font family.
17///
18/// Implements [`FontFamilyConfig`] to provide compile-time constants
19/// for the `ExportConfig`-based build pipeline and the generic
20/// [`IconName`](fonts_rs_model::IconName) normalization (`tabler-`
21/// prefix, `-symbolic` suffix).
22pub struct TablerConfig;
23
24impl FontFamilyConfig for TablerConfig {
25    const FONT_FAMILY_NAME: &'static str = "tabler";
26
27    const FAMILY_DISPLAY_NAME: &'static str = "Tabler Icons";
28
29    const GRESOURCE_PREFIX: &'static str = concatcp!(GRESOURCE_BASE_PREFIX, "/", TablerConfig::FONT_FAMILY_NAME);
30
31    const CODEPOINT_RANGES: &[CodePointRange] = TABLER_RANGES;
32
33    const ICON_PREFIX: &'static str = "tabler-";
34}
