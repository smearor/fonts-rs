3use serde::Deserialize;

11/// A workspace package name - the crate directory name and crates.io
12/// package name in one (e.g. `fonts-rs-tabler`). Serializes
13/// transparently as the plain name.
14///
15/// Shared by the xtask tooling (crate dirs under `crates/`) and the
16/// icon-set spec (`crate = "…"` source locations, `DEP_*` links
17/// metadata naming).
18#[derive(Debug, Clone, PartialEq, Eq, PartialOrd, Ord, Hash, Deserialize, Serialize)]
19#[serde(transparent)]
20pub struct Package(
21    /// Package name - `crates/<name>` / crates.io identifier.
22    String,
23);
24
25impl Package {
26    /// The package name as a string slice.
27    pub fn as_str(&self) -> &str {
28        &self.0
29    }
30
31    /// `DEP_*` environment variable candidates for a `cargo:<key>`
32    /// metadata value emitted by this crate's `links` build script.
33    ///
34    /// The `links` values of fonts-rs crates drop the `fonts-rs-`
35    /// prefix (e.g. `links = "lucide"` → `DEP_LUCIDE_SVG_DIR`); both
36    /// the stripped and the full package name are tried so foreign
37    /// crates with full-name links values resolve too.
38    pub fn dep_env_vars(&self, key: &str) -> Vec<String> {
39        let upper = |name: &str| name.to_uppercase().replace('-', "_");
40        let stripped = upper(self.0.strip_prefix("fonts-rs-").unwrap_or(&self.0));
41        let mut vars = vec![format!("DEP_{stripped}_{key}")];
42        let full = format!("DEP_{}_{key}", upper(&self.0));
43        if !vars.contains(&full) {
44            vars.push(full);
45        }
46        vars
47    }
48
49    /// The crate directory `<workspace_root>/crates/<name>`, when it
50    /// exists.
51    pub fn crate_dir(&self, workspace_root: &Path) -> Option<PathBuf> {
52        let dir = workspace_root.join("crates").join(&**self);
53        dir.is_dir().then_some(dir)
54    }
55}
56
57impl Deref for Package {
58    type Target = str;
59
60    /// Borrows the package name as `&str`.
61    fn deref(&self) -> &str {
62        &self.0
63    }
64}
65
66impl Display for Package {
67    /// Writes the plain package name, honoring `Formatter`
68    /// width/alignment (report tables pad via `{:<32}`).
69    fn fmt(&self, f: &mut Formatter<'_>) -> std::fmt::Result {
70        self.0.fmt(f)
71    }
72}
73
74impl AsRef<str> for Package {
75    fn as_ref(&self) -> &str {
76        self.as_str()
77    }
78}
79
80impl From<&str> for Package {
81    /// Wraps a package name.
82    fn from(name: &str) -> Self {
83        Self(name.to_string())
84    }
85}

87impl From<String> for Package {
88    /// Wraps a package name.
89    fn from(name: String) -> Self {

91    }
92}

94#[cfg(test)]
95mod tests {
96    use super::*;
97
98    #[test]
99    fn dep_env_vars_prefers_stripped_fonts_rs_prefix() {
100        let vars = Package("fonts-rs-lucide".to_string()).dep_env_vars("SVG_DIR");
101        assert_eq!(vars[0], "DEP_LUCIDE_SVG_DIR");
102        assert_eq!(vars[1], "DEP_FONTS_RS_LUCIDE_SVG_DIR");
103    }
104
105    #[test]
106    fn dep_env_vars_dedupes_foreign_names() {
107        let vars = Package("some-icons".to_string()).dep_env_vars("SVG_DIR");
108        assert_eq!(vars, vec!["DEP_SOME_ICONS_SVG_DIR".to_string()]);
109    }
110
111    #[test]
112    fn crate_dir_resolves_under_workspace_root() {
113        let manifest_dir = Path::new(env!("CARGO_MANIFEST_DIR"));
114        let root = manifest_dir.parent().and_then(Path::parent).expect("workspace root");
115        assert!(Package("fonts-rs-model".to_string()).crate_dir(root).is_some());
116        assert!(Package("fonts-rs-nope".to_string()).crate_dir(root).is_none());
117    }
