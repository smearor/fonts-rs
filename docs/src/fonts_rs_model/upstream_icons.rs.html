1//! The [`UpstreamIcons`] trait - reverse-mapping displayed icon names
2//! to upstream SVG sources.
3
4use crate::FontFamilyConfig;
5use crate::icon_set::IconSetPostProcessor;
6use crate::upstream_icon::UpstreamIcon;
7
8/// Maps a family's exported icon name (`lucide-check-symbolic`) back to
9/// its upstream source - the `resources/svg` subdirectory, file stem
10/// and default post-processors, bundled as [`UpstreamIcon`].
11///
12/// Icon-capable families implement this next to their
13/// `impl_font_family!` invocation. The default path covers flat
14/// single-style families (`ICON_PREFIX` + `-symbolic` stripping);
15/// multi-style families override [`upstream_icon`](Self::upstream_icon)
16/// for per-stem routing (Eva `fill`/`outline`, Fluent
17/// `regular-24`/`filled-24`, Heroicons `micro`/`mini`/`solid`).
18pub trait UpstreamIcons: FontFamilyConfig {
19    /// `resources/svg` subdirectory for single-style families -
20    /// `Some("outline")` for Tabler, `None` for flat roots.
21    const UPSTREAM_SUBDIRECTORY: Option<&'static str> = None;
22
23    /// Post-processors the family's SVG style typically needs.
24    const UPSTREAM_POST_PROCESSORS: &'static [IconSetPostProcessor] = &[];
25
26    /// Strips `ICON_PREFIX` and `-symbolic` from an exported icon
27    /// name - the shared first step of every reverse mapping.
28    /// Returns `None` for names outside the family's convention.
29    fn upstream_stem(icon_name: &str) -> Option<&str> {
30        icon_name
31            .strip_prefix(Self::ICON_PREFIX)?
32            .strip_suffix("-symbolic")
33            .filter(|stem| !stem.is_empty())
34    }
35
36    /// Reverse-maps a displayed icon name to its upstream location.
37    /// `None` for non-conforming names - callers must skip rather
38    /// than emit a wrong stem (a wrong name surfaces as `UnknownIcon`
39    /// in `IconSetBuild`).
40    fn upstream_icon(icon_name: &str) -> Option<UpstreamIcon> {
41        Self::upstream_stem(icon_name).map(|stem| UpstreamIcon {
42            subdirectory: Self::UPSTREAM_SUBDIRECTORY,
43            stem: stem.to_string(),
44            rename: None,
45            post_processors: Self::UPSTREAM_POST_PROCESSORS,
46        })
47    }
48}
