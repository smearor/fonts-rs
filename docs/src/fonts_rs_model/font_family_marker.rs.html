7/// Sealed trait module preventing external implementations of
8/// [`FontFamilyMarker`].
9///
10/// Only crates within the workspace can implement `FontFamilyMarker` by
11/// also implementing `Sealed`, which is private to this module.
12pub mod sealed {
13    /// Private supertrait that prevents external crates from implementing
14    /// [`FontFamilyMarker`](super::FontFamilyMarker).
15    pub trait Sealed {}
16}

18/// Marker trait for font family identity.
19///
20/// Implemented by each font family crate as a zero-sized enum (e.g.
21/// `SevenSegment`, `Barcode`). Used as the phantom type parameter in
22/// [`GlyphName<F>`](crate::GlyphName) to ensure type safety between font
23/// families at compile time.
24///
25/// The `Sealed` supertrait prevents external crates from implementing
26/// `FontFamilyMarker`, ensuring that only crates within the workspace can
27/// define font family markers.
28///
29/// For the runtime API surface (glyph maps, capability traits), see
30/// [`FontFamily`](crate::FontFamily).
31pub trait FontFamilyMarker: sealed::Sealed {}
