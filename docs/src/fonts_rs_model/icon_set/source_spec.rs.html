1//! The [`IconSetSourceSpec`] struct - one `[[iconset.sources]]` table.
2
3use crate::icon_set::post_processor::IconSetPostProcessor;
4use crate::icon_set::source_location::IconSetSourceLocation;
5use crate::icon_set::whitelist_entry::IconWhitelistEntry;
6use serde::Deserialize;
7use serde::Serialize;
8use std::collections::HashMap;
9use std::collections::HashSet;
10
11/// One `[[iconset.sources]]` table: a source location plus the subset
12/// and processing applied to it.
13#[derive(Debug, Clone, Deserialize, Serialize)]
14pub struct IconSetSourceSpec {
15    /// Where this source's SVGs are located.
16    #[serde(flatten)]
17    pub location: IconSetSourceLocation,
18    /// Post-processors applied to this source's exported SVGs.
19    #[serde(default)]
20    pub post_processors: Vec<IconSetPostProcessor>,
21    /// Whitelist of icons to export - plain upstream names or renames.
22    pub icons: Vec<IconWhitelistEntry>,
23}
24
25impl IconSetSourceSpec {
26    /// The whitelist as a lookup set of upstream names.
27    pub fn whitelist(&self) -> HashSet<String> {
28        self.icons.iter().map(|entry| entry.upstream_name().to_string()).collect()
29    }
30
31    /// Upstream name → exported name map for the rename-aware name
32    /// resolution. Entries without a rename map to themselves.
33    pub fn rename_map(&self) -> HashMap<String, String> {
34        self.icons
35            .iter()
36            .map(|entry| (entry.upstream_name().to_string(), entry.exported_name().to_string()))
37            .collect()
38    }
39}
