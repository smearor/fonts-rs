14/// Serde model of the application's `icons.toml` `[iconset]` table.
15///
16/// The schema is the wire format both ways: `IconSetBuild` reads it at
17/// build time, tooling like the icon-set builder writes it. Always
18/// wrap in [`IconSetFile`](crate::icon_set::IconSetFile) for
19/// (de)serialization - the fields live under the `[iconset]` table.
20#[derive(Debug, Clone, Deserialize, Serialize)]
21pub struct IconSetSpec {
22    /// PascalCase marker enum name, e.g. `MyAppIcons`.
23    pub name: String,
24    /// Human-readable display name for build logs.
25    pub display_name: String,
26    /// Glyph name prefix producing `{icon_prefix}-{name}-symbolic`
27    /// icon names.
28    pub icon_prefix: String,
29    /// GResource prefix for the compiled bundle.
30    pub gresource_prefix: String,
31    /// Freedesktop icon context (`actions` by default).
32    #[serde(default = "default_icons_context")]
33    pub icons_context: String,
34    /// SVG sources merged into this set, in TOML order. Must be
35    /// non-empty.
36    pub sources: Vec<IconSetSourceSpec>,
37}
38
39impl IconSetSpec {
40    /// Validates the spec: naming conventions, non-empty fields and
41    /// sources, no duplicate whitelist entries, no exported-name
42    /// collisions across sources.
43    ///
44    /// Covers everything decidable from the document alone -
45    /// filesystem-dependent checks (unknown icons, source resolution)
46    /// run later in `IconSetBuild`.
47    ///
48    /// # Errors
49    ///
50    /// Returns [`IconSetSpecError`] describing the first violation.
51    pub fn validate(&self) -> Result<(), IconSetSpecError> {
52        if !is_pascal_case(&self.name) {
53            return Err(IconSetSpecError::InvalidIdent {
54                what: "iconset.name",
55                value: self.name.clone(),
56                expected: "PascalCase marker name",
57            });
58        }
59        if self.display_name.is_empty() {
60            return Err(IconSetSpecError::EmptyField { what: "iconset.display_name" });
61        }
62        if !is_kebab(&self.icon_prefix) {
63            return Err(IconSetSpecError::InvalidIdent {
64                what: "iconset.icon_prefix",
65                value: self.icon_prefix.clone(),
66                expected: "kebab-case icon prefix",
67            });
68        }
69        if !self.gresource_prefix.starts_with('/') || self.gresource_prefix.len() < 2 {
70            return Err(IconSetSpecError::InvalidIdent {
71                what: "iconset.gresource_prefix",
72                value: self.gresource_prefix.clone(),
73                expected: "absolute GResource path prefix, e.g. /com/example/app/icons",
74            });
75        }
76        if self.sources.is_empty() {
77            return Err(IconSetSpecError::EmptySources);
78        }
79
80        let mut exported: HashMap<String, String> = HashMap::new();
81        for source in &self.sources {
82            let source_label = source.location.label();
83            let mut upstream: HashSet<&str> = HashSet::new();
84            for entry in &source.icons {
85                if !upstream.insert(entry.upstream_name()) {
86                    return Err(IconSetSpecError::DuplicateIcon {
87                        source_id: source_label.clone(),
88                        name: entry.upstream_name().to_string(),
89                    });
90                }
91                if !is_kebab(entry.exported_name()) {
92                    return Err(IconSetSpecError::InvalidIdent {
93                        what: "iconset.icons entry",
94                        value: entry.exported_name().to_string(),
95                        expected: "kebab-case icon name",
96                    });
97                }
98                if let Some(existing) = exported.insert(entry.exported_name().to_string(), source_label.clone()) {
99                    let _ = existing;
100                    return Err(IconSetSpecError::IconNameCollision {
101                        name: entry.exported_name().to_string(),
102                    });
103                }
104            }
105        }
106        Ok(())
107    }
108}

110/// Checks a PascalCase identifier: first char uppercase ASCII, rest
111/// alphanumeric.
112fn is_pascal_case(value: &str) -> bool {
113    let mut chars = value.chars();
114    chars.next().is_some_and(|first| first.is_ascii_uppercase()) && chars.all(|ch| ch.is_ascii_alphanumeric())
115}
116
117/// Checks strict kebab-case: lowercase ASCII letters, digits and single
118/// dashes between segments.
119fn is_kebab(value: &str) -> bool {
120    !value.is_empty()
121        && !value.starts_with('-')
122        && !value.ends_with('-')
123        && !value.contains("--")
124        && value.chars().all(|ch| ch.is_ascii_lowercase() || ch.is_ascii_digit() || ch == '-')
125}

127#[cfg(test)]
128mod tests {
129    use super::*;
130
131    fn valid_spec() -> IconSetSpec {
132        IconSetSpec {
133            name: "MyAppIcons".to_string(),
134            display_name: "My App Icons".to_string(),
135            icon_prefix: "myapp".to_string(),
136            gresource_prefix: "/com/example/myapp/icons".to_string(),
137            icons_context: "actions".to_string(),
138            sources: vec![IconSetSourceSpec {
139                location: crate::icon_set::IconSetSourceLocation::Dir { dir: "icons".to_string() },
140                post_processors: Vec::new(),
141                icons: vec![crate::icon_set::IconWhitelistEntry::Name("check".to_string())],
142            }],
143        }
144    }
145
146    #[test]
147    fn valid_spec_passes() {
148        assert!(valid_spec().validate().is_ok());
149    }
150
151    #[test]
152    fn empty_sources_fail() {
153        let mut spec = valid_spec();
154        spec.sources.clear();
155        assert!(matches!(spec.validate(), Err(IconSetSpecError::EmptySources)));
156    }
157
158    #[test]
159    fn snake_case_marker_fails() {
160        let mut spec = valid_spec();
161        spec.name = "my_app_icons".to_string();
162        assert!(matches!(spec.validate(), Err(IconSetSpecError::InvalidIdent { what: "iconset.name", .. })));
163    }
164
165    #[test]
166    fn collision_across_sources_fails() {
167        let mut spec = valid_spec();
168        spec.sources.push(IconSetSourceSpec {
169            location: crate::icon_set::IconSetSourceLocation::Dir { dir: "other".to_string() },
170            post_processors: Vec::new(),
171            icons: vec![crate::icon_set::IconWhitelistEntry::Name("check".to_string())],
172        });
173        assert!(matches!(spec.validate(), Err(IconSetSpecError::IconNameCollision { .. })));
174    }
175
176    #[test]
177    fn rename_resolves_collision() {
178        let mut spec = valid_spec();
179        spec.sources.push(IconSetSourceSpec {
180            location: crate::icon_set::IconSetSourceLocation::Dir { dir: "other".to_string() },
181            post_processors: Vec::new(),
182            icons: vec![crate::icon_set::IconWhitelistEntry::Renamed {
183                name: "check".to_string(),
184                rename: "check-filled".to_string(),
185            }],
186        });
187        assert!(spec.validate().is_ok());
188    }
