1//! The [`IconSetSpecError`] error type for `icons.toml` validation.
2
3use thiserror::Error;
4
5/// Error returned by
6/// [`IconSetSpec::validate`](crate::icon_set::IconSetSpec::validate) -
7/// covers every spec violation that can be detected from the document
8/// alone, without touching the filesystem.
9#[derive(Debug, Clone, Error)]
10pub enum IconSetSpecError {
11    /// The `sources` list is empty.
12    #[error("icon set must declare at least one [[iconset.sources]] table")]
13    EmptySources,
14
15    /// An identifier or name violates its naming convention.
16    #[error("invalid {what}: '{value}' - expected {expected}")]
17    InvalidIdent {
18        /// Which field failed validation (e.g. `iconset.name`).
19        what: &'static str,
20        /// The offending value.
21        value: String,
22        /// The expected format (e.g. `PascalCase`, `kebab-case`).
23        expected: &'static str,
24    },
25
26    /// A field is empty where a value is required.
27    #[error("{what} must not be empty")]
28    EmptyField {
29        /// Which field is empty (e.g. `iconset.display_name`).
30        what: &'static str,
31    },
32
33    /// The same upstream name is whitelisted twice within one source.
34    #[error("duplicate whitelist entry '{name}' in {source_id}")]
35    DuplicateIcon {
36        /// Source identifier (crate name or directory).
37        source_id: String,
38        /// Duplicated upstream icon name.
39        name: String,
40    },
41
42    /// Two whitelist entries resolve to the same exported icon name,
43    /// within one source or across sources.
44    #[error("icon name collision: '{name}' is exported more than once - use a rename to disambiguate")]
45    IconNameCollision {
46        /// The colliding exported base name.
47        name: String,
48    },
49}
