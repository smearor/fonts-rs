1//! Software rendering capability.
2//!
3//! Provides the [`SoftwareRender`] trait for font families loadable via
4//! `ab_glyph` - the only capability trait without a `GtkFont` bound,
5//! usable in pure-Rust renderers.
6
7use crate::FontFamily;
8
9/// Software rendering via `ab_glyph` (feature `render`).
10///
11/// Emitted per family by `impl_font_loader!` - the font loading itself
12/// stays per-crate because the TTF source differs (`embed-fonts` vs.
13/// `resources/` directory).
14pub trait SoftwareRender: FontFamily {
15    /// Variant selector - `()` for single-font families, a per-family
16    /// enum for multi-TTF families (e.g. `NerdFontVariant`).
17    type Variant: Copy + Default;
18
19    /// Returns the cached `ab_glyph` font for `variant`.
20    ///
21    /// With `embed-fonts` the TTF is compiled into the binary; otherwise
22    /// it is loaded from `resources/` relative to the working directory.
23    fn ab_glyph(variant: Self::Variant) -> Option<&'static ab_glyph::FontVec>;
24
25    /// Convenience: the default variant. For single-font families this
26    /// is the only call needed (`Code39::font()`).
27    fn font() -> Option<&'static ab_glyph::FontVec> {
28        Self::ab_glyph(Self::Variant::default())
29    }
30}
