1//! `@font-face` HTML export capability.
2//!
3//! Provides the [`HtmlExport`] trait for font families that can produce
4//! self-contained HTML pages embedding their TTF as a base64 data URI.
5//! Inline-SVG export (no font required) lives in the base trait
6//! [`HtmlSvgExport`](crate::HtmlSvgExport).
7
8use crate::FontError;
9use crate::HtmlSvgExport;
10use base64::Engine;
11
12/// Self-contained `@font-face` HTML export.
13///
14/// Extends [`HtmlSvgExport`] with the font-based export mode: the
15/// default-variant TTF is embedded via
16/// [`GtkFont::load_font`](crate::GtkFont::load_font). Families that
17/// deliberately avoid `@font-face` (Nerd Fonts) or ship no TTF at all
18/// (fontless icon sets) do not implement this trait - they can still
19/// implement [`HtmlSvgExport`] alone.
20pub trait HtmlExport: HtmlSvgExport {
21    /// Returns a `@font-face` CSS rule embedding the default-variant
22    /// TTF as a base64 data URI. Low-level building block for custom
23    /// HTML pages.
24    ///
25    /// # Errors
26    ///
27    /// Returns a [`FontError`] if registration, font loading, or reading
28    /// the temp file fails.
29    fn css_font_face() -> Result<String, FontError> {
30        Self::ensure_registered()?;
31        let font_path = Self::load_font()?;
32        let font_bytes = std::fs::read(font_path)?;
33        let font_b64 = base64::engine::general_purpose::STANDARD.encode(&font_bytes);
34        Ok(format!(
35            "@font-face {{\n    font-family: '{}';\n    src: url('data:font/ttf;base64,{font_b64}') format('truetype');\n}}",
36            Self::CSS_FONT_FAMILY
37        ))
38    }
39}
