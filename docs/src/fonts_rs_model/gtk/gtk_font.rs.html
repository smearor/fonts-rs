13/// Trait for GTK font families with GResource-backed registration.
14///
15/// Implemented by font family marker types via the `impl_gtk_font!` macro
16/// in `fonts-rs-generator`. Each method is a static (no `&self`) method
17/// callable as e.g. `Ean13::register_glyphs()`.
18///
19/// # Methods
20///
21/// - [`register_glyphs`](Self::register_glyphs): Registers the SVG icon
22///   GResource bundle (`icons.gresource`).
23/// - [`register_font`](Self::register_font): Registers the TTF font
24///   GResource bundle (`font.gresource`). No-op for fonts without a TTF
25///   GResource.
26/// - [`load_font`](Self::load_font): Loads the TTF from the GResource into
27///   a temp file and returns the path. Returns an error for fonts without
28///   a TTF GResource.
29/// - [`load_font_from_gresource`](Self::load_font_from_gresource):
30///   Provided helper - shared `load_font()` body for families whose
31///   font GResource bundles one or more TTFs.
32/// - [`ensure_registered`](Self::ensure_registered): Lazy, idempotent
33///   registration of all the family's GResource bundles, cached per
34///   family. Called automatically by every capability method.
35/// - [`install`](Self::install): Explicit fail-fast setup for application
36///   startup.
37///
38/// The `'static` supertrait allows [`ensure_registered`](Self::ensure_registered)
39/// to key its global cache by `TypeId::of::<Self>()`.
40pub trait GtkFont: 'static {
41    /// Registers embedded SVG glyphs as a GResource.
42    ///
43    /// # Errors
44    ///
45    /// Returns a [`RegisterGlyphError`] if resource registration fails.
46    fn register_glyphs() -> Result<(), RegisterGlyphError>;
47
48    /// Registers the TTF font as a GResource.
49    ///
50    /// No-op for font families without a `font.gresource` bundle.
51    ///
52    /// # Errors
53    ///
54    /// Returns a [`RegisterFontError`] if resource registration fails.
55    fn register_font() -> Result<(), RegisterFontError>;
56
57    /// Loads the TTF font from the registered GResource into a temp file
58    /// and returns the path.
59    ///
60    /// The temp file is kept alive for the process lifetime. Use the
61    /// returned path with `FontMapExt::add_font_file`.
62    ///
63    /// # Errors
64    ///
65    /// Returns a [`LoadFontError`] if the font cannot be registered,
66    /// looked up, or written.
67    fn load_font() -> Result<&'static std::path::Path, LoadFontError>;
68
69    /// Idempotent registration of all GResource bundles this family has.
70    ///
71    /// Called lazily by every capability method. The result is cached per
72    /// family in a global `OnceLock` keyed by `TypeId`: subsequent calls
73    /// return the cached `Ok` or the cached error. Safe to call from any
74    /// thread.
75    ///
76    /// A `static OnceLock` inside this default body would be shared
77    /// across *all* impls, so the cache is a `HashMap` keyed by
78    /// `TypeId::of::<Self>()` instead - one entry per family, no
79    /// macro-generated statics required.
80    ///
81    /// # Errors
82    ///
83    /// Returns the cached [`FontError`] if registration failed.
84    fn ensure_registered() -> Result<(), FontError> {
85        static CACHE: OnceLock<Mutex<HashMap<TypeId, Result<(), FontError>>>> = OnceLock::new();
86        let cache = CACHE.get_or_init(|| Mutex::new(HashMap::new()));
87        let mut guard = cache.lock().unwrap_or_else(|e| e.into_inner());
88        guard.entry(TypeId::of::<Self>()).or_insert_with(|| Self::do_register()).clone()
89    }
90
91    /// Explicit fail-fast setup for application startup.
92    ///
93    /// Equivalent to [`ensure_registered`](Self::ensure_registered) -
94    /// exists to make intent visible at the call site
95    /// (`Code39::install()?` in `main()`).
96    ///
97    /// # Errors
98    ///
99    /// Returns the cached [`FontError`] if registration failed.
100    fn install() -> Result<(), FontError> {
101        Self::ensure_registered()
102    }
103
104    /// Performs the actual registration of all the family's GResource
105    /// bundles. Called once per family by
106    /// [`ensure_registered`](Self::ensure_registered).
107    ///
108    /// Default: [`register_glyphs`](Self::register_glyphs) +
109    /// [`register_font`](Self::register_font). Families with additional
110    /// bundles override this method.
111    #[doc(hidden)]
112    fn do_register() -> Result<(), FontError> {
113        Self::register_glyphs()?;
114        Self::register_font()?;
115        Ok(())
116    }
117
118    /// Extracts one TTF from the registered font GResource into a
119    /// process-lifetime temp file and returns the cached path.
120    ///
121    /// Shared implementation of the [`load_font`](Self::load_font)
122    /// bodies of multi-TTF families: [`register_font`](Self::register_font)
123    /// must have succeeded already (the caller's `load_font` takes care
124    /// of it). Results are cached per `resource_path`; `temp_tag`
125    /// (typically `env!("CARGO_PKG_NAME")` at the call site) namespaces
126    /// the temp file so faces of different families never collide.
127    ///
128    /// Provided method - implementors are not expected to override it.
129    ///
130    /// # Errors
131    ///
132    /// Returns a [`LoadFontError`] if the resource lookup or the
133    /// temp-file write fails.
134    fn load_font_from_gresource(resource_path: &str, temp_tag: &str) -> Result<&'static std::path::Path, LoadFontError> {
135        static CACHE: OnceLock<Mutex<HashMap<String, &'static std::path::Path>>> = OnceLock::new();
136        let cache = CACHE.get_or_init(|| Mutex::new(HashMap::new()));
137        let mut guard = cache.lock().unwrap_or_else(|e| e.into_inner());
138        if let Some(path) = guard.get(resource_path) {
139            return Ok(path);
140        }
141        let bytes = gio::resources_lookup_data(resource_path, gio::ResourceLookupFlags::NONE)?;
142        let stem = std::path::Path::new(resource_path).file_stem().and_then(|s| s.to_str()).unwrap_or("font");
143        let temp_path = std::env::temp_dir().join(format!("fonts-rs-{temp_tag}-{stem}-{}.ttf", std::process::id()));
144        std::fs::write(&temp_path, bytes.as_ref())?;
145        // Leak the path: the temp file must live for the process
146        // lifetime (pango/ab_glyph read it on demand), so a shared
147        // `&'static Path` cache entry is intentional.
148        let path: &'static std::path::Path = Box::leak(temp_path.into_boxed_path());
149        guard.insert(resource_path.to_string(), path);
150        Ok(path)
151    }
152}
