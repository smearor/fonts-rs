6/// Base GResource prefix for all fonts-rs resources.
7///
8/// Individual font families build their full prefix by appending their
9/// family-specific sub-path (e.g. `nerd_fonts`, `dseg7`).
10///
11/// # Examples
12///
13/// - Nerd Fonts: `{GRESOURCE_BASE_PREFIX}/nerd_fonts` -> `/io/smearor/fonts/nerd_fonts`
14/// - Seven Segment: `{GRESOURCE_BASE_PREFIX}/dseg7` -> `/io/smearor/fonts/dseg7`
15pub const GRESOURCE_BASE_PREFIX: &str = "/io/smearor/fonts";
