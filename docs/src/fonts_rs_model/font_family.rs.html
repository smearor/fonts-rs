1//! Core runtime trait for font families.
2//!
3//! [`FontFamily`] bundles the feature-free data surface every font family
4//! has: the typed glyph name, family constants, and the glyph maps. Usage
5//! modes (icon theme, Pango, HTML export, software rendering) live in the
6//! capability traits [`IconThemeFont`](crate::IconThemeFont),
7//! [`PangoFont`](crate::PangoFont), [`HtmlExport`](crate::HtmlExport), and
8//! [`SoftwareRender`](crate::SoftwareRender).
9
10use crate::CodePoint;
11use crate::FontFamilyConfig;
12use crate::FontFamilyMarker;
13use crate::FontVariant;
14use std::str::FromStr;
15
16/// Runtime access to a font family: metadata and glyph maps.
17///
18/// Implemented once per font crate via `impl_font_family!`. The core
19/// trait is feature-free - pure data access that works without GTK.
20///
21/// # Usage Modes
22///
23/// | User wants… | Trait |
24/// |---|---|
25/// | Icons via `GtkIconTheme` / `Image::from_icon_name` | [`IconThemeFont`](crate::IconThemeFont) |
26/// | Text in GTK widgets (Pango, CSS `font-family`) | [`PangoFont`](crate::PangoFont) |
27/// | One glyph as SVG / `gdk::Texture` | [`IconThemeFont`](crate::IconThemeFont) |
28/// | Text composed into a single SVG | [`IconThemeFont`](crate::IconThemeFont) |
29/// | HTML page with `@font-face` or inline SVG | [`HtmlExport`](crate::HtmlExport) |
30/// | Software rendering without GTK | [`SoftwareRender`](crate::SoftwareRender) |
31/// | Fail-fast setup at app start | [`GtkFont::install`](crate::GtkFont::install) |
32pub trait FontFamily: FontFamilyMarker + FontFamilyConfig {
33    /// Typed glyph name of this family (e.g. `GlyphName<Code39>`,
34    /// `IconName` for Nerd Fonts).
35    ///
36    /// `FromStr` lets implementations construct the typed name from the
37    /// `&'static str` stored in the glyph maps; `AsRef<str>` lets lookup
38    /// methods accept it alongside plain `&str` constants.
39    type Name: AsRef<str> + FromStr;
40
41    /// CSS `font-family` name, e.g. `"Libre Barcode Code 39"`.
42    ///
43    /// Defaults to the display name; override for families whose CSS
44    /// name differs.
45    const CSS_FONT_FAMILY: &'static str = Self::FAMILY_DISPLAY_NAME;
46
47    /// Build-time variant metadata (see [`FontVariant`]). Empty for
48    /// single-font families; lists the shipped TTF variants otherwise.
49    /// Useful for UIs that offer a variant picker.
50    const VARIANTS: &'static [FontVariant] = &[];
51
52    /// Looks up the typed glyph name for a Unicode codepoint.
53    ///
54    /// Accepts both `char` and [`CodePoint`] via `Into<CodePoint>`.
55    fn glyph_name(ch: impl Into<CodePoint>) -> Option<Self::Name>;
56
57    /// Looks up the codepoint for a glyph name.
58    ///
59    /// Accepts `&str`, `String`, and the typed [`Self::Name`] via
60    /// `AsRef<str>` - both `constants::UNI002A` and `GlyphName<F>` work.
61    fn codepoint(name: impl AsRef<str>) -> Option<CodePoint>;
62
63    /// Iterates all `(codepoint, glyph_name)` pairs.
64    ///
65    /// Yields the typed [`Self::Name`], consistent with
66    /// [`glyph_name`](Self::glyph_name). Use [`AsRef<str>`] on the name
67    /// when plain `&str` access is enough.
68    fn all_glyphs() -> impl Iterator<Item = (CodePoint, Self::Name)>;
69
70    /// Returns the number of glyphs in this family.
71    ///
72    /// Default: `all_glyphs().count()` - O(n). Implementations backed
73    /// by a `phf::Map` override this with `map.len()` - O(1).
74    fn glyph_count() -> usize {
75        Self::all_glyphs().count()
76    }
77}
