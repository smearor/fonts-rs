1//! The [`BuilderSettings`] set-level fields of an `icons.toml`.
2
3use fonts_rs_model::IconSetSpec;
4use fonts_rs_model::IconSetSpecError;
5
6/// The `[iconset]` table fields, editable in the settings pane.
7///
8/// Validation is delegated to [`IconSetSpec::validate`] at emission
9/// time - the builder cannot emit a spec the generator would reject.
10pub struct BuilderSettings {
11    /// PascalCase marker type name (`IconSetConfig`-marker).
12    pub name: String,
13    /// Human-readable set name for docs.
14    pub display_name: String,
15    /// Kebab-case exported icon prefix (`{prefix}-{name}-symbolic`).
16    pub icon_prefix: String,
17    /// GResource mount prefix for the exported icons.
18    pub gresource_prefix: String,
19    /// GTK icon-theme context for registration.
20    pub icons_context: String,
21}
22
23impl Default for BuilderSettings {
24    fn default() -> Self {
25        Self {
26            name: "MyIcons".to_string(),
27            display_name: "My Icons".to_string(),
28            icon_prefix: "my".to_string(),
29            gresource_prefix: "/com/example/myapp/icons".to_string(),
30            icons_context: "actions".to_string(),
31        }
32    }
33}
34
35impl BuilderSettings {
36    /// Validates the set fields by building a minimal spec around them
37    /// and delegating to [`IconSetSpec::validate`]. The placeholder
38    /// source keeps the check honest without a selection.
39    pub fn validate(&self, sources: Vec<fonts_rs_model::IconSetSourceSpec>) -> Result<(), IconSetSpecError> {
40        IconSetSpec {
41            name: self.name.clone(),
42            display_name: self.display_name.clone(),
43            icon_prefix: self.icon_prefix.clone(),
44            gresource_prefix: self.gresource_prefix.clone(),
45            icons_context: self.icons_context.clone(),
46            sources,
47        }
48        .validate()
49    }
50}
