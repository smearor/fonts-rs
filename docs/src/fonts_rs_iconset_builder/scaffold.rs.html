1//! The [`Scaffold`] copy-paste block - `Cargo.toml` deps, `build.rs`
2//! and module stub for the emitted `icons.toml`.
3
4use crate::builder_settings::BuilderSettings;
5use crate::icon_selection::IconSelection;
6
7/// Generates the project-side snippets the target app needs to consume
8/// the emitted `icons.toml`.
9pub struct Scaffold;
10
11impl Scaffold {
12    /// The paste-ready scaffold: `[dependencies]` entries for the
13    /// selected source crates, `fonts-rs-generator` under
14    /// `[build-dependencies]`, the one-line `build.rs` and the
15    /// `include!` module stub - exactly the CUSTOM_ICON_SET surface.
16    pub fn text(selection: &IconSelection, settings: &BuilderSettings) -> String {
17        let mut deps = String::new();
18        for crate_name in used_source_crates(selection) {
19            deps.push_str(&format!("{crate_name} = \"0.5\"\n"));
20        }
21
22        format!(
23            "# Cargo.toml\n\
24             [dependencies]\n\
25             {deps}\
26             \n\
27             [build-dependencies]\n\
28             fonts-rs-generator = \"0.5\"\n\
29             \n\
30             # build.rs\n\
31             fn main() -> miette::Result<()> {{\n\
32             \x20   fonts_rs_generator::IconSetBuild::from_file(\"icons.toml\")?.run()?;\n\
33             \x20   Ok(())\n\
34             }}\n\
35             \n\
36             # src/iconset.rs (generated marker module)\n\
37             include!(concat!(env!(\"OUT_DIR\"), \"/iconset.rs\"));\n\
38             \n\
39             # usage\n\
40             {name}::install()?;            # registers fonts + icon theme\n\
41             gtk4::Image::from_icon_name(\"{prefix}-<icon>-symbolic\");\n",
42            deps = deps,
43            name = settings.name,
44            prefix = settings.icon_prefix,
45        )
46    }
47}
48
49/// Distinct source crate names that have at least one picked icon.
50fn used_source_crates(selection: &IconSelection) -> Vec<&'static str> {
51    let mut crates: Vec<&'static str> = selection.groups().map(|(_key, source, _icons)| source.crate_name).collect();
52    crates.sort_unstable();
53    crates.dedup();
54    crates
55}
