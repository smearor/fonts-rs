1//! The [`IconSource`] registry entry - one icon-capable family mapped
2//! to its `[[iconset.sources]]` crate.
3
4use fonts_rs_model::UpstreamIcon;
5
6/// An icon-capable family: maps the cheat-sheet-facing family to the
7/// `[[iconset.sources]]` crate it can feed. Stem/subdirectory/post-
8/// processor logic lives in each family's `UpstreamIcons` impl - this
9/// is pure builder wiring.
10pub struct IconSource {
11    /// Source crate name for `icons.toml` (`crate = "…"`).
12    pub crate_name: &'static str,
13    /// Reverse-maps a grid icon name to its upstream location
14    /// (subdirectory + file stem + post-processors).
15    pub upstream_icon: fn(&str) -> Option<UpstreamIcon>,
16}
17
18impl IconSource {
19    /// The exported icon name a whitelist entry produces under the
20    /// set's `icon_prefix` — `Some(rename)` wins over `upstream`.
21    pub fn exported_name(set_prefix: &str, upstream: &str, rename: Option<&str>) -> String {
22        format!("{}-{}-symbolic", set_prefix, rename.unwrap_or(upstream))
23    }
24
25    /// Finds the registry source index + upstream location for a grid
26    /// icon name - first family whose `upstream_icon` accepts it.
27    pub fn resolve(icon_name: &str) -> Option<(usize, UpstreamIcon)> {
28        ICON_SOURCES
29            .iter()
30            .enumerate()
31            .find_map(|(index, source)| (source.upstream_icon)(icon_name).map(|upstream| (index, upstream)))
32    }
33}
34

37///

39/// subdirectory is selectable by hand-editing the emitted spec (the

68#[cfg(test)]
69mod tests {
70    use super::*;
71
72    #[test]
73    fn exported_name_prefers_rename() {
74        assert_eq!(IconSource::exported_name("app", "trash", Some("delete")), "app-delete-symbolic");
75        assert_eq!(IconSource::exported_name("app", "check", None), "app-check-symbolic");
76    }
77
78    #[test]
79    fn resolve_routes_each_family() {
80        let (index, upstream) = IconSource::resolve("lucide-check-symbolic").unwrap();
81        assert_eq!(ICON_SOURCES[index].crate_name, "fonts-rs-lucide");
82        assert_eq!(upstream.stem, "check");
83        assert_eq!(upstream.subdirectory, None);
84
85        let (_, tabler) = IconSource::resolve("tabler-heart-symbolic").unwrap();
86        // The subdirectory follows the compiled variant - `outline`
87        // by default, `filled` under --all-features.
88        assert_eq!(tabler.subdirectory, <fonts_rs_tabler::Tabler as fonts_rs_model::UpstreamIcons>::UPSTREAM_SUBDIRECTORY);
89
90        let (_, eva) = IconSource::resolve("eva-checkmark-outline-symbolic").unwrap();
91        assert_eq!(eva.subdirectory, Some("outline"));
92        assert_eq!(eva.stem, "checkmark-outline");
93
94        let (_, fluent) = IconSource::resolve("fluent-panel-left-add-symbolic").unwrap();
95        assert_eq!(fluent.subdirectory, Some("regular-24"));
96        assert_eq!(fluent.stem, "ic_fluent_panel_left_add_24_regular");
97
98        assert!(IconSource::resolve("unknown-check-symbolic").is_none());
99    }
