1//! The [`ProjectExport`] - a compile-ready scaffold crate emitted as
2//! a zip archive.

4use crate::builder_settings::BuilderSettings;
5use crate::icon_selection::IconSelection;
6use crate::icon_selection::IconSelectionError;
7use std::io::Write;

9/// Packages the emitted `icons.toml` with the consumer-side plumbing
10/// (`Cargo.toml`, `build.rs`, `src/lib.rs`) into a zip whose contents
11/// `cargo build`s out of the box - the shape `fonts-rs-iconset-test`
12/// exercises.
13pub struct ProjectExport;
14
15/// One file inside the exported project tree.
16struct ExportFile {
17    /// Path relative to the project root (`src/lib.rs`).
18    path: String,
19    /// File contents (UTF-8).
20    content: String,
21}
22
23impl ProjectExport {
24    /// PascalCase set name → kebab-case package name (`AppIcons` →
25    /// `app-icons`).
26    pub fn package_name(set_name: &str) -> String {
27        let mut kebab = String::new();
28        for ch in set_name.chars() {
29            if ch.is_ascii_uppercase() && !kebab.is_empty() {
30                kebab.push('-');
31            }
32            kebab.push(ch.to_ascii_lowercase());
33        }
34        kebab
35    }
36
37    /// Writes a `{prefix}-icons.zip` project scaffold to `path`.
38    /// Returns the emitted `icons.toml` text (useful for toasts/logs).
39    pub fn write_zip(selection: &IconSelection, settings: &BuilderSettings, path: &std::path::Path) -> Result<(), ProjectExportError> {
40        let file = std::fs::File::create(path)?;
41        let mut zip = zip::ZipWriter::new(file);
42        let options = zip::write::SimpleFileOptions::default();
43        for export in Self::files(selection, settings)? {
44            zip.start_file(&export.path, options)?;
45            zip.write_all(export.content.as_bytes())?;
46        }
47        zip.finish()?;
48        Ok(())
49    }
50
51    /// All project files, in write order.
52    fn files(selection: &IconSelection, settings: &BuilderSettings) -> Result<Vec<ExportFile>, IconSelectionError> {
53        let package = Self::package_name(&settings.name);
54        Ok(vec![
55            ExportFile {
56                path: "Cargo.toml".to_string(),
57                content: cargo_toml(selection, &package),
58            },
59            ExportFile {
60                path: "icons.toml".to_string(),
61                content: selection.to_toml(settings)?,
62            },
63            ExportFile {
64                path: "build.rs".to_string(),
65                content: build_rs().to_string(),
66            },
67            ExportFile {
68                path: "src/lib.rs".to_string(),
69                content: lib_rs(&settings.name),
70            },
71        ])
72    }
73}
74
75/// `[build-dependencies]` gets the generator; `[dependencies]` the
76/// model plus the actually-picked source crates (their `links`
77/// metadata feeds `DEP_*_SVG_DIR` into `build.rs`).
78fn cargo_toml(selection: &IconSelection, package: &str) -> String {
79    let mut deps = String::from("fonts-rs-model = \"0.5\"\n");
80    let mut crates: Vec<&'static str> = selection.groups().map(|(_key, source, _icons)| source.crate_name).collect();
81    crates.sort_unstable();
82    crates.dedup();
83    for crate_name in crates {
84        deps.push_str(&format!("{crate_name} = \"0.5\"\n"));
85    }
86    format!(
87        "[package]\n\
88         name = \"{package}\"\n\
89         version = \"0.1.0\"\n\
90         edition = \"2024\"\n\
91         description = \"Custom icon set exported by fonts-rs-iconset-builder\"\n\
92         \n\
93         [build-dependencies]\n\
94         fonts-rs-generator = \"0.5\"\n\
95         miette = \"7.6\"\n\
96         \n\
97         [dependencies]\n\
98         {deps}",
99        package = package,
100        deps = deps,
101    )
102}

104/// One-liner build script - exactly the CUSTOM_ICON_SET surface.
105fn build_rs() -> &'static str {
106    "fn main() -> miette::Result<()> {\n\
107     \x20   fonts_rs_generator::IconSetBuild::from_file(\"icons.toml\")?.run()?;\n\
108     \x20   Ok(())\n\
109     }\n"
110}
111
112/// The generated module include + re-export; `pub use icons::*` keeps
113/// it correct whatever marker/name types `icons.toml` generates.
114fn lib_rs(set_name: &str) -> String {
115    format!(
116        "//! Custom icon set `{set_name}` - generated by `IconSetBuild`.\n\
117         \n\
118         pub mod icons {{\n\
119         \x20   include!(concat!(env!(\"OUT_DIR\"), \"/iconset.rs\"));\n\
120         }}\n\
121         \n\
122         pub use icons::*;\n"
123    )
124}

126/// Errors when writing a project export.
127#[derive(Debug, thiserror::Error)]
128pub enum ProjectExportError {
129    /// Spec assembly/serialization failed.
130    #[error(transparent)]
131    Spec(#[from] IconSelectionError),
132    /// File I/O failed.
133    #[error(transparent)]
134    Io(#[from] std::io::Error),
135    /// Zip writer failed.
136    #[error(transparent)]
137    Zip(#[from] zip::result::ZipError),
138}

155    #[test]
156    fn package_name_kebabs_pascal() {
157        assert_eq!(ProjectExport::package_name("AppIcons"), "app-icons");
158        assert_eq!(ProjectExport::package_name("Fixture"), "fixture");
159    }

161    #[test]
162    fn files_cover_the_compile_surface() {
163        let mut selection = IconSelection::new();
164        selection.add("lucide-check-symbolic");
165        let files = ProjectExport::files(&selection, &settings()).unwrap();
166        let paths: Vec<&str> = files.iter().map(|file| file.path.as_str()).collect();
167        assert_eq!(paths, ["Cargo.toml", "icons.toml", "build.rs", "src/lib.rs"]);
168
169        let cargo = files.iter().find(|file| file.path == "Cargo.toml").unwrap();
170        assert!(cargo.content.contains("name = \"app-icons\""));
171        assert!(cargo.content.contains("fonts-rs-lucide = \"0.5\""));
172        assert!(cargo.content.contains("[build-dependencies]"));
173
174        let lib = files.iter().find(|file| file.path == "src/lib.rs").unwrap();
175        assert!(lib.content.contains("include!(concat!(env!(\"OUT_DIR\")"));
176    }
177
178    #[test]
179    fn write_zip_produces_readable_archive() {
180        let mut selection = IconSelection::new();
181        selection.add("lucide-check-symbolic");
182        let path = std::env::temp_dir().join(format!("export-test-{}.zip", std::process::id()));
183        ProjectExport::write_zip(&selection, &settings(), &path).unwrap();
184        let mut archive = zip::ZipArchive::new(std::fs::File::open(&path).unwrap()).unwrap();
185        assert_eq!(archive.len(), 4);
186        let mut names: Vec<String> = (0..archive.len())
187            .filter_map(|i| archive.by_index(i).ok().and_then(|f| f.name().ok().map(|n| n.to_string())))
188            .collect();
189        names.sort();
190        assert_eq!(names, ["Cargo.toml", "build.rs", "icons.toml", "src/lib.rs"]);
191        let _ = std::fs::remove_file(&path);
192    }
