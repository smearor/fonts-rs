1//! Lucide Icons Cheat Sheet application.
2//!
3//! Browse all available Lucide icons with a search filter.
4//! Displays each icon with its Unicode codepoint and icon name.
5//! Click an icon to open a detail sidebar.
6//!
7//! The UI lives in `fonts-rs-cheat-sheet`; this binary only supplies
8//! the Lucide hooks via [`CheatSheetFamily`]. With the `metadata`
9//! feature (default), search also covers upstream tags from
10//! `icons.json` - e.g. "font size" finds `lucide-a-arrow-down-symbolic`.
11//!
12//! Lucide is a fontless icon set: grid cells render via the
13//! `GtkIconTheme` (`glyph_icon_name`), not as font text. Upstream
14//! `categories` populate the green category chips - the first family
15//! with real icon-set categories.
16//!
17//! Run with:
18//! ```sh
19//! cargo run -p fonts-rs-lucide-cheat-sheet
20//! ```
21
22use fonts_rs_cheat_sheet::CheatSheetFamily;
23use fonts_rs_cheat_sheet::FamilyKind;
24use fonts_rs_lucide::Lucide;
25use fonts_rs_lucide::LucideIconName;
26#[cfg(feature = "metadata")]
27use fonts_rs_lucide::metadata::LucideMetadata;
28use fonts_rs_model::CodePoint;
29#[cfg(feature = "metadata")]
30use fonts_rs_model::GlyphMetadata;
31use miette::IntoDiagnostic;
32use miette::Result;
33
34/// Lucide Icons hooks for the generic cheat sheet.
35pub struct LucideSheet;
36
37impl CheatSheetFamily for LucideSheet {
38    const TITLE: &'static str = "Lucide Icons - Cheat Sheet";
39    const NOUN: &'static str = "icons";
40    const APP_ID: &'static str = "io.smearor.fonts_rs.lucide_cheat_sheet";
41    const SEARCH_PLACEHOLDER: &'static str = "Search by name, keyword, category, or codepoint (e.g. 'home', 'alarm', 'EA01')...";
42    const NAME_FIELD_LABEL: &'static str = "Icon Name";
43    const FAMILY_NAME: &'static str = "Lucide Icons";
44    const FAMILY_KIND: FamilyKind = FamilyKind::IconSet;
45    const LICENSE: &'static str = "ISC";
46
47    type Name = LucideIconName;
48
49    fn init() -> Result<()> {
50        Lucide::install().into_diagnostic()?;
51        // Grid cells render via Image::from_icon_name - the icon theme
52        // path must be added to the display's GtkIconTheme. Runs after
53        // gtk4::init(), so the default display is already open. No
54        // install_pango/icon_css: Lucide ships no TTF.
55        let display = gtk4::gdk::Display::default().ok_or_else(|| miette::miette!("no default GDK display"))?;
56        Lucide::install_icon_theme(&display).into_diagnostic()
57    }
58
59    fn all_glyphs() -> Vec<(CodePoint, LucideIconName)> {
60        Lucide::all_glyphs().collect()
61    }
62
63    fn codepoint(name: &str) -> Option<CodePoint> {
64        Lucide::codepoint(name)
65    }
66
67    fn parse_name(name: &str) -> Option<LucideIconName> {
68        LucideIconName::parse(name)
69    }
70
71    fn display_name(name: &LucideIconName) -> String {
72        name.css_class_name().to_string()
73    }
74
75    fn subtitle(_name: &LucideIconName) -> String {
76        "Lucide Icons".to_string()
77    }
78
79    fn search(query: &str) -> Vec<String> {
80        #[cfg(feature = "metadata")]
81        {
82            LucideMetadata.search(query).iter().map(|s| s.to_string()).collect()
83        }
84        #[cfg(not(feature = "metadata"))]
85        {
86            let _ = query;
87            Vec::new()
88        }
89    }
90
91    fn keywords(name: &LucideIconName) -> Vec<String> {
92        #[cfg(feature = "metadata")]
93        {
94            LucideMetadata.keywords(name).iter().map(|k| k.as_str().to_string()).collect()
95        }
96        #[cfg(not(feature = "metadata"))]
97        {
98            let _ = name;
99            Vec::new()
100        }
101    }
102
103    fn categories(name: &LucideIconName) -> Vec<String> {
104        #[cfg(feature = "metadata")]
105        {
106            LucideMetadata.categories(name).iter().map(|c| c.as_str().to_string()).collect()
107        }
108        #[cfg(not(feature = "metadata"))]
109        {
110            let _ = name;
111            Vec::new()
112        }
113    }
114
115    /// Render cells via the GtkIconTheme: Lucide is fontless, so a
116    /// text label would show a PUA glyph only if some installed font
117    /// covers the codepoint - the bundled SVGs are authoritative.
118    fn glyph_icon_name(name: &LucideIconName) -> Option<String> {
119        Some(name.as_ref().to_string())
120    }
121
122    fn glyph_svg(name: &LucideIconName) -> Option<String> {
123        Lucide::glyph_svg(name)
124    }
125}
