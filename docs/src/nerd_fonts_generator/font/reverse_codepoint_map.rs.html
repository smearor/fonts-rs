1//! Reverse codepoint map construction from font glyph tables.
2
3use std::collections::HashMap;
4
5use nerd_fonts_model::CodePoint;
6
7/// Build a reverse cmap (GlyphId -> CodePoint) by probing Unicode codepoints.
8///
9/// Nerd Fonts map glyphs to codepoints in several Unicode ranges:
10/// - BMP PUA: U+E000-U+F8FF
11/// - Supplementary PUA: U+F0001-U+10FFFF
12/// - Miscellaneous Technical: U+23FB-U+23FE (IEC power symbols)
13/// - Miscellaneous Symbols and Arrows: U+2B58
14///
15/// To cover all cases, we scan the entire BMP (U+0000-U+FFFF) plus
16/// the supplementary PUA. At ~16ns per `glyph_index` lookup, this
17/// takes ~17ms total.
18pub fn build_reverse_cmap(face: &ttf_parser::Face) -> HashMap<ttf_parser::GlyphId, CodePoint> {
19    let mut map = probe_range(face, 0x0000, 0xFFFF);
20    map.extend(probe_range(face, 0xF0001, 0x10FFFF));
21    map
22}
23
24/// Probe a contiguous range of Unicode codepoints and return matching glyphs.
25fn probe_range(face: &ttf_parser::Face, start: u32, end: u32) -> HashMap<ttf_parser::GlyphId, CodePoint> {
26    let mut map = HashMap::new();
27    for codepoint in start..=end {
28        if let Some(ch) = char::from_u32(codepoint) {
29            if let Some(glyph_id) = face.glyph_index(ch) {
30                map.insert(glyph_id, CodePoint::from(ch));
31            }
32        }
33    }
34    map
35}
