1//! Parsed Devicon metadata: icon name to tags and altnames.
2
3use std::path::Path;
4
5use super::super::mapping::MetadataMapping;
6use super::super::mapping::RawAlias;
7use super::super::mapping::RawKeyword;
8use super::super::source::IconMetadataSource;
9use super::icon_entry::DeviconEntry;
10
11/// Parsed Devicon metadata: icon name to tags and altnames.
12#[derive(Debug, Clone)]
13pub struct DeviconMetadata {
14    /// Underlying mapping holding categories, keywords, and aliases.
15    pub(crate) mapping: MetadataMapping,
16}
17
18impl IconMetadataSource for DeviconMetadata {
19    fn prefix(&self) -> &'static str {
20        "nf-dev-"
21    }
22
23    fn from_file(path: &Path) -> Result<Self, Box<dyn std::error::Error + Send + Sync>> {
24        let content = std::fs::read_to_string(path)?;
25        let entries: Vec<DeviconEntry> = serde_json::from_str(&content)?;
26
27        let mut mapping = MetadataMapping::new();
28        for entry in entries {
29            if !entry.tags.is_empty() {
30                for tag in &entry.tags {
31                    mapping.keywords.add(&entry.name, tag.clone());
32                }
33            }
34            for altname in &entry.altnames {
35                mapping.aliases.insert(altname.clone(), RawAlias::new(entry.name.clone()));
36            }
37            for style_alias in &entry.aliases {
38                let alias_name = RawAlias::new(format!("{}-{}", entry.name, style_alias.alias));
39                let canonical_name = RawAlias::new(format!("{}-{}", entry.name, style_alias.base));
40                mapping.aliases.insert(alias_name, canonical_name);
41            }
42        }
43
44        Ok(Self { mapping })
45    }
46
47    fn mapping(&self) -> &MetadataMapping {
48        &self.mapping
49    }
50
51    fn keywords_for(&self, upstream_name: &str) -> Option<&[RawKeyword]> {
52        // Try direct lookup first, then resolve via altname.
53        if let Some(tags) = self.mapping.keywords.get(upstream_name) {
54            return Some(tags);
55        }
56        if let Some(canonical) = self.mapping.aliases.get(upstream_name) {
57            return self.mapping.keywords.get(canonical);
58        }
59        None
60    }
61}
