24    #[test]
25    fn fa_metadata_parses_categories() {
26        let categories_yaml = "
27accessibility:
28  icons:
29    - wheelchair
30    - blind
31  label: Accessibility
32arrows:
33  icons:
34    - arrow-left
35    - arrow-right
36  label: Arrows
37";
38        let icons_yaml = "
39wheelchair:
40  search:
41    terms:
42      - handicap
43      - accessibility
44arrow-left:
45  search:
46    terms: []
47";
48        let tmp = std::env::temp_dir();
49        let dir = tmp.join("test_fa");
50        std::fs::create_dir_all(&dir).unwrap();
51        std::fs::write(dir.join("categories.yml"), categories_yaml).unwrap();
52        std::fs::write(dir.join("icons.yml"), icons_yaml).unwrap();
53
54        let metadata = FaMetadata::from_file(&dir).unwrap();
55
56        assert_eq!(
57            metadata.categories_for("wheelchair"),
58            Some(&[RawCategory::new("Accessibility".to_string())] as &[RawCategory])
59        );
60        assert_eq!(metadata.categories_for("arrow-left"), Some(&[RawCategory::new("Arrows".to_string())] as &[RawCategory]));
61        assert!(metadata.categories_for("nonexistent").is_none());
62    }

64    #[test]
65    fn fa_metadata_parses_search_terms() {
66        let categories_yaml = "test:\n  icons: [gamepad]\n  label: Test\n";
67        let icons_yaml = "gamepad:\n  search:\n    terms:\n      - controller\n      - video game\n";
68        let tmp = std::env::temp_dir();
69        let dir = tmp.join("test_fa2");
70        std::fs::create_dir_all(&dir).unwrap();
71        std::fs::write(dir.join("categories.yml"), categories_yaml).unwrap();
72        std::fs::write(dir.join("icons.yml"), icons_yaml).unwrap();
73
74        let metadata = FaMetadata::from_file(&dir).unwrap();
75
76        let terms = metadata.keywords_for("gamepad").unwrap();
77        assert!(terms.contains(&RawKeyword::new("controller".to_string())));
78        assert!(terms.contains(&RawKeyword::new("video game".to_string())));
79    }
