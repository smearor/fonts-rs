16/// Marker enum for the Nerd Fonts font family.
17///
18/// Used as the `FontFamily` phantom type parameter. Zero-sized, exists
19/// only at the type level for compile-time type safety.
20pub enum NerdFonts {}

22impl sealed::Sealed for NerdFonts {}

24impl FontFamily for NerdFonts {}

26/// Build-time configuration for the Nerd Fonts font family.
27///
28/// Implements [`FontDefinition`] to plug into the generic
29/// [`FontDefinition::export_glyphs`] pipeline.
30pub struct NerdFontsDefinition;
31
32impl FontFamilyConfig for NerdFontsDefinition {
33    const FONT_FAMILY_NAME: &'static str = "nerd_fonts";
34
35    const FAMILY_DISPLAY_NAME: &'static str = "Nerd Fonts";
36
37    const GRESOURCE_PREFIX: &'static str = GRESOURCE_PREFIX;
38
39    const ICONS_CONTEXT: &'static str = "glyphs";
40
41    /// Unicode codepoint ranges to probe for reverse cmap construction.
42    ///
43    /// Nerd Fonts uses two disjoint ranges:
44    ///
45    /// - **BMP (`U+0000`–`U+FFFF`)**: Covers legacy icon codepoints in the
46    ///   Basic Multilingual Plane, including the PUA range `U+E000`–`U+F8FF`
47    ///   where most Font Awesome and Material Design icons are mapped.
48    /// - **Supplementary PUA (`U+F0001`–`U+10FFFF`)**: Covers the
49    ///   Supplementary Private Use Area, where Nerd Fonts 3+ places
50    ///   additional icon codepoints that don't fit in the BMP PUA.
51    ///
52    /// At ~16ns per `glyph_index` lookup, probing both ranges takes ~17ms.
53    const CODEPOINT_RANGES: &[CodePointRange] = &[BMP_RANGE, SUPPLEMENTARY_PUA_RANGE];
54}
55
56impl FontDefinition for NerdFontsDefinition {
57    type Name = IconName;
58
59    type Family = NerdFonts;
60
61    fn normalize_name(raw_glyph_name: &str) -> Option<Self::Name> {
62        IconName::from_glyph_name(raw_glyph_name)
63    }
64}
