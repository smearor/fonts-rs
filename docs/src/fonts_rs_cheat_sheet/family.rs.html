3use crate::family_kind::FamilyKind;
4use fonts_rs_model::CodePoint;
5use std::fmt::Display;

7/// Family-specific data surface for the cheat sheet UI.
8///
9/// Implement once per font family; [`crate::run`] builds the complete
10/// application from these hooks. The associated [`Name`](Self::Name)
11/// type is the family's glyph name - `IconName` for icon fonts,
12/// `String` or `GlyphName<F>` for others.
13///
14/// # Examples
15///
16/// ```ignore
17/// struct NotoEmojiSheet;
18///
19/// impl CheatSheetFamily for NotoEmojiSheet {
20///     type Name = String;
21///     // ...
22/// }
23///
24/// fn main() -> miette::Result<glib::ExitCode> {
25///     fonts_rs_cheat_sheet::run::<NotoEmojiSheet>()
26/// }
27/// ```
28pub trait CheatSheetFamily: 'static {
29    /// Window title and header text, e.g. `"Nerd Fonts - Cheat Sheet"`.
30    const TITLE: &'static str;
31
32    /// Noun used in count labels, e.g. `"icons"` or `"emoji"`.
33    const NOUN: &'static str;
34
35    /// GTK application ID, e.g. `"io.smearor.fonts_rs.cheat_sheet"`.
36    const APP_ID: &'static str;
37
38    /// Search entry placeholder text.
39    const SEARCH_PLACEHOLDER: &'static str;
40
41    /// CSS class applied to glyph preview labels so they render in the
42    /// family's font (e.g. `"nerd-icon"`). `None` for families whose
43    /// glyphs render via the system font (emoji).
44    const GLYPH_CSS_CLASS: Option<&'static str> = None;
45
46    /// Label of the name field in the detail sidebar
47    /// (e.g. `"CSS Class"` for icon fonts, `"Glyph Name"` otherwise).
48    const NAME_FIELD_LABEL: &'static str = "Glyph Name";
49
50    /// Display name of the font or icon set itself (e.g. `"Eva
51    /// Icons"`) - shown in the "Family" section of the detail sidebar.
52    const FAMILY_NAME: &'static str;
53
54    /// What kind of glyph source the family is - a general font, an
55    /// icon set, an emoji font or a special-purpose font.
56    const FAMILY_KIND: FamilyKind;
57
58    /// License of the bundled font or icon set (e.g. `"MIT"`,
59    /// `"OFL-1.1"`); list all options when dual-licensed.
60    const LICENSE: &'static str;
61
62    /// The glyph name type - `IconName` for icon fonts, `String` or
63    /// `GlyphName<F>` for other families.
64    type Name: Clone + Ord + AsRef<str> + Display + 'static;
65
66    /// Family initialization, called once in `main` before the
67    /// application starts (GResource registration, font loading, CSS).
68    ///
69    /// # Errors
70    ///
71    /// Returns an error if registration or font loading fails.
72    fn init() -> miette::Result<()>;
73
74    /// All glyphs as `(codepoint, name)` pairs, sorted by name.
75    fn all_glyphs() -> Vec<(CodePoint, Self::Name)>;
76
77    /// Resolves a name string to its codepoint.
78    fn codepoint(name: &str) -> Option<CodePoint>;
79
80    /// Parses a name string into the typed name.
81    fn parse_name(name: &str) -> Option<Self::Name>;
82
83    /// Display label for grid cells and the detail title
84    /// (e.g. the CSS class name for icon fonts).
85    fn display_name(name: &Self::Name) -> String;
86
87    /// Subtitle in the detail sidebar (e.g. the icon set label, or the
88    /// family display name).
89    fn subtitle(name: &Self::Name) -> String;
90
91    /// Metadata search: returns the names matching `query`.
92    ///
93    /// Families without metadata return an empty vec - the generic
94    /// supplement still matches names, hex codepoints, and characters.
95    fn search(query: &str) -> Vec<String>;
96
97    /// Keywords for a glyph name (empty without metadata).
98    fn keywords(name: &Self::Name) -> Vec<String>;
99
100    /// Categories for a glyph name (empty without metadata).
101    fn categories(name: &Self::Name) -> Vec<String>;
102
103    /// Display string of the canonical name when `name` is an alias.
104    /// `None` (the default) hides the alias section.
105    fn alias_display(_name: &Self::Name) -> Option<String> {
106        None
107    }
108
109    /// Optional canonical ordering contract for the family.
110    ///
111    /// Families with an upstream-curated glyph order (e.g. emoji picker
112    /// ordering) return their [`GlyphOrdering`](fonts_rs_model::GlyphOrdering)
113    /// implementation here;
114    /// `None` (the default) keeps the alphabetical grid.
115    ///
116    /// Opting in enables, per glyph name, in the cheat sheet UI:
117    ///
118    /// - **Picker-order grid sort** via `order_index` (search results
119    ///   and `all_glyphs` in picker order),
120    /// - **group jump-list dropdown** and a **group label** in the
121    ///   detail sidebar via `group_labels`/`group`,
122    /// - **variant collapsing** in the browse grid via
123    ///   `collapse_variants`/`base_of` (opt-in per family - emoji
124    ///   tones collapse, style families do not) plus a **"Variants"
125    ///   section** and **"Variant of"** back-link in the sidebar via
126    ///   `alternates`/`base_of`,
127    /// - **default-tone picking** via `tone_group`: a chosen alternate
128    ///   position is stored per tone group and the browse grid swaps
129    ///   base cells to that tone.
130    fn ordering() -> Option<&'static dyn fonts_rs_model::GlyphOrdering> {
131        None
132    }
133
134    /// Icon-theme name for a glyph, rendered via an image widget in grid
135    /// cells instead of a text label.
136    ///
137    /// `None` (the default) renders the codepoint character as text.
138    /// Return `Some` for families that register full-color SVG glyphs in
139    /// the `GtkIconTheme` - e.g. emoji fonts whose bundled TTF family
140    /// name would otherwise resolve to a system font.
141    fn glyph_icon_name(_name: &Self::Name) -> Option<String> {
142        None
143    }
144
145    /// SVG document for a glyph name, shown in the detail sidebar.
146    fn glyph_svg(name: &Self::Name) -> Option<String>;
147}
