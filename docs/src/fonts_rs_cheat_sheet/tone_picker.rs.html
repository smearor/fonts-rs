1//! Default-tone state for emoji-style variant pickers.
2
3use std::cell::RefCell;
4use std::collections::HashMap;
5use std::rc::Rc;
6
7/// Re-render callback invoked after a tone selection change.
8type RefreshFn = Rc<dyn Fn()>;
9
10/// Shared slot for the refresh callback, wired once the grid closures exist.
11type RefreshSlot = Rc<RefCell<Option<RefreshFn>>>;
12
13/// Session-scoped default-tone selection for tone-variant glyph groups.
14///
15/// Maps a [`GlyphOrdering::tone_group`](fonts_rs_model::GlyphOrdering::tone_group)
16/// id to the chosen position inside that group's `alternates` list. The
17/// index transfers to every base glyph sharing the group because entries
18/// with the same `tone_group` have the same alternate shape.
19///
20/// A global default index ([`TonePicker::set_default`]) applies to
21/// every tone group without a per-group selection - that powers the
22/// always-visible tone row in the browse sidebar while per-glyph
23/// overrides remain possible from the detail page.
24///
25/// `refresh` is wired by the app once the grid closures exist - picking
26/// or resetting a tone re-renders so the swap applies immediately.
27#[derive(Clone)]
28pub struct TonePicker {
29    /// Selected alternate index per `tone_group` id.
30    pub(super) selections: Rc<RefCell<HashMap<u32, usize>>>,
31    /// Global default alternate index for all groups without a
32    /// per-group selection.
33    default_index: Rc<RefCell<Option<usize>>>,
34    /// Re-render the grid (recompute icons + populate) after a change.
35    refresh: RefreshSlot,
36}
37
38impl TonePicker {
39    /// Creates an empty picker state; call [`TonePicker::set_refresh`]
40    /// once the grid closures are built.
41    pub fn new() -> Self {
42        Self {
43            selections: Rc::new(RefCell::new(HashMap::new())),
44            default_index: Rc::new(RefCell::new(None)),
45            refresh: Rc::new(RefCell::new(None)),
46        }
47    }
48
49    /// Installs the callback invoked after `select`/`reset`.
50    pub fn set_refresh(&self, refresh: RefreshFn) {
51        *self.refresh.borrow_mut() = Some(refresh);
52    }
53
54    /// Remembers `index` as the default alternate position for `group`
55    /// and re-renders the grid.
56    pub fn select(&self, group: u32, index: usize) {
57        self.selections.borrow_mut().insert(group, index);
58        self.refresh();
59    }
60
61    /// Clears the default tone for `group` and re-renders the grid.
62    pub fn reset(&self, group: u32) {
63        self.selections.borrow_mut().remove(&group);
64        self.refresh();
65    }
66
67    /// Stores `index` as the global default alternate position applied
68    /// to every tone group without a per-group selection, and
69    /// re-renders the grid.
70    pub fn set_default(&self, index: usize) {
71        *self.default_index.borrow_mut() = Some(index);
72        self.refresh();
73    }
74
75    /// Returns the global default alternate index, if set.
76    pub fn default(&self) -> Option<usize> {
77        *self.default_index.borrow()
78    }
79
80    /// Clears all tone selections (per-group and global default) and
81    /// re-renders the grid.
82    pub fn reset_all(&self) {
83        self.selections.borrow_mut().clear();
84        *self.default_index.borrow_mut() = None;
85        self.refresh();
86    }
87
88    /// Returns the effective alternate index for `group`: the per-group
89    /// selection if present, otherwise the global default.
90    pub fn selected(&self, group: u32) -> Option<usize> {
91        self.selections.borrow().get(&group).copied().or_else(|| self.default())
92    }
93
94    /// Runs the installed refresh callback, if set.
95    fn refresh(&self) {
96        if let Some(refresh) = self.refresh.borrow().as_ref() {
97            refresh();
98        }
99    }
100}
101
102impl Default for TonePicker {
103    fn default() -> Self {
104        Self::new()
105    }
106}
