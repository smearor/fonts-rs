1//! Detail sidebar for the generic cheat sheet.
2

22/// Populate the sidebar content with details for the given glyph.
23pub(super) fn show_detail<F: CheatSheetFamily>(ctx: &DetailContext, name: &F::Name, codepoint: CodePoint) {
24    let DetailContext {
25        content,
26        search_entry,
27        tone_picker,
28        toasts,
29        category_split,
30        detail_icon_size,
31        ..
32    } = ctx;
33    let detail_size = detail_icon_size.get();
34
35    // Maps the log2 slider value back to a whole-pixel icon size.
36    let log2_to_px = |value: f64| 2.0_f64.powf(value).round();
37
38    // Scales the codepoint preview label via a Pango size attribute
39    // relative to its 96px CSS base (`.sidebar-codepoint-char`).
40    let scale_attrs = |px: f64| {
41        let attrs = AttrList::new();
42        attrs.insert(AttrFloat::new_scale(px / 96.0));
43        attrs
44    };
45    while let Some(child) = content.first_child() {
46        content.remove(&child);
47    }
48
49    let display_name = F::display_name(name);
50    let subtitle = F::subtitle(name);
51    let keywords = F::keywords(name);
52    let categories = F::categories(name);
53
54    // Display name as title.
55    let title_label = Label::builder()
56        .label(&display_name)
57        .css_classes(["sidebar-title"])
58        .halign(Align::Start)
59        .selectable(true)
60        .wrap(true)
61        .build();
62    content.append(&title_label);
63
64    // Source/family as subtitle.
65    let subtitle_label = Label::builder().label(&subtitle).css_classes(["sidebar-subtitle"]).halign(Align::Start).build();
66    content.append(&subtitle_label);
67
68    // Ordering group label (e.g. the emoji picker group) below the subtitle.
69    if let Some(group) = F::ordering().and_then(|o| o.group(name.as_ref())) {
70        let group_label = Label::builder()
71            .label(group.as_str())
72            .css_classes(["sidebar-subtitle", "dim-label"])
73            .halign(Align::Start)
74            .build();
75        content.append(&group_label);
76    }
77
78    // Family section: which font or icon set the glyph ships in.
79    let family_heading = Label::builder()
80        .label("Family")
81        .css_classes(["heading"])
82        .halign(Align::Start)
83        .margin_top(8)
84        .build();
85    content.append(&family_heading);
86    for (field, value) in [("Name", F::FAMILY_NAME), ("Type", F::FAMILY_KIND.as_str()), ("License", F::LICENSE)] {
87        let field_row = Box::builder().orientation(Orientation::Horizontal).spacing(8).build();
88        let field_label = Label::builder().label(field).css_classes(["dim-label"]).width_chars(8).xalign(0.0).build();
89        let value_label = Label::builder().label(value).halign(Align::Start).selectable(true).build();
90        field_row.append(&field_label);
91        field_row.append(&value_label);
92        content.append(&field_row);
93    }
94
95    // Name section (CSS class name for icon fonts, glyph name otherwise).
96    let name_heading = Label::builder()
97        .label(F::NAME_FIELD_LABEL)
98        .css_classes(["heading"])
99        .halign(Align::Start)
100        .margin_top(8)
101        .build();
102    content.append(&name_heading);
103
104    let name_row = Box::builder()
105        .orientation(Orientation::Horizontal)
106        .spacing(8)
107        .halign(Align::Fill)
108        .css_classes(["card"])
109        .margin_top(4)
110        .margin_bottom(4)
111        .margin_start(10)
112        .margin_end(6)
113        .build();
114    let name_val = Label::builder()
115        .label(&display_name)
116        .css_classes(["monospace"])
117        .hexpand(true)
118        .xalign(0.0)
119        .selectable(true)
120        .wrap(true)
121        .max_width_chars(20)
122        .build();
123    let name_copy = Button::builder()
124        .icon_name("edit-copy-symbolic")
125        .tooltip_text(format!("Copy {}", F::NAME_FIELD_LABEL.to_lowercase()))
126        .build();
127    let name_for_copy = display_name.clone();
128    let toasts_for_name = toasts.clone();
129    name_copy.connect_clicked(move |_| {
130        if let Some(display) = gtk4::gdk::Display::default() {
131            display.clipboard().set_text(&name_for_copy);
132            toasts_for_name.add_toast(adw::Toast::new("Copied name"));
133        }
134    });
135    name_row.append(&name_val);
136    name_row.append(&name_copy);
137    content.append(&name_row);
138
139    // Codepoint section.
140    let cp_heading = Label::builder()
141        .label("Codepoint")
142        .css_classes(["heading"])
143        .halign(Align::Start)
144        .margin_top(8)
145        .build();
146    content.append(&cp_heading);
147
148    // Codepoint glyph (very large, own line). Families with icon-theme
149    // glyphs (e.g. emoji, fontless SVG sets) render the registered SVG -
150    // their codepoint is a PUA handle that the system font fallback
151    // would draw as an unrelated glyph. Others render the char as text.
152    let cp_widget: gtk4::Widget = if let Some(icon_name) = F::glyph_icon_name(name) {
153        gtk4::Image::builder()
154            .icon_name(&icon_name)
155            .pixel_size(detail_size.round() as i32)
156            .halign(Align::Center)
157            .build()
158            .upcast()
159    } else {
160        let mut cp_char_classes = vec!["sidebar-codepoint-char"];
161        if let Some(class) = F::GLYPH_CSS_CLASS {
162            cp_char_classes.insert(0, class);
163        }
164        let cp_label = Label::builder()
165            .label(codepoint.as_char().to_string())
166            .css_classes(cp_char_classes)
167            .halign(Align::Center)
168            .selectable(true)
169            .wrap(true)
170            .width_request(200)
171            .build();
172        cp_label.set_attributes(Some(&scale_attrs(detail_size)));
173        cp_label.upcast()
174    };
175
176    // Preview size slider (log2 scale like the grid slider) - the
177    // chosen size persists across detail views via the context. It
178    // sits *above* the preview: resizing the icon grows it downward
179    // instead of shifting the slider up and down.
180    let detail_size_box = Box::builder().orientation(Orientation::Horizontal).spacing(8).halign(Align::Center).build();
181    let detail_size_value = Label::builder().label(format!("{detail_size:.0}px")).css_classes(["monospace"]).build();
182    let detail_slider = icon_size_scale(detail_size, 260);
183    let cp_widget_for_size = cp_widget.clone();
184    let detail_size_value_ref = detail_size_value.clone();
185    let detail_icon_size_ref = detail_icon_size.clone();
186    detail_slider.connect_value_changed(move |slider| {
187        let px = log2_to_px(slider.value());
188        if let Some(image) = cp_widget_for_size.downcast_ref::<gtk4::Image>() {
189            image.set_pixel_size(px as i32);
190        } else if let Some(label) = cp_widget_for_size.downcast_ref::<Label>() {
191            label.set_attributes(Some(&scale_attrs(px)));
192        }
193        detail_size_value_ref.set_label(&format!("{px:.0}px"));
194        detail_icon_size_ref.set(px);
195    });
196    detail_size_box.append(&detail_slider);
197    detail_size_box.append(&detail_size_value);
198    content.append(&detail_size_box);
199    content.append(&cp_widget);
200
201    // Codepoint hex string (own line).
202    let cp_hex = format!("U+{:04X}", codepoint.as_char() as u32);
203    let cp_hex_row = Box::builder()
204        .orientation(Orientation::Horizontal)
205        .spacing(8)
206        .halign(Align::Fill)
207        .css_classes(["card"])
208        .margin_top(4)
209        .margin_bottom(4)
210        .margin_start(10)
211        .margin_end(6)
212        .build();
213    let cp_hex_label = Label::builder()
214        .label(&cp_hex)
215        .css_classes(["monospace", "sidebar-codepoint-hex"])
216        .hexpand(true)
217        .selectable(true)
218        .build();
219    let cp_hex_copy = Button::builder().icon_name("edit-copy-symbolic").tooltip_text("Copy codepoint").build();
220    let toasts_for_hex = toasts.clone();
221    cp_hex_copy.connect_clicked(move |_| {
222        if let Some(display) = gtk4::gdk::Display::default() {
223            display.clipboard().set_text(&cp_hex);
224            toasts_for_hex.add_toast(adw::Toast::new("Copied codepoint"));
225        }
226    });
227    cp_hex_row.append(&cp_hex_label);
228    cp_hex_row.append(&cp_hex_copy);
229    content.append(&cp_hex_row);
230
231    // Copy character button.
232    let cp_copy = Button::builder()
233        .icon_name("edit-copy-symbolic")
234        .label("Copy character")
235        .tooltip_text("Copy character to clipboard")
236        .halign(Align::Center)
237        .build();
238    let cp_char = codepoint.as_char().to_string();
239    let toasts_for_char = toasts.clone();
240    cp_copy.connect_clicked(move |_| {
241        if let Some(display) = gtk4::gdk::Display::default() {
242            display.clipboard().set_text(&cp_char);
243            toasts_for_char.add_toast(adw::Toast::new("Copied character"));
244        }
245    });
246    content.append(&cp_copy);
247
248    // Alias info.
249    if let Some(canonical) = F::alias_display(name) {
250        let alias_label = Label::builder()
251            .label(format!("Alias of: {}", canonical))
252            .css_classes(["dim-label"])
253            .halign(Align::Start)
254            .margin_top(4)
255            .wrap(true)
256            .max_width_chars(24)
257            .build();
258        content.append(&alias_label);
259    }
260
261    // Keywords (clickable buttons).
262    if !keywords.is_empty() {
263        let kw_header = Label::builder()
264            .label("Keywords")
265            .css_classes(["heading"])
266            .halign(Align::Start)
267            .margin_top(12)
268            .build();
269        content.append(&kw_header);
270
271        let kw_box = FlowBox::builder()
272            .column_spacing(4)
273            .row_spacing(4)
274            .halign(Align::Start)
275            .selection_mode(gtk4::SelectionMode::None)
276            .build();
277        for kw_label in keywords.iter() {
278            let btn = Button::builder().label(kw_label).css_classes(["kw-btn"]).build();
279            let search_entry_ref = search_entry.clone();
280            let kw_label = kw_label.clone();
281            btn.connect_clicked(move |_| {
282                search_entry_ref.set_text(&kw_label);
283            });
284            kw_box.insert(&btn, -1);
285        }
286        content.append(&kw_box);
287    }
288
289    // Categories (clickable buttons).
290    if !categories.is_empty() {
291        let cat_header = Label::builder()
292            .label("Categories")
293            .css_classes(["heading"])
294            .halign(Align::Start)
295            .margin_top(12)
296            .build();
297        content.append(&cat_header);
298
299        let cat_box = FlowBox::builder()
300            .column_spacing(4)
301            .row_spacing(4)
302            .halign(Align::Start)
303            .selection_mode(gtk4::SelectionMode::None)
304            .build();
305        for cat_label in categories.iter() {
306            let btn = Button::builder().label(cat_label).css_classes(["cat-btn"]).build();
307            let search_entry_ref = search_entry.clone();
308            let cat_label = cat_label.clone();
309            let split_ref = category_split.clone();
310            btn.connect_clicked(move |_| {
311                search_entry_ref.set_text(&cat_label);
312                // Reveal the axis the user just filtered by.
313                if let Some(split) = &split_ref {
314                    split.set_show_sidebar(true);
315                }
316            });
317            cat_box.insert(&btn, -1);
318        }
319        content.append(&cat_box);
320    }
321
322    // Variant navigation via the ordering contract: a variant links
323    // back to its base; a base with alternates lists them as clickable
324    // glyph cells. Replaces the popover interaction - variants stay
325    // reachable from the sidebar while the grid stays collapsed.
326    if let Some(ord) = F::ordering() {
327        if let Some(base) = ord.base_of(name.as_ref())
328            && let (Some(base_name), Some(base_cp)) = (F::parse_name(base), F::codepoint(base))
329        {
330            let base_header = Label::builder()
331                .label("Variant of")
332                .css_classes(["heading"])
333                .halign(Align::Start)
334                .margin_top(12)
335                .build();
336            content.append(&base_header);
337            let btn = Button::builder()
338                .label(base)
339                .css_classes(["cat-btn"])
340                .halign(Align::Start)
341                .tooltip_text(format!("Open {base}"))
342                .build();
343            let ctx_ref = ctx.clone();
344            btn.connect_clicked(move |_| show_detail::<F>(&ctx_ref, &base_name, base_cp));
345            content.append(&btn);
346        }
347
348        let variants = ord.alternates(name.as_ref());
349        if !variants.is_empty() {
350            let var_header = Label::builder()
351                .label("Variants")
352                .css_classes(["heading"])
353                .halign(Align::Start)
354                .margin_top(12)
355                .build();
356            content.append(&var_header);
357            let var_box = FlowBox::builder()
358                .column_spacing(6)
359                .row_spacing(6)
360                .halign(Align::Start)
361                .selection_mode(gtk4::SelectionMode::None)
362                .build();
363            for &variant in variants {
364                let (Some(variant_name), Some(variant_cp)) = (F::parse_name(variant), F::codepoint(variant)) else {
365                    continue;
366                };
367                let icon: gtk4::Widget = if let Some(icon_name) = F::glyph_icon_name(&variant_name) {
368                    gtk4::Image::builder()
369                        .icon_name(&icon_name)
370                        .pixel_size(32)
371                        .halign(Align::Center)
372                        .build()
373                        .upcast()
374                } else {
375                    // The variant char needs the family font class,
376                    // otherwise Pango falls back to any installed font
377                    // covering the PUA codepoint and draws an
378                    // unrelated glyph.
379                    let mut variant_classes = vec!["cheat-icon-label"];
380                    if let Some(class) = F::GLYPH_CSS_CLASS {
381                        variant_classes.insert(0, class);
382                    }
383                    Label::builder()
384                        .label(variant_cp.as_char().to_string())
385                        .css_classes(variant_classes)
386                        .halign(Align::Center)
387                        .build()
388                        .upcast()
389                };
390                let cell = Box::builder().orientation(Orientation::Vertical).spacing(2).halign(Align::Center).build();
391                cell.append(&icon);
392                cell.append(
393                    &Label::builder()
394                        .label(variant)
395                        .css_classes(["caption", "dim-label"])
396                        .max_width_chars(12)
397                        .wrap(true)
398                        .halign(Align::Center)
399                        .build(),
400                );
401                let btn = Button::builder().child(&cell).css_classes(["flat"]).tooltip_text(variant).build();
402                let ctx_ref = ctx.clone();
403                btn.connect_clicked(move |_| show_detail::<F>(&ctx_ref, &variant_name, variant_cp));
404                var_box.insert(&btn, -1);
405            }
406            content.append(&var_box);
407        }
408
409        // Default-tone picking: a variant becomes the shared default
410        // of its tone group ("set skin tone once, apply everywhere").
411        // The base glyph has no alternates position - it only offers
412        // the reset while a non-default index is stored.
413        if let Some(group) = ord.tone_group(name.as_ref()) {
414            let picked_index = ord
415                .base_of(name.as_ref())
416                .and_then(|base| ord.alternates(base).iter().position(|v| *v == name.as_ref()));
417            let stored = tone_picker.selected(group);
418            if picked_index.is_some() || stored.is_some() {
419                let is_default = picked_index == stored;
420                let label = if is_default { "Reset default tone" } else { "Set as default tone" };
421                let tone_btn = Button::builder()
422                    .label(label)
423                    .css_classes(["cat-btn"])
424                    .halign(Align::Start)
425                    .margin_top(8)
426                    .tooltip_text("Apply this tone to every glyph in the group")
427                    .build();
428                let ctx_ref = ctx.clone();
429                let name_for_tone = name.clone();
430                tone_btn.connect_clicked(move |_| {
431                    match (picked_index, is_default) {
432                        (Some(index), false) => ctx_ref.tone_picker.select(group, index),
433                        _ => ctx_ref.tone_picker.reset(group),
434                    }
435                    // Re-render the sidebar so the button label
436                    // reflects the new selection state.
437                    show_detail::<F>(&ctx_ref, &name_for_tone, codepoint);
438                });
439                content.append(&tone_btn);
440            }
441        }
442    }
443
444    // SVG section.
445    if let Some(svg_xml) = F::glyph_svg(name) {
446        let svg_heading = Label::builder()
447            .label("SVG")
448            .css_classes(["heading"])
449            .halign(Align::Start)
450            .margin_top(12)
451            .build();
452        content.append(&svg_heading);
453
454        let text_view = TextView::builder()
455            .editable(false)
456            .css_classes(["monospace"])
457            .wrap_mode(WrapMode::Char)
458            .margin_top(8)
459            .margin_bottom(8)
460            .margin_start(10)
461            .margin_end(10)
462            .build();
463        let svg_buffer = text_view.buffer();
464        svg_buffer.set_text(&svg_xml);
465        xml_highlight::apply(&svg_buffer, &svg_xml, adw::StyleManager::default().is_dark());
466
467        // `.card` separates the code block from the rest of the
468        // sidebar; propagate_natural_height lets it grow with the
469        // content up to the cap instead of claiming fixed space.
470        let scrolled = ScrolledWindow::builder()
471            .css_classes(["card"])
472            .hexpand(true)
473            .vexpand(true)
474            .min_content_height(80)
475            .max_content_height(300)
476            .propagate_natural_height(true)
477            .child(&text_view)
478            .build();
479        content.append(&scrolled);
480
481        let svg_copy = Button::builder()
482            .icon_name("edit-copy-symbolic")
483            .label("Copy SVG")
484            .tooltip_text("Copy SVG XML to clipboard")
485            .halign(Align::Start)
486            .build();
487        let svg_xml_for_copy = svg_xml;
488        let toasts_for_svg = toasts.clone();
489        svg_copy.connect_clicked(move |_| {
490            if let Some(display) = gtk4::gdk::Display::default() {
491                display.clipboard().set_text(&svg_xml_for_copy);
492                toasts_for_svg.add_toast(adw::Toast::new("Copied SVG"));
493            }
494        });
495        content.append(&svg_copy);
496    }
