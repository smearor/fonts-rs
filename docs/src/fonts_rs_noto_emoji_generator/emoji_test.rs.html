1//! Unicode emoji-test.txt parsing for emoji categories.
2//!
3//! The file has `# group: Category Name` and `# subgroup: Sub Name` headers,
4//! followed by lines like `1F600 ; fully-qualified # 😀 E1.0 grinning face`.
5
6use fonts_rs_model::CodePoint;
7use fonts_rs_model::CodePointCategoryMap;
8
9use std::str::FromStr;
10
11/// Parse emoji-test.txt to build a codepoint→category map.
12///
13/// The file has `# group: Category Name` and `# subgroup: Sub Name` headers,
14/// followed by lines like `1F600 ; fully-qualified # 😀 E1.0 grinning face`.
15pub fn parse_emoji_test(content: &str) -> CodePointCategoryMap {
16    let mut categories = CodePointCategoryMap::new();
17    let mut current_group = String::new();
18
19    for line in content.lines() {
20        let trimmed = line.trim();
21
22        if let Some(rest) = trimmed.strip_prefix("# group:") {
23            current_group = rest.trim().to_string();
24        } else if !trimmed.starts_with('#') && !trimmed.is_empty() {
25            // Parse codepoint: "1F600 ; fully-qualified # ..."
26            if let Some(codepoint_str) = trimmed.split_whitespace().next()
27                && let Ok(cp) = CodePoint::from_str(codepoint_str)
28                && !current_group.is_empty()
29            {
30                categories.insert(cp, current_group.clone());
31            }
32        }
33    }
34
35    categories
36}
