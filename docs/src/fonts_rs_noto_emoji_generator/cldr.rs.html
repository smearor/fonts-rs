1//! CLDR annotation parsing for emoji keywords and semantic names.
2//!
3//! The CLDR JSON uses the actual emoji character as the key.
4//! The `tts` field contains the canonical name (e.g. "grinning face").
5//! The `default` field contains search keywords (e.g. ["face", "grin"]).
6
7use std::collections::HashMap;
8
9use fonts_rs_model::CodePoint;
10use fonts_rs_model::CodePointKeywordMap;
11use fonts_rs_model::CodePointNameMap;
12use fonts_rs_model::GlyphNameMap;
13use serde::Deserialize;
14
15/// CLDR annotation entry: `tts` (name) and `default` (keywords).
16///
17/// Fields are optional because the JSON also contains an `identity` entry
18/// with `language` instead of annotation data.
19#[derive(Deserialize)]
20pub struct CldrAnnotation {
21    #[serde(rename = "tts", default)]
22    tts: Vec<String>,
23    #[serde(rename = "default", default)]
24    default: Vec<String>,
25}
26
27/// CLDR annotations inner structure: contains `identity` and the actual annotations map.
28#[derive(Deserialize)]
29struct CldrAnnotationsInner {
30    #[serde(default)]
31    _identity: serde_json::Value,
32    annotations: HashMap<String, CldrAnnotation>,
33}
34
35/// CLDR annotations root structure.
36#[derive(Deserialize)]
37struct CldrAnnotations {
38    annotations: CldrAnnotationsInner,
39}
40
41/// Convert a CLDR tts name to a kebab-case glyph name.
42///
43/// e.g. "grinning face" → "grinning-face"
44///      "face with tears of joy" → "face-with-tears-of-joy"
45fn name_to_kebab(name: &str) -> String {
46    let kebab: String = name
47        .to_lowercase()
48        .chars()
49        .map(|c| if c.is_ascii_alphanumeric() || c == ' ' { c } else { ' ' })
50        .collect();
51    let kebab: String = kebab.split_whitespace().collect::<Vec<_>>().join("-");
52    kebab
53}

55/// Parse CLDR annotations JSON into (name→codepoint map, keyword map, name map).
56///
57/// The CLDR JSON uses the actual emoji character as the key.
58/// The `tts` field contains the canonical name (e.g. "grinning face").
59/// The `default` field contains search keywords (e.g. ["face", "grin"]).
60pub fn parse_cldr_annotations(json: &str) -> miette::Result<(GlyphNameMap, CodePointKeywordMap, CodePointNameMap)> {
61    let data: CldrAnnotations = serde_json::from_str(json).map_err(|e| miette::miette!("Failed to parse CLDR annotations: {e}"))?;
62
63    let mut name_map = GlyphNameMap::new();
64    let mut keyword_map = CodePointKeywordMap::new();
65    let mut name_by_codepoint = CodePointNameMap::new();
66
67    for (char_str, annotation) in &data.annotations.annotations {
68        // The key is the actual emoji character(s). For single-codepoint emoji,
69        // take the first char. For sequences (e.g. skin tone modifiers), skip
70        // multi-codepoint entries that aren't in the font.
71        let Some(ch) = char_str.chars().next() else { continue };
72        if char_str.chars().count() > 1 {
73            continue;
74        }
75        let codepoint = CodePoint::from(ch);
76
77        // Skip entries without a tts name (e.g. the identity entry)
78        if annotation.tts.is_empty() {
79            continue;
80        }
81
82        // Use the first tts entry as the canonical name
83        if let Some(name) = annotation.tts.first() {
84            let kebab = name_to_kebab(name);
85            if !kebab.is_empty() {
86                name_map.insert(kebab.clone(), codepoint);
87                name_by_codepoint.insert(codepoint, kebab);
88            }
89        }
90
91        if !annotation.default.is_empty() {
92            keyword_map.insert(codepoint, annotation.default.clone());
93        }
94    }
95
96    Ok((name_map, keyword_map, name_by_codepoint))
97}
