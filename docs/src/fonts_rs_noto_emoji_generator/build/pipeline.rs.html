1//! The Noto Emoji build pipeline.
2//!
3//! Extracted from the legacy `fonts-rs-noto-emoji` `build.rs` so every
4//! variant crate runs the identical export: pre-rendered
5//! `emoji_u*.svg` artwork for GResources (full color, original
6//! artwork) plus the variant's TTF for `pixel-drawing` (`ab_glyph`)
7//! and the `font.gresource`.
8//!
9//! Glyph names come from CLDR annotation data
10//! (`metadata/annotations.json`): the emoji sequence encoded in each
11//! `emoji_u*.svg` filename is mapped to the CLDR `tts` name
12//! (kebab-cased). Sequences without a CLDR name keep their raw
13//! filename stem. Emoji sequences are allocated PUA codepoints by
14//! `SvgDirSource`.
15//!
16//! `CARGO_FEATURE_METADATA` gates keyword/category/ordering
17//! generation (the crate's `metadata` feature).
18

47/// Export pipeline version.
48///
49/// Mixed into `FontBuild::extra_hash`. Bump when the export pipeline
50/// itself changes in ways the hash cannot see - e.g. adding or
51/// modifying `SvgPostProcessor`s - otherwise a matching hash would
52/// reuse a stale export.
53const EXPORT_PIPELINE_VERSION: u64 = 1;
54
55/// Variation selector-16 (U+FE0F).
56///
57/// Stripped from CLDR keys and SVG filename codepoints so that
58/// `emoji_u2764.svg` and `emoji_u2764_fe0f.svg` both match the CLDR
59/// entry "❤️" (U+2764 U+FE0F).
60const VARIATION_SELECTOR_16: char = '\u{FE0F}';

62/// Runs the Noto Emoji build for `input`.
63///
64/// `font_path` is derived from `input.variant`
65/// (`resources/{file}.ttf`, crate-relative). `input.svg_dirs` is
66/// merged into a single export via a multi-directory `SvgDirSource`.
67pub fn build(input: &BuildInput) -> miette::Result<()> {
68    let font_path = input.variant.font_path();
69
70    eprintln!("build.rs: active variant: {} -> {}", input.variant.feature_name(), font_path.display());
71
72    // Expose the active variant's rasterization capability as cfgs -
73    // cargo feature cfgs cannot see the build-time default variant
74    // (under --all-features the fallback variant wins), so tests gate
75    // on these instead.
76    println!("cargo::rustc-check-cfg=cfg(noto_emoji_bitmap)");
77    println!("cargo::rustc-check-cfg=cfg(noto_emoji_mono)");
78    if input.variant.is_bitmap() {
79        println!("cargo:rustc-cfg=noto_emoji_bitmap");
80    }
81    if input.variant.is_mono() {
82        println!("cargo:rustc-cfg=noto_emoji_mono");
83    }
84
85    // Set TTF path for pixel-drawing (ab_glyph) at runtime.
86    set_font_path_env(input.font_path_env, &font_path)?;
87
88    // Resource dirs can live outside the crate (links crates) - watch
89    // them explicitly so vendored updates re-run the build.
90    for dir in &input.svg_dirs {
91        println!("cargo::rerun-if-changed={}", dir.display());
92    }
93    println!("cargo::rerun-if-changed={}", input.metadata_dir.display());
94
95    // Parse CLDR annotations into a sequence → name map for the
96    // SvgDirSource name resolver. The base file covers single-codepoint
97    // emoji; the derived file adds sequences (ZWJ, flags, keycaps).
98    let mut sequence_names = HashMap::new();
99    for file_name in ["annotations.json", "annotations-derived.json"] {
100        let path = input.metadata_dir.join(file_name);
101        let json = std::fs::read_to_string(&path).map_err(|e| miette::miette!("Failed to read {}: {e}", path.display()))?;
102        sequence_names.extend(parse_cldr_sequence_names(&json)?);
103    }
104    eprintln!("build.rs: parsed {} CLDR sequence names", sequence_names.len());
105
106    let entry = FontVariant::file(input.variant.feature_name(), input.variant.font_stem());
107    let config = ExportConfig::<NotoEmojiDefinition>::with_variant(entry);
108    let glyph_name_prefix = config.glyph_name_prefix.clone();
109    let include_flags = input.variant.include_flags();
110
111    // GResource SVGs: copy/rename pre-rendered SVGs from the input dirs
112    // using CLDR semantic names. Full color, original artwork quality.
113    // Runs inside FontBuild's export closure so metadata.json and
114    // icons.gresource.xml land in OUT_DIR where the GResource
115    // compilation and code generation expect them.
116    // The resolver closure consumes its own copy; the ordering
117    // generator below borrows `sequence_names` again.
118    let resolver_names = sequence_names.clone();
119    let svg_dirs = input.svg_dirs.clone();
120    // The closure is `move` so `svg_dirs`/`resolver_names` can be
121    // consumed; `config` must stay borrowed for the ordering resolver
122    // below - move a reference instead of the config itself.
123    let config_ref = &config;
124    let mut runner = FontBuild::new(&font_path)
125        .compile_font_gresource()
126        .gresource_prefix(NotoEmojiDefinition::GRESOURCE_PREFIX)
127        .extra_hash(entry.as_str())
128        .extra_hash(EXPORT_PIPELINE_VERSION.to_string());
129    for dir in &input.svg_dirs {
130        runner = runner.extra_hash(format!("{:016x}", hash_svg_dir(dir)));
131    }
132    runner
133        .run(move |_font_path, resources_dir| {
134            let mut source = SvgDirSource::new(&svg_dirs[0]);
135            for dir in &svg_dirs[1..] {
136                source = source.dir(dir.clone());
137            }
138            source
139                .filename_parser(EmojiUFilenameParser)
140                .codepoint_strategy(CodepointStrategy::FromFilename)
141                .name_resolver(move |parsed| {
142                    parsed.codepoints.as_ref().and_then(|codepoints| {
143                        let key: Vec<CodePoint> = codepoints.iter().filter(|cp| cp.as_char() != VARIATION_SELECTOR_16).copied().collect();
144                        resolver_names.get(&key).cloned()
145                    })
146                })
147                .name_prefix(&glyph_name_prefix)
148                .post_process(StripEntityNamespaces)
149                .filter(move |codepoints| include_flags || !is_flag_emoji(codepoints))
150                .export(config_ref, resources_dir)
151        })
152        .map_err(|e| miette::miette!("{e}"))?;
153
154    // Generate variant info for runtime use.
155    config.write_variant_info()?;
156
157    // Generate metadata (keywords + categories + ordering) if enabled.
158    // Keywords come from CLDR annotation data, categories from Unicode
159    // emoji-test.txt. Shortcodes/emoticons from the ordering file become
160    // aliases (resolve_alias + search hit them).
161    if std::env::var("CARGO_FEATURE_METADATA").is_ok() {
162        eprintln!("build.rs: generating emoji metadata (keywords/categories/ordering)...");
163
164        let annotations_path = input.metadata_dir.join(file_name(ANNOTATIONS_FILE));
165        let emoji_test_path = input.metadata_dir.join(file_name(EMOJI_TEST_FILE));
166
167        let annotations_json = std::fs::read_to_string(&annotations_path).map_err(|e| miette::miette!("Failed to read {}: {e}", annotations_path.display()))?;
168        let (_name_map, keyword_map, _name_by_codepoint) = parse_cldr_annotations(&annotations_json)?;
169
170        let emoji_test_content = std::fs::read_to_string(&emoji_test_path).map_err(|e| miette::miette!("Failed to read {}: {e}", emoji_test_path.display()))?;
171        let category_map = parse_emoji_test(&emoji_test_content);
172
173        // Read the generated metadata.json to get the full list of exported glyphs.
174        let metadata_json =
175            std::fs::read_to_string(fonts_rs_generator::build_constants::metadata_path()).map_err(|e| miette::miette!("Failed to read metadata.json: {e}"))?;
176        let entries: Vec<GlyphEntry<String>> = serde_json::from_str(&metadata_json).map_err(|e| miette::miette!("Failed to parse metadata.json: {e}"))?;
177
178        // Emoji picker ordering from googlefonts/emoji-metadata. The
179        // resolver joins each ordered codepoint sequence onto the
180        // exported glyph name, mirroring SvgDirSource's CLDR resolver +
181        // name_prefix + raw-stem fallback, then drops names that are not
182        // exported (unknown upstream emoji or filtered flags).
183        let ordering_path = input.metadata_dir.join(file_name(ORDERING_FILE));
184        let ordering_json = std::fs::read_to_string(&ordering_path).map_err(|e| miette::miette!("Failed to read {}: {e}", ordering_path.display()))?;
185        let ordering = parse_emoji_ordering(&ordering_json)?;
186
187        let exported_names: HashSet<&str> = entries.iter().map(|entry| entry.name.as_str()).collect();
188        let resolve_glyph_name = |sequence: &[CodePoint]| -> Option<String> {
189            let key: Vec<CodePoint> = sequence.iter().filter(|cp| cp.as_char() != VARIATION_SELECTOR_16).copied().collect();
190            let stem = sequence_names.get(&key).cloned().unwrap_or_else(|| {
191                let raw: Vec<String> = key.iter().map(|cp| format!("{:x}", cp.as_char() as u32)).collect();
192                format!("emoji-u{}", raw.join("-"))
193            });
194            let name = format!("{}-{}", config.glyph_name_prefix, normalize_to_kebab(&stem)?);
195            exported_names.contains(name.as_str()).then_some(name)
196        };
197
198        let alias_pairs = EmojiOrderingGenerator::new(&ordering)
199            .run(resolve_glyph_name)
200            .map_err(|e| miette::miette!("Failed to generate ordering tables: {e}"))?;
201
202        EmojiMetadataGenerator::new(keyword_map, category_map)
203            .with_aliases(alias_pairs)
204            .run(&entries)
205            .map_err(|e| miette::miette!("Failed to generate metadata: {e}"))?;
206    }
207
208    Ok(())
209}

211/// Basename of a `metadata/`-prefixed resource constant.

213    constant.rsplit('/').next().unwrap_or(constant)
214}

216/// Parses CLDR annotations into a codepoint-sequence → kebab-name map.
217///
218/// Unlike `parse_cldr_annotations` - which skips multi-codepoint
219/// emoji - this keeps sequences (ZWJ, keycap, flag) because the
220/// pre-rendered SVG set covers them. Variation selectors are stripped
221/// from the key sequence.
222///
223/// Accepts both CLDR JSON shapes: the base file wraps the map under
224/// `annotations.annotations`, the derived file under
225/// `annotationsDerived.annotations`.
226fn parse_cldr_sequence_names(json: &str) -> miette::Result<HashMap<Vec<CodePoint>, String>> {
227    let value: serde_json::Value = serde_json::from_str(json).map_err(|e| miette::miette!("Failed to parse CLDR annotations: {e}"))?;
228    let annotations = value
229        .get("annotations")
230        .or_else(|| value.get("annotationsDerived"))
231        .and_then(|root| root.get("annotations"))
232        .and_then(|map| map.as_object())
233        .ok_or_else(|| miette::miette!("CLDR annotations map not found"))?;
234    let mut names = HashMap::new();
235    for (emoji, annotation) in annotations {
236        let Some(name) = annotation
237            .get("tts")
238            .and_then(|tts| tts.as_array())
239            .and_then(|tts| tts.first())
240            .and_then(|name| name.as_str())
241        else {
242            continue;
243        };
244        let Some(kebab) = normalize_to_kebab(name) else { continue };
245        let codepoints: Vec<CodePoint> = emoji.chars().filter(|c| *c != VARIATION_SELECTOR_16).map(CodePoint::from).collect();
246        if codepoints.is_empty() {
247            continue;
248        }
249        names.insert(codepoints, kebab);
250    }
251    Ok(names)
252}

254/// Returns `true` if the codepoint sequence is a flag emoji.
255///
256/// Covers regional-indicator pairs (e.g. 🇩🇪 = U+1F1E9 U+1F1EA) and
257/// subdivision flags, which start with U+1F3F4 (waving black flag)
258/// followed by tag characters (e.g. 🏴󠁧󠁢󠁥󠁮󠁧󠁿 England).
259fn is_flag_emoji(codepoints: &[CodePoint]) -> bool {
260    const REGIONAL_INDICATOR_START: char = '\u{1F1E6}';
261    const REGIONAL_INDICATOR_END: char = '\u{1F1FF}';
262    const WAVING_BLACK_FLAG: char = '\u{1F3F4}';
263    if codepoints.len() < 2 {
264        return false;
265    }
266    if codepoints[0].as_char() == WAVING_BLACK_FLAG {
267        return true;
268    }
269    codepoints
270        .iter()
271        .all(|cp| (REGIONAL_INDICATOR_START..=REGIONAL_INDICATOR_END).contains(&cp.as_char()))
272}
