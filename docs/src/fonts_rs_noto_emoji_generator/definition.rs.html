1//! `NotoEmojiDefinition` — `FontDefinition` implementation for Noto Emoji.
2//!
3//! This module implements the generic [`FontDefinition`] trait for the Noto Emoji
4//! font family, allowing the `FontDefinition::export_glyphs` pipeline to be used.
5//!
6//! Note: Noto Emoji glyph names in the font are `uniXXXX` format (PostScript
7//! auto-generated), so the build script uses `ExportConfig::export_glyphs_by_name_map` with
8//! CLDR annotation data for semantic names instead of the generic pipeline.
9//! This definition provides the constants and type information for the family.
10

54/// Build-time configuration for the Noto Emoji font family.
55///
56/// Implements [`FontDefinition`] to plug into the generic
57/// [`FontDefinition::export_glyphs`] pipeline.
58///
59/// Note: The build script typically uses `ExportConfig::export_glyphs_by_name_map` with
60/// CLDR annotation data instead of the generic pipeline, because Noto Emoji
61/// glyph names are `uniXXXX` format and require CLDR lookup for semantic names.
62/// This definition still provides the constants and type information.
63pub struct NotoEmojiDefinition;

77impl FontDefinition for NotoEmojiDefinition {
78    type Name = String;
79
80    type Family = NotoEmoji;
81
82    fn normalize_name(raw_glyph_name: &str) -> Option<Self::Name> {
83        // Noto Emoji glyph names are `uniXXXX` format and require CLDR
84        // annotation data for semantic names. The build script uses
85        // `ExportConfig::export_glyphs_by_name_map` with a pre-built name map instead.
86        // This implementation is provided for completeness but returns
87        // `None` for `uni`-prefixed names.
88        if raw_glyph_name.starts_with("uni") || raw_glyph_name.starts_with("u") || raw_glyph_name.starts_with('.') {
89            return None;
90        }
91        Some(raw_glyph_name.to_string())
92    }
93
94    fn should_skip(raw_glyph_name: &str) -> bool {
95        // Skip auto-generated PostScript names; the build script uses
96        // `ExportConfig::export_glyphs_by_name_map` which bypasses this filter.
97        raw_glyph_name.starts_with('.') || raw_glyph_name.starts_with("uni") || raw_glyph_name.starts_with("u")
98    }
99}
