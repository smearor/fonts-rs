1//! Icon metadata: keywords, style categories and ordering.
2//!
3//! Provides keyword lookup for Eva icons, enabling search beyond
4//! simple name matching - e.g. "alarm" finds `eva-bell-symbolic` and
5//! `eva-clock-symbolic`.
6//!
7//! Keywords are sourced from upstream `icons-tags.json` (converted
8//! from `icons-tags.ts`, the search tags powering the official
9//! eva-icons.com cheat sheet). Categories carry the style ("fill" /
10//! "outline"); Eva ships no aliases.
11//!
12//! [`GlyphOrdering`] groups glyphs by style and pairs
13//! `eva-{name}-outline-symbolic` onto its fill base
14//! `eva-{name}-symbolic` - the link powers the detail sidebar's
15//! Variants section and sorts each outline glyph next to its fill
16//! base (styles never collapse).
17//!
18//! This module is only available when the `metadata` feature is enabled.
19

54/// Ordering tables derived once from the generated codemap.
55///
56/// Eva has no upstream ordering JSON, so the tables are computed
57/// lazily from `REVERSE_GLYPHS` (all exported names) plus the
58/// `-outline` style suffix instead of being codegen'd.
59struct OrderingTables {
60    /// Alphabetical index over the *base* glyphs (fill style) -
61    /// variants sort next to their base via `base_of`.
62    order: HashMap<&'static str, u32>,
63    /// Style label per glyph name.
64    group_of: HashMap<&'static str, GroupLabel>,
65    /// `Some(fill_name)` for outline glyphs whose fill pair exists.
66    base_of: HashMap<&'static str, &'static str>,
67    /// `fill_name -> [outline_name]` for every outline pair.
68    alternates_of: HashMap<&'static str, &'static [&'static str]>,
69    /// One group per style with alphabetically sorted glyph names;
70    /// `fill` first - it is the upstream default style.
71    groups: &'static [GlyphGroup],
72}
73
74/// Returns the process-cached [`OrderingTables`].
75fn tables() -> &'static OrderingTables {
76    static TABLES: OnceLock<OrderingTables> = OnceLock::new();
77    TABLES.get_or_init(build_tables)
78}
79
80/// Builds the ordering tables from the generated codemap.
81fn build_tables() -> OrderingTables {
82    let is_outline = |name: &str| name.strip_prefix("eva-").is_some_and(|n| n.ends_with("-outline-symbolic"));
83
84    let mut fill: Vec<&'static str> = Vec::new();
85    let mut outline: Vec<&'static str> = Vec::new();
86    for (&name, _) in REVERSE_GLYPHS.entries() {
87        if is_outline(name) { outline.push(name) } else { fill.push(name) }
88    }
89    fill.sort_unstable();
90
91    let order: HashMap<&'static str, u32> = fill.iter().enumerate().map(|(i, &name)| (name, i as u32)).collect();
92    let mut group_of = HashMap::with_capacity(fill.len() + outline.len());
93    for &name in &fill {
94        group_of.insert(name, GroupLabel::new("fill"));
95    }
96    for &name in &outline {
97        group_of.insert(name, GroupLabel::new("outline"));
98    }
99    outline.sort_unstable();
100
101    let mut base_of = HashMap::with_capacity(outline.len());
102    let mut alternates_of = HashMap::with_capacity(outline.len());
103    for &name in &outline {
104        let stem = name
105            .strip_suffix("-outline-symbolic")
106            .unwrap_or_else(|| name.strip_suffix("-symbolic").unwrap_or(name));
107        let base_name: &'static str = Box::leak(format!("{stem}-symbolic").into_boxed_str());
108        if fill.contains(&base_name) {
109            base_of.insert(name, base_name);
110            alternates_of.insert(base_name, Box::leak(Box::new([name])) as &'static [&'static str]);
111        }
112    }
113
114    let groups: &'static [GlyphGroup] = Box::leak(
115        vec![
116            GlyphGroup::new(GroupLabel::new("fill"), Box::leak(fill.into_boxed_slice())),
117            GlyphGroup::new(GroupLabel::new("outline"), Box::leak(outline.into_boxed_slice())),
118        ]
119        .into_boxed_slice(),
120    );
121
122    OrderingTables {
123        order,
124        group_of,
125        base_of,
126        alternates_of,
127        groups,
128    }
129}

131impl GlyphOrdering for EvaMetadata {
132    fn order_index(&self, name: &str) -> Option<u32> {
133        tables().order.get(name).copied()
134    }
135
136    /// Style of a glyph: `eva-{name}-outline-symbolic` -> "outline",
137    /// `eva-{name}-symbolic` -> "fill".
138    fn group(&self, name: &str) -> Option<GroupLabel> {
139        tables().group_of.get(name).copied()
140    }
141
142    fn group_labels(&self) -> &'static [GroupLabel] {
143        static LABELS: &[GroupLabel] = &[GroupLabel::new("fill"), GroupLabel::new("outline")];
144        LABELS
145    }
146
147    fn groups(&self) -> &'static [GlyphGroup] {
148        tables().groups
149    }
150
151    /// The fill glyph an outline glyph variants:
152    /// `eva-{name}-outline-symbolic` -> `eva-{name}-symbolic`.
153    fn base_of(&self, name: &str) -> Option<&'static str> {
154        tables().base_of.get(name).copied()
155    }
156
157    /// The outline variant of a fill glyph, if it exists.
158    fn alternates(&self, name: &str) -> &'static [&'static str] {
159        tables().alternates_of.get(name).copied().unwrap_or(&[])
160    }
161
162    /// Both styles are first-class browse content - a whole style is
163    /// not a hideable variant (unlike emoji skin tones). The pair
164    /// link still powers the detail sidebar's Variants section and
165    /// keeps outline glyphs sorted next to their fill base.
166    fn collapse_variants(&self) -> bool {
167        false
168    }
169}
