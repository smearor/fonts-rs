1//! Eva Icons font integration for GTK 4 and pixel-drawing.
2//!
3//! Provides icon name resolution, GResource registration, and codepoint
4//! lookup for the Eva Icons font family - both upstream styles in one
5//! export: ~490 `fill` icons (`eva-{name}-symbolic`) plus ~490
6//! `outline` icons (`eva-{name}-outline-symbolic`), sharing the single
7//! webfont TTF's codepoint space.
8//!
9//! Keyword metadata (`metadata` feature) is sourced from upstream
10//! `icons-tags.json` - the search tags powering the official
11//! eva-icons.com cheat sheet.
12//!
13//! ## Quick Start
14//!
15//! ```no_run
16//! # #[cfg(feature = "gtk")]
17//! use fonts_rs_eva::Eva;
18//!
19//! # #[cfg(feature = "gtk")]
20//! // Call once at startup before using any icons
21//! if let Err(e) = Eva::install() {
22//!     eprintln!("Failed to install font: {e}");
23//! }
24//! ```
25//!
26//! Then resolve icon names to Unicode codepoints:
27//!
28//! ```no_run
29//! use fonts_rs_eva::Eva;
30//!
31//! let codepoint = Eva::codepoint("eva-home-symbolic");
32//! ```

59/// Type alias for this font family's glyph name type.
60///
61/// Prefer [`EvaIconName`] from the [`naming`] module - this alias
62/// is provided for consistency with the template structure.
63pub type FamilyName = EvaIconName;
64
65/// Returns all known Eva icons as `(CodePoint, icon_name)` pairs,
66/// sorted alphabetically by icon name.
67pub fn all_icons() -> Vec<(CodePoint, String)> {
68    let mut icons: Vec<(CodePoint, String)> = codepoint_map::GLYPHS
69        .entries()
70        .map(|(cp, name)| (CodePoint::from(*cp), name.to_string()))
71        .collect();
72    icons.sort_by(|a, b| a.1.cmp(&b.1));
73    icons
74}

76#[cfg(test)]
77mod tests {
78    use super::*;
79
80    /// Returns the first icon name in the glyph map.
81    ///
82    /// The fill and outline styles ship slightly different icon sets,
83    /// so tests must derive names from the map instead of hardcoding
84    /// them.
85    fn first_icon_name() -> String {
86        all_icons().into_iter().next().map(|(_, name)| name).expect("at least one icon should exist")
87    }
88
89    /// `codepoint`/`glyph_name` must round-trip for exported icons -
90    /// the chain behind icon theme lookup and the cheat sheet.
91    /// Codepoints differ between styles, so the test derives them
92    /// from the name instead of hardcoding values.
93    #[test]
94    fn codepoint_glyph_name_round_trip() {
95        let name = first_icon_name();
96        let codepoint = Eva::codepoint(&name).expect("icon codepoint should resolve");
97        let resolved = Eva::glyph_name(codepoint).expect("icon name should resolve");
98        assert_eq!(resolved.as_ref(), name);
99    }
100
101    /// The `glyph_prefix` fallback must resolve suffix-only lookups:
102    /// `codepoint("{name}-symbolic")` expands to the full
103    /// `eva-{name}-symbolic` name.
104    #[test]
105    fn codepoint_resolves_suffixed_name() {
106        let name = first_icon_name();
107        let stripped = name.strip_prefix("eva-").expect("icon name should carry the eva- prefix");
108        assert_eq!(Eva::codepoint(stripped), Eva::codepoint(&name));
109    }
110
111    /// `EvaIconName` normalizes every accepted form to the canonical
112    /// `eva-{name}-symbolic` representation.
113    #[test]
114    fn icon_name_normalizes() {
115        for input in ["activity", "eva-activity", "eva-activity-symbolic"] {
116            let name = EvaIconName::parse(input).unwrap_or_else(|| panic!("{input} should parse"));
117            assert_eq!(name.as_ref(), "eva-activity-symbolic", "input: {input}");
118            assert_eq!(name.css_class_name(), "eva-activity", "input: {input}");
119        }
120    }
121
122    /// The combined export names every icon after its style:
123    /// `eva-{name}-symbolic` (fill) or `eva-{name}-outline-symbolic`
124    /// (outline) - and both styles must be present.
125    #[test]
126    fn icon_names_carry_their_style() {
127        let mut fill = 0usize;
128        let mut outline = 0usize;
129        for (_, name) in all_icons() {
130            if let Some(stem) = name.strip_prefix("eva-").and_then(|n| n.strip_suffix("-symbolic")) {
131                if stem.ends_with("-outline") {
132                    outline += 1;
133                } else {
134                    fill += 1;
135                }
136            } else {
137                panic!("glyph name '{name}' does not match eva-*-symbolic");
138            }
139        }
140        assert!(fill > 0, "expected fill icons (eva-*-symbolic)");
141        assert!(outline > 0, "expected outline icons (eva-*-outline-symbolic)");
142    }
143
144    /// Metadata search must find icons by keyword - e.g. upstream tags
145    /// map "alarm" to `bell`, `bell-off`, and `clock`.
146    #[cfg(feature = "metadata")]
147    #[test]
148    fn metadata_keywords_and_search() {
149        use fonts_rs_model::GlyphMetadata;
150        let results = metadata::EvaMetadata.search("alarm");
151        assert!(results.contains(&"eva-bell-symbolic"), "search for 'alarm' should find bell, got: {results:?}");
152        assert!(results.contains(&"eva-clock-symbolic"), "search for 'alarm' should find clock, got: {results:?}");
153        assert!(metadata::EvaMetadata.search("").is_empty(), "empty query should return no results");
154    }
155
156    /// `glyph_svg` must return the vector artwork from the GResource -
157    /// the chain behind `gtk_icon_theme` and the detail sidebar.
158    #[cfg(feature = "gtk")]
159    #[test]
160    fn glyph_svg_returns_artwork() {
161        let svg = Eva::glyph_svg(first_icon_name()).expect("glyph SVG should be in the GResource");
162        assert!(svg.contains("<svg"), "expected an SVG document, got: {}", &svg[..svg.len().min(200)]);
163    }
