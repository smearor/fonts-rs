1//! DSEG14 fourteen-segment display font integration for GTK 4.
2//!
3//! Provides glyph name resolution, GResource registration, and codepoint
4//! lookup for the DSEG14 font family - a fourteen-segment display font
5//! suitable for retro-style digital readouts.
6//!
7//! The active variant (style + weight) is selected at build time via
8//! Cargo features (e.g. `classic-regular`, `modern-mini-bold`).
9//!
10//! ## Quick Start
11//!
12//! ```no_run
13//! # #[cfg(feature = "gtk")]
14//! use fonts_rs_fourteen_segment::FourteenSegment;
15//!
16//! # #[cfg(feature = "gtk")]
17//! // Call once at startup before using any glyphs
18//! if let Err(e) = FourteenSegment::install() {
19//!     eprintln!("Failed to install font: {e}");
20//! }
21//! ```
22//!
23//! Then resolve glyph names to Unicode codepoints:
24//!
25//! ```no_run
26//! use fonts_rs_fourteen_segment::FourteenSegment;
27//!
28//! let codepoint = FourteenSegment::codepoint("zero");
29//! ```

58#[cfg(test)]
59mod tests {
60    use super::*;
61
62    #[test]
63    fn all_glyphs_is_not_empty() {
64        let count = FourteenSegment::glyph_count();
65        assert!(count > 0, "DSEG14 should export at least one glyph");
66    }
67
68    #[test]
69    fn all_glyphs_names_start_with_dseg14() {
70        for (_, name) in FourteenSegment::all_glyphs() {
71            assert!(name.as_ref().starts_with("dseg14-"), "glyph name '{name}' should start with 'dseg14-'");
72        }
73    }
74
75    #[test]
76    fn all_glyphs_names_are_lowercase() {
77        for (_, name) in FourteenSegment::all_glyphs() {
78            assert_eq!(name.as_ref(), name.as_ref().to_lowercase(), "glyph name '{name}' should be lowercase");
79        }
80    }
81
82    #[test]
83    fn all_glyphs_names_contain_no_underscores() {
84        for (_, name) in FourteenSegment::all_glyphs() {
85            assert!(!name.as_ref().contains('_'), "glyph name '{name}' should not contain underscores");
86        }
87    }
88
89    #[test]
90    fn all_glyphs_codepoints_are_unique() {
91        let codepoints: Vec<char> = FourteenSegment::all_glyphs().map(|(c, _)| c.as_char()).collect();
92        let unique: std::collections::HashSet<char> = codepoints.iter().copied().collect();
93        assert_eq!(codepoints.len(), unique.len(), "all codepoints should be unique");
94    }
