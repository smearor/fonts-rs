1//! Blobmoji `bitmap-noflags` font integration for GTK 4 and pixel-drawing.
2//!
3//! Provides glyph name resolution, GResource registration, and codepoint
4//! lookup for the `bitmap-noflags` variant of the Blobmoji font family - a
5//! community-maintained emoji font recreating Google's classic "blob"
6//! emoji style (CBDT bitmap font `NotoColorEmoji-noflags`, flag emoji
7//! filtered out), covering all Unicode emoji codepoints with semantic
8//! names from CLDR annotation data. GResource SVGs ship full-color
9//! artwork regardless of variant.
10//!
11//! ## Quick Start
12//!
13//! ```no_run
14//! # #[cfg(feature = "gtk")]
15//! use fonts_rs_blobmoji_bitmap_noflags::Blobmoji;
16//!
17//! # #[cfg(feature = "gtk")]
18//! // Call once at startup before using any glyphs
19//! if let Err(e) = Blobmoji::install() {
20//!     eprintln!("Failed to install font: {e}");
21//! }
22//! ```
23//!
24//! Then resolve glyph names to Unicode codepoints:
25//!
26//! ```no_run
27//! use fonts_rs_blobmoji_bitmap_noflags::Blobmoji;
28//!
29//! let codepoint = Blobmoji::codepoint("grinning-face");
30//! ```

58/// Whether the bundled TTF rasterizes visible output via `ab_glyph`.

60/// CBDT strikes are PNG blobs - pixel-drawing produces colored output.
61/// `false` for COLRv1 variants whose `glyf` table is empty.

70/// Returns all known Blobmoji glyphs as `(CodePoint, glyph_name)` pairs,
71/// sorted alphabetically by glyph name.
72///
73/// Emoji sequences (ZWJ, keycap) are represented by their
74/// allocated PUA codepoint; the original sequence is recorded in
75/// `metadata.json`. The `noflags` variant filters flag emoji at
76/// export time.
77pub fn all_emoji() -> Vec<(CodePoint, String)> {
78    let mut emojis: Vec<(CodePoint, String)> = codepoint_map::GLYPHS
79        .entries()
80        .map(|(cp, name)| (CodePoint::from(*cp), name.to_string()))
81        .collect();
82    emojis.sort_by(|a, b| a.1.cmp(&b.1));
83    emojis
84}

137    /// The TTF bundled for `pixel-drawing` must load via `ab_glyph` and
138    /// resolve emoji codepoints through its cmap.

148    /// `render_text` must produce a non-empty, non-black buffer: CBDT
149    /// strikes are PNG blobs positioned by their bottom bearing - a
150    /// regression test for the "1px line" and "black box" failure modes.
151    #[cfg(all(feature = "gtk", feature = "render"))]
152    #[test]
153    fn render_text_blits_bitmap_strikes() {
154        let font = Blobmoji::font().expect("BLOBMOJI_FONT_PATH TTF should load via ab_glyph");
155        let (width, height, pixels) = fonts_rs_test::render_text(font, "\u{1F600}", 64.0);
156        assert!(width > 1 && height > 1, "rendered buffer should have real dimensions, got {width}x{height}");
157        let non_black = pixels.chunks_exact(4).filter(|px| px[0] > 0 || px[1] > 0 || px[2] > 0).count();
158        assert!(non_black > 0, "rendered buffer should contain non-black pixels");
159    }

161    /// `compose_svg` must keep prefixed namespaces bound: Blobmoji SVGs
162    /// use `ns1:href` on `<use>` elements, declared via `xmlns:ns1` on
163    /// the root tag. Stripping the root for composition must re-attach
164    /// the declaration to the nested `<svg>`.
165    #[cfg(feature = "gtk")]
166    #[test]
167    fn compose_svg_keeps_namespaces_bound() {
168        let svg = Blobmoji::compose_svg("\u{1F680}", 200.0);
169        assert!(svg.contains("viewBox"), "composed svg should embed the rocket glyph");
170        if svg.contains("ns1:href") {
171            assert!(svg.contains("xmlns:ns1"), "ns1:href used but xmlns:ns1 declaration missing");
172        }
173    }
