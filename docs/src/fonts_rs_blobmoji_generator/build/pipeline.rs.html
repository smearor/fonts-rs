1//! The Blobmoji build pipeline.
2//!
3//! Extracted from the legacy `fonts-rs-blobmoji` `build.rs` so every
4//! variant crate runs the identical export: pre-rendered
5//! `emoji_u*.svg` artwork for GResources (full color, original
6//! artwork) plus the variant's TTF for `pixel-drawing` (`ab_glyph`)
7//! and the `font.gresource`.
8//!
9//! Glyph names come from CLDR annotation data (`annotations.json` +
10//! `annotations-derived.json` in the shared `fonts-rs-emoji-metadata`
11//! crate): the emoji sequence encoded in each `emoji_u*.svg` filename
12//! is mapped to the CLDR `tts` name (kebab-cased). Sequences without a
13//! CLDR name keep their raw filename stem. Emoji sequences are
14//! allocated PUA codepoints by `SvgDirSource`.
15//!
16//! Unlike Noto Emoji, the vendored Blobmoji SVGs use prefixed
17//! `xmlns:ns1` namespaces that must stay bound - no
18//! `StripEntityNamespaces` post-processing here.
19//!
20//! `CARGO_FEATURE_METADATA` gates keyword/category/ordering
21//! generation (the crate's `metadata` feature).
22

50/// Export pipeline version.
51///
52/// Mixed into `FontBuild::extra_hash`. Bump when the export pipeline
53/// itself changes in ways the hash cannot see - e.g. adding or
54/// modifying `SvgPostProcessor`s - otherwise a matching hash would
55/// reuse a stale export.
56const EXPORT_PIPELINE_VERSION: u64 = 1;
57
58/// Variation selector-16 (U+FE0F).
59///
60/// Stripped from CLDR keys and SVG filename codepoints so that
61/// `emoji_u2764.svg` and `emoji_u2764_fe0f.svg` both match the CLDR
62/// entry "❤️" (U+2764 U+FE0F).
63const VARIATION_SELECTOR_16: char = '\u{FE0F}';
64
65/// Runs the Blobmoji build for `input`.
66///
67/// `font_path` is derived from `input.variant`
68/// (`resources/{file}.ttf`, crate-relative). `input.svg_dirs` is
69/// merged into a single export via a multi-directory `SvgDirSource`.
70pub fn build(input: &BuildInput) -> miette::Result<()> {
71    let font_path = input.variant.font_path();
72
73    eprintln!("build.rs: active variant: {} -> {}", input.variant.feature_name(), font_path.display());
74
75    // Expose the active variant's bitmap capability as a cfg - cargo
76    // feature cfgs cannot see the build-time variant, so tests gate on
77    // this instead.
78    println!("cargo::rustc-check-cfg=cfg(blobmoji_bitmap)");
79    if input.variant.is_bitmap() {
80        println!("cargo:rustc-cfg=blobmoji_bitmap");
81    }
82
83    // Set TTF path for pixel-drawing (ab_glyph) at runtime.
84    set_font_path_env(input.font_path_env, &font_path)?;
85
86    // Resource dirs can live outside the crate (links crates) - watch
87    // them explicitly so vendored updates re-run the build.
88    for dir in &input.svg_dirs {
89        println!("cargo::rerun-if-changed={}", dir.display());
90    }
91    println!("cargo::rerun-if-changed={}", input.metadata_dir.display());
92
93    // Parse CLDR annotations into a sequence → name map for the
94    // SvgDirSource name resolver. The base file covers single-codepoint
95    // emoji; the derived file adds sequences (ZWJ, flags, keycaps).
96    let mut sequence_names = HashMap::new();
97    for file_name in ["annotations.json", "annotations-derived.json"] {
98        let path = input.metadata_dir.join(file_name);
99        let json = std::fs::read_to_string(&path).map_err(|e| miette::miette!("Failed to read {}: {e}", path.display()))?;
100        sequence_names.extend(parse_cldr_sequence_names(&json)?);
101    }
102    eprintln!("build.rs: parsed {} CLDR sequence names", sequence_names.len());
103
104    let entry = FontVariant::file(input.variant.feature_name(), input.variant.font_stem());
105    let config = ExportConfig::<BlobmojiDefinition>::with_variant(entry);
106    let glyph_name_prefix = config.glyph_name_prefix.clone();
107    let include_flags = input.variant.include_flags();
108
109    // GResource SVGs: copy/rename pre-rendered SVGs from the input dirs
110    // using CLDR semantic names. Full color, original artwork quality.
111    // Runs inside FontBuild's export closure so metadata.json and
112    // icons.gresource.xml land in OUT_DIR where the GResource
113    // compilation and code generation expect them.
114    // The resolver closure consumes its own copy; the ordering
115    // generator below borrows `sequence_names` again.
116    let resolver_names = sequence_names.clone();
117    let svg_dirs = input.svg_dirs.clone();
118    // The closure is `move` so `svg_dirs`/`resolver_names` can be
119    // consumed; `config` must stay borrowed for the ordering resolver
120    // below - move a reference instead of the config itself.
121    let config_ref = &config;
122    let mut runner = FontBuild::new(&font_path)
123        .compile_font_gresource()
124        .gresource_prefix(BlobmojiDefinition::GRESOURCE_PREFIX)
125        .extra_hash(entry.as_str())
126        .extra_hash(EXPORT_PIPELINE_VERSION.to_string());
127    for dir in &input.svg_dirs {
128        runner = runner.extra_hash(format!("{:016x}", hash_svg_dir(dir)));
129    }
130    runner
131        .run(move |_font_path, resources_dir| {
132            let mut source = SvgDirSource::new(&svg_dirs[0]);
133            for dir in &svg_dirs[1..] {
134                source = source.dir(dir.clone());
135            }
136            source
137                .filename_parser(EmojiUFilenameParser)
138                .codepoint_strategy(CodepointStrategy::FromFilename)
139                .name_resolver(move |parsed| {
140                    parsed.codepoints.as_ref().and_then(|codepoints| {
141                        let key: Vec<CodePoint> = codepoints.iter().filter(|cp| cp.as_char() != VARIATION_SELECTOR_16).copied().collect();
142                        resolver_names.get(&key).cloned()
143                    })
144                })
145                .name_prefix(&glyph_name_prefix)
146                .filter(move |codepoints| include_flags || !is_flag_emoji(codepoints))
147                .export(config_ref, resources_dir)
148        })
149        .map_err(|e| miette::miette!("{e}"))?;
150
151    // Generate variant info for runtime use.
152    config.write_variant_info()?;
153
154    // Generate metadata (keywords + categories + ordering) if enabled.
155    // Keywords come from CLDR annotation data, categories from Unicode
156    // emoji-test.txt. Shortcodes/emoticons from the ordering file become
157    // aliases (resolve_alias + search hit them).
158    if std::env::var("CARGO_FEATURE_METADATA").is_ok() {
159        eprintln!("build.rs: generating emoji metadata (keywords/categories/ordering)...");
160
161        let annotations_path = input.metadata_dir.join(file_name(ANNOTATIONS_FILE));
162        let emoji_test_path = input.metadata_dir.join(file_name(EMOJI_TEST_FILE));
163
164        let annotations_json = std::fs::read_to_string(&annotations_path).map_err(|e| miette::miette!("Failed to read {}: {e}", annotations_path.display()))?;
165        let (_name_map, keyword_map, _name_by_codepoint) = parse_cldr_annotations(&annotations_json)?;
166
167        let emoji_test_content = std::fs::read_to_string(&emoji_test_path).map_err(|e| miette::miette!("Failed to read {}: {e}", emoji_test_path.display()))?;
168        let category_map = parse_emoji_test(&emoji_test_content);
169
170        // Read the generated metadata.json to get the full list of exported glyphs.
171        let metadata_json =
172            std::fs::read_to_string(fonts_rs_generator::build_constants::metadata_path()).map_err(|e| miette::miette!("Failed to read metadata.json: {e}"))?;
173        let entries: Vec<GlyphEntry<String>> = serde_json::from_str(&metadata_json).map_err(|e| miette::miette!("Failed to parse metadata.json: {e}"))?;
174
175        // Emoji picker ordering from googlefonts/emoji-metadata. The
176        // resolver joins each ordered codepoint sequence onto the
177        // exported glyph name, mirroring SvgDirSource's CLDR resolver +
178        // name_prefix + raw-stem fallback, then drops names that are not
179        // exported (unknown upstream emoji or filtered flags).
180        let ordering_path = input.metadata_dir.join(file_name(ORDERING_FILE));
181        let ordering_json = std::fs::read_to_string(&ordering_path).map_err(|e| miette::miette!("Failed to read {}: {e}", ordering_path.display()))?;
182        let ordering = parse_emoji_ordering(&ordering_json)?;
183
184        let exported_names: HashSet<&str> = entries.iter().map(|entry| entry.name.as_str()).collect();
185        let resolve_glyph_name = |sequence: &[CodePoint]| -> Option<String> {
186            let key: Vec<CodePoint> = sequence.iter().filter(|cp| cp.as_char() != VARIATION_SELECTOR_16).copied().collect();
187            let stem = sequence_names.get(&key).cloned().unwrap_or_else(|| {
188                let raw: Vec<String> = key.iter().map(|cp| format!("{:x}", cp.as_char() as u32)).collect();
189                format!("emoji-u{}", raw.join("-"))
190            });
191            let name = format!("{}-{}", config.glyph_name_prefix, normalize_to_kebab(&stem)?);
192            exported_names.contains(name.as_str()).then_some(name)
193        };
194
195        let alias_pairs = EmojiOrderingGenerator::new(&ordering)
196            .run(resolve_glyph_name)
197            .map_err(|e| miette::miette!("Failed to generate ordering tables: {e}"))?;
198
199        EmojiMetadataGenerator::new(keyword_map, category_map)
200            .with_aliases(alias_pairs)
201            .run(&entries)
202            .map_err(|e| miette::miette!("Failed to generate metadata: {e}"))?;
203    }
204
205    Ok(())
206}

213/// Parses CLDR annotations into a codepoint-sequence → kebab-name map.
214///
215/// Unlike `parse_cldr_annotations` - which skips multi-codepoint
216/// emoji - this keeps sequences (ZWJ, keycap, flag) because the
217/// pre-rendered SVG set covers them. Variation selectors are stripped
218/// from the key sequence.
219///
220/// Accepts both CLDR JSON shapes: the base file wraps the map under
221/// `annotations.annotations`, the derived file under
222/// `annotationsDerived.annotations`.
223fn parse_cldr_sequence_names(json: &str) -> miette::Result<HashMap<Vec<CodePoint>, String>> {
224    let value: serde_json::Value = serde_json::from_str(json).map_err(|e| miette::miette!("Failed to parse CLDR annotations: {e}"))?;
225    let annotations = value
226        .get("annotations")
227        .or_else(|| value.get("annotationsDerived"))
228        .and_then(|root| root.get("annotations"))
229        .and_then(|map| map.as_object())
230        .ok_or_else(|| miette::miette!("CLDR annotations map not found"))?;
231    let mut names = HashMap::new();
232    for (emoji, annotation) in annotations {
233        let Some(name) = annotation
234            .get("tts")
235            .and_then(|tts| tts.as_array())
236            .and_then(|tts| tts.first())
237            .and_then(|name| name.as_str())
238        else {
239            continue;
240        };
241        let Some(kebab) = normalize_to_kebab(name) else { continue };
242        let codepoints: Vec<CodePoint> = emoji.chars().filter(|c| *c != VARIATION_SELECTOR_16).map(CodePoint::from).collect();
243        if codepoints.is_empty() {
244            continue;
245        }
246        names.insert(codepoints, kebab);
247    }
248    Ok(names)
249}

251/// Returns `true` if the codepoint sequence is a flag emoji.
252///
253/// Covers regional-indicator pairs (e.g. 🇩🇪 = U+1F1E9 U+1F1EA) and
254/// subdivision flags, which start with U+1F3F4 (waving black flag)
255/// followed by tag characters (e.g. 🏴󠁧󠁢󠁥󠁮󠁧󠁿 England).
256fn is_flag_emoji(codepoints: &[CodePoint]) -> bool {
257    const REGIONAL_INDICATOR_START: char = '\u{1F1E6}';
258    const REGIONAL_INDICATOR_END: char = '\u{1F1FF}';
259    const WAVING_BLACK_FLAG: char = '\u{1F3F4}';
260    if codepoints.len() < 2 {
261        return false;
262    }
263    if codepoints[0].as_char() == WAVING_BLACK_FLAG {
264        return true;
265    }
266    codepoints
267        .iter()
268        .all(|cp| (REGIONAL_INDICATOR_START..=REGIONAL_INDICATOR_END).contains(&cp.as_char()))
