1//! Noto Emoji `color-flags` font integration for GTK 4 and pixel-drawing.
2//!
3//! Provides glyph name resolution, GResource registration, and codepoint
4//! lookup for the `color-flags` variant of the Noto Emoji font family -
5//! Google's COLRv1 vector emoji font (`Noto-COLRv1`, including flag emoji)
6//! covering all Unicode emoji codepoints with semantic names from CLDR
7//! annotation data. GResource SVGs ship full-color artwork regardless of
8//! variant.
9//!
10//! ## Quick Start
11//!
12//! ```no_run
13//! # #[cfg(feature = "gtk")]
14//! use fonts_rs_noto_emoji_color_flags::NotoEmoji;
15//!
16//! # #[cfg(feature = "gtk")]
17//! // Call once at startup before using any glyphs
18//! if let Err(e) = NotoEmoji::install() {
19//!     eprintln!("Failed to install font: {e}");
20//! }
21//! ```
22//!
23//! Then resolve glyph names to Unicode codepoints:
24//!
25//! ```no_run
26//! use fonts_rs_noto_emoji_color_flags::NotoEmoji;
27//!
28//! let codepoint = NotoEmoji::codepoint("grinning-face");
29//! ```

69/// Returns all known Noto Emoji glyphs as `(CodePoint, glyph_name)` pairs,
70/// sorted alphabetically by glyph name.
71pub fn all_emoji() -> Vec<(CodePoint, String)> {
72    let mut emojis: Vec<(CodePoint, String)> = codepoint_map::GLYPHS
73        .entries()
74        .map(|(cp, name)| (CodePoint::from(*cp), name.to_string()))
75        .collect();
76    emojis.sort_by(|a, b| a.1.cmp(&b.1));
77    emojis
78}

80#[cfg(test)]
81mod tests {
82    use super::*;
83
84    /// `glyph_name` must resolve emoji codepoints to variant-prefixed
85    /// icon names, and `glyph_svg` must return the colored vector
86    /// artwork from the GResource - the chain behind `gtk_icon_theme`
87    /// and `svg_gresource`.
88    #[cfg(feature = "gtk")]
89    #[test]
90    fn glyph_svg_returns_colored_artwork() {
91        let name = NotoEmoji::glyph_name('\u{1F600}').expect("grinning face should resolve");
92        let svg = NotoEmoji::glyph_svg(&name).expect("glyph SVG should be in the GResource");
93        assert!(svg.contains("<svg"), "expected an SVG document, got: {}", &svg[..svg.len().min(200)]);
94        assert!(
95            svg.contains("stop-color") || svg.contains("fill:") || svg.contains("fill=\""),
96            "expected colored artwork in {name}"
97        );
98    }
99
100    /// `compose_svg` must produce a parseable document for glyphs whose
101    /// source SVGs carried Illustrator DOCTYPE entity namespaces
102    /// (`emoji_u1f680.svg` declares `xmlns:i="&ns_ai;"`). The export
103    /// strips them via `StripEntityNamespaces`; a regression would
104    /// surface as "namespace prefix" parse errors in GTK.
105    #[cfg(feature = "gtk")]
106    #[test]
107    fn compose_svg_has_no_unbound_prefixes() {
108        let svg = NotoEmoji::compose_svg("\u{1F680}", 200.0);
109        assert!(svg.contains("<svg"), "expected a composed SVG document");
110        assert!(!svg.contains("&ns_"), "entity references must not leak into the composed SVG");
111        assert!(!svg.contains("i:aipgf"), "Illustrator metadata must be stripped");
112    }
113}
