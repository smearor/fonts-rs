1//! End-to-end fixture for custom icon sets.
2//!
3//! The build script runs `IconSetBuild` on `icons.toml`: two crate
4//! sources (Lucide + Tabler/filled) resolved via `DEP_*_SVG_DIR`, a
5//! local `dir` source, renames in every source, per-source
6//! post-processors. `src/icons.rs` is the generated `iconset.rs`
7//! module verbatim - exactly what an application would write.
8
9pub mod icons {
10    include!(concat!(env!("OUT_DIR"), "/iconset.rs"));
11}
12
13pub use icons::FixtureIconName;
14pub use icons::FixtureIcons;
15
16#[cfg(test)]
17mod tests {
18    use super::*;
19
20    #[test]
21    fn whitelist_size_matches_glyph_count() {
22        assert_eq!(FixtureIcons::all_glyphs().count(), 7);
23    }
24
25    #[test]
26    fn renamed_icons_use_exported_names() {
27        let names: Vec<String> = FixtureIcons::all_glyphs().map(|(_, name)| name.as_ref().to_string()).collect();
28        assert!(names.contains(&"fixture-install-symbolic".to_string()));
29        assert!(names.contains(&"fixture-check-filled-symbolic".to_string()));
30        assert!(names.contains(&"fixture-brand-symbolic".to_string()));
31        assert!(!names.contains(&"fixture-download-symbolic".to_string()));
32        assert!(!names.contains(&"fixture-old-brand-symbolic".to_string()));
33    }
34
35    #[test]
36    fn dir_source_icons_export() {
37        let names: Vec<String> = FixtureIcons::all_glyphs().map(|(_, name)| name.as_ref().to_string()).collect();
38        assert!(names.contains(&"fixture-custom-arrow-symbolic".to_string()));
39    }
40
41    #[cfg(feature = "gtk")]
42    #[test]
43    fn glyphs_register_and_resolve_via_gresource() {
44        use fonts_rs_model::GtkFont;
45
46        FixtureIcons::register_glyphs().expect("register icons.gresource");
47
48        for icon_name in [
49            "fixture-check-symbolic",
50            "fixture-install-symbolic",
51            "fixture-trash-symbolic",
52            "fixture-check-filled-symbolic",
53            "fixture-heart-symbolic",
54            "fixture-custom-arrow-symbolic",
55            "fixture-brand-symbolic",
56        ] {
57            gio::resources_lookup_data(&format!("/com/example/iconset-test/icons/scalable/actions/{icon_name}.svg"), gio::ResourceLookupFlags::NONE)
58                .unwrap_or_else(|_| panic!("{icon_name}.svg must be in the gresource"));
59        }
60
61        // Renamed-away and unwhitelisted icons must be absent.
62        for absent in ["fixture-download-symbolic", "fixture-old-brand-symbolic", "fixture-house-symbolic"] {
63            assert!(
64                gio::resources_lookup_data(&format!("/com/example/iconset-test/icons/scalable/actions/{absent}.svg"), gio::ResourceLookupFlags::NONE,).is_err(),
65                "{absent} must not be exported"
66            );
67        }
68    }
69
70    #[test]
71    fn icon_name_constants_exist() {
72        assert_eq!(icons::icon_names::FIXTURE_CHECK_SYMBOLIC, "fixture-check-symbolic");
73        assert_eq!(icons::icon_names::FIXTURE_INSTALL_SYMBOLIC, "fixture-install-symbolic");
74        assert_eq!(icons::icon_names::FIXTURE_BRAND_SYMBOLIC, "fixture-brand-symbolic");
75    }
