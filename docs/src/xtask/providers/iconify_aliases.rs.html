15/// `iconify_aliases` provider: derives a kebab-case `alias ->
16/// canonical` map from an `@iconify-json` set's alias table.
17///
18/// `with.iconify_set` names the set (e.g. `heroicons`), `with.version`
19/// pins the package range (default `1`) and `with.suffixes` lists
20/// style suffixes stripped from both alias and parent names
21/// (longest first). Aliases for sizes/styles outside `with.suffixes`
22/// (e.g. `-16-filled` when only `-24-*` is vendored) are dropped, as
23/// are aliases resolving to a hidden or missing icon.
24pub struct IconifyAliases;
25
26/// Top-level shape of an iconify `icons.json`.
27#[derive(Deserialize)]
28struct IconifySet {
29    /// Icon table: icon name -> icon record (only `hidden` used).
30    #[serde(default)]
31    icons: HashMap<String, IconifyIcon>,
32    /// Alias table: alias name -> alias record.
33    #[serde(default)]
34    aliases: HashMap<String, IconifyAlias>,
35}
36
37/// Minimal shape of an iconify icon record.
38#[derive(Deserialize)]
39struct IconifyIcon {
40    /// Iconify hides deprecated/duplicate icons from pickers.
41    #[serde(default)]
42    hidden: bool,
43}
44
45/// One iconify alias record.
46#[derive(Deserialize)]
47struct IconifyAlias {
48    /// Canonical icon name the alias resolves to.
49    parent: String,
50}
51
52impl Provider for IconifyAliases {
53    fn name(&self) -> &'static str {
54        "iconify_aliases"
55    }
56
57    fn generate(&self, context: &ProviderContext) -> miette::Result<ProviderOutput> {
58        let set = required_string(context.with, "iconify_set", self.name())?;
59        let version = match context.with.get("version") {
60            Some(toml::Value::String(value)) => value.clone(),
61            Some(toml::Value::Integer(value)) => value.to_string(),
62            Some(_) => return Err(miette!("iconify_aliases: 'with.version' must be a string or integer")),
63            None => "1".to_string(),
64        };
65        let mut suffixes = required_string_list(context.with, "suffixes", self.name())?;
66        suffixes.sort_by_key(|suffix| std::cmp::Reverse(suffix.len()));
67
68        let url = format!("https://unpkg.com/@iconify-json/{set}@{version}/icons.json");
69        let body = context.http_get(&url)?;
70        let input = LockedInput {
71            url,
72            sha256: sha256_hex(&body),
73            provider: self.name().to_string(),
74        };
75
76        let iconify: IconifySet = serde_json::from_slice(&body)
77            .into_diagnostic()
78            .map_err(|e| e.wrap_err("iconify_aliases: failed to parse icons.json"))?;
79
80        let mut aliases = BTreeMap::new();
81        for (alias, record) in &iconify.aliases {
82            if !suffixes.iter().any(|suffix| alias.ends_with(suffix.as_str())) && has_other_size_suffix(alias) {
83                continue;
84            }
85            match iconify.icons.get(&record.parent) {
86                Some(icon) if icon.hidden => continue,
87                None => continue,
88                _ => {}
89            }
90            let alias = strip_suffix(alias, &suffixes);
91            let parent = strip_suffix(&record.parent, &suffixes);
92            if alias == parent {
93                continue;
94            }
95            match aliases.get(&alias) {
96                Some(existing) if existing != &parent => {
97                    return Err(miette!("iconify_aliases: alias '{alias}' maps to both '{existing}' and '{parent}'"));
98                }
99                Some(_) => {}
100                None => {
101                    aliases.insert(alias, parent);
102                }
103            }
104        }
105
106        let content = to_json(&aliases).map_err(|e| e.wrap_err("iconify_aliases: failed to serialize aliases"))?;
107        Ok(ProviderOutput {
108            content: content.into_bytes(),
109            inputs: vec![input],
110        })
111    }
112}

114/// Whether `name` ends with `-<digits>-<word>` - i.e. it belongs to
115/// a size variant that is not vendored (e.g. `-16-filled` or
116/// `-32-light` when only `-24-*` is selected). Aliases carrying a
117/// declared suffix are kept by the caller before this check.
118fn has_other_size_suffix(name: &str) -> bool {
119    let Some((head, _style)) = name.rsplit_once('-') else {
120        return false;
121    };
122    let Some((_, size)) = head.rsplit_once('-') else {
123        return false;
124    };
125    !size.is_empty() && size.chars().all(|c| c.is_ascii_digit())
126}
127
128/// Removes the first matching style suffix from `name`.
129fn strip_suffix(name: &str, suffixes: &[String]) -> String {
130    for suffix in suffixes {
131        if let Some(stripped) = name.strip_suffix(suffix.as_str()) {
132            return stripped.to_string();
133        }
134    }
135    name.to_string()
136}

138#[cfg(test)]
139mod tests {
140    use super::*;
141
142    /// Detects unvendored `-<size>-<style>` suffixes only.
143    #[test]
144    fn detects_other_size_suffixes() {
145        assert!(has_other_size_suffix("app-span-16-filled"));
146        assert!(has_other_size_suffix("arrow-growth-20-regular"));
147        assert!(has_other_size_suffix("text-add-32-light"));
148        assert!(!has_other_size_suffix("arrow-down-square"));
149        assert!(!has_other_size_suffix("fill-24"));
150    }
151
152    /// Longest suffix wins; no suffix leaves the name untouched.
153    #[test]
154    fn strips_longest_suffix() {
155        let suffixes = vec!["-16-solid".to_string(), "-20-solid".to_string(), "-solid".to_string()];
156        assert_eq!(strip_suffix("code-solid", &suffixes), "code");
157        assert_eq!(strip_suffix("viewfinder-dot-20-solid", &suffixes), "viewfinder-dot");
158        assert_eq!(strip_suffix("viewfinder-dot-16-solid", &suffixes), "viewfinder-dot");
159        assert_eq!(strip_suffix("academic-cap", &suffixes), "academic-cap");
160    }
