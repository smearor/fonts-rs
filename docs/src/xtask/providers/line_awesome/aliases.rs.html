11/// `line_awesome_aliases` provider: emits `alias -> canonical stem`
12/// pairs for the FA compatibility classes. `line-awesome.css` carries
13/// the FA4 layer (`.la.la-{name}`: alias key without the `la-`
14/// prefix), `font-awesome-line-awesome/css/all.css` the FA5 layer
15/// (`.fa-{name}`: alias key verbatim). Each class codepoint resolves
16/// to its canonical `$la-` variable name and then to the svg stem -
17/// `{name}` first, `{name}-solid` when only the solid variant exists
18/// (e.g. `fa-hotel` -> `bed-solid`). Classes resolving to their own
19/// canonical name are dropped.
20pub struct LineAwesomeAliases;
21
22impl Provider for LineAwesomeAliases {
23    fn name(&self) -> &'static str {
24        "line_awesome_aliases"
25    }
26
27    fn generate(&self, context: &ProviderContext) -> miette::Result<ProviderOutput> {
28        let checkout = context.checkout()?;
29        let scss = std::fs::read_to_string(checkout.join(source::VARIABLES_SCSS))
30            .into_diagnostic()
31            .map_err(|e| e.wrap_err("line_awesome_aliases: failed to read _variables.scss"))?;
32        let line_awesome_css = std::fs::read_to_string(checkout.join(source::LINE_AWESOME_CSS))
33            .into_diagnostic()
34            .map_err(|e| e.wrap_err("line_awesome_aliases: failed to read line-awesome.css"))?;
35        let font_awesome_css = std::fs::read_to_string(checkout.join(source::FONT_AWESOME_CSS))
36            .into_diagnostic()
37            .map_err(|e| e.wrap_err("line_awesome_aliases: failed to read all.css"))?;
38
39        let variables = source::parse_variables(&scss)?;
40        let mut canonical_names: BTreeMap<u32, String> = BTreeMap::new();
41        for (name, codepoint) in &variables {
42            canonical_names.insert(*codepoint, name.clone());
43        }
44        let stems = source::svg_stems(&checkout.join(source::SVG_DIR))?;
45
46        let mut aliases = BTreeMap::new();
47        let classes = source::icon_classes(&line_awesome_css, "la-")
48            .into_iter()
49            .chain(source::icon_classes(&font_awesome_css, "fa-"));
50        for (class, codepoint) in classes {
51            let alias = class.strip_prefix("la-").unwrap_or(&class).to_string();
52            let Some(canonical) = canonical_names.get(&codepoint) else {
53                continue;
54            };
55            if alias == *canonical {
56                continue;
57            }
58            let Some(stem) = resolve_stem(canonical, &stems) else {
59                continue;
60            };
61            match aliases.get(&alias) {
62                Some(existing) if existing != &stem => {
63                    return Err(miette!("line_awesome_aliases: '{alias}' maps to both '{existing}' and '{stem}'"));
64                }
65                Some(_) => {}
66                None => {
67                    aliases.insert(alias, stem);
68                }
69            }
70        }
71
72        let content = to_json(&aliases).map_err(|e| e.wrap_err("line_awesome_aliases: failed to serialize"))?;
73        // The vendored line-awesome metadata files end with a newline.
74        Ok(ProviderOutput::file(format!("{content}\n").into_bytes()))
75    }
76}

78/// Resolves a canonical variable name to its svg file stem: `{name}`
79/// wins, `{name}-solid` is the fallback for solid-only icons.
80fn resolve_stem(name: &str, stems: &BTreeSet<String>) -> Option<String> {
81    if stems.contains(name) {
82        return Some(name.to_string());
83    }
84    let solid = format!("{name}-solid");
85    stems.contains(&solid).then_some(solid)
86}
