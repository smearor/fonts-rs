16/// Parses `$la-{name}: \fXXX;` declarations into a name -> codepoint
17/// map. Non-icon variables (paths, colors, sizes) carry no `\f` escape
18/// and are ignored.
19pub fn parse_variables(scss: &str) -> miette::Result<BTreeMap<String, u32>> {
20    let mut variables = BTreeMap::new();
21    for (line_number, line) in scss.lines().enumerate() {
22        let line = line.trim();
23        let Some(rest) = line.strip_prefix("$la-") else {
24            continue;
25        };
26        let Some((name, value)) = rest.split_once(':') else {
27            continue;
28        };
29        let Some(codepoint) = parse_css_escape(value) else {
30            continue;
31        };
32        let name = name.trim().to_string();
33        if variables.insert(name.clone(), codepoint).is_some() {
34            return Err(miette!("line-awesome: duplicate variable '$la-{name}' at line {}", line_number + 1));
35        }
36    }
37    Ok(variables)
38}

40/// Extracts `.{class}... { content: "\fXXX"; }` rules and returns the
41/// `(class, codepoint)` pairs whose class carries `prefix` (`la-` /
42/// `fa-`). For compound selectors the icon class is the last prefixed
43/// token (`.la.la-address-book-o` -> `la-address-book-o`).
44pub fn icon_classes(css: &str, prefix: &str) -> Vec<(String, u32)> {
45    let mut classes = Vec::new();
46    for rule in css.split('}') {
47        let Some((selector, body)) = rule.split_once('{') else {
48            continue;
49        };
50        let Some(class) = selector
51            .split(['.', ',', ' ', '\t', '\n'])
52            .filter_map(|token| token.split(':').next())
53            .rfind(|token| token.starts_with(prefix))
54        else {
55            continue;
56        };
57        let Some(codepoint) = parse_css_escape(body) else {
58            continue;
59        };
60        classes.push((class.to_string(), codepoint));
61    }
62    classes
63}

65/// Reads the `content: "\fXXX"` escape out of a declaration block or
66/// scss value.
67fn parse_css_escape(text: &str) -> Option<u32> {
68    let start = text.find('\\')? + 1;
69    let digits: String = text[start..].chars().take_while(|c| c.is_ascii_hexdigit()).collect();
70    if digits.is_empty() {
71        return None;
72    }
73    u32::from_str_radix(&digits, 16).ok()
74}

76/// Lists the sorted file stems of every `*.svg` directly below `dir`.
77pub fn svg_stems(dir: &Path) -> miette::Result<BTreeSet<String>> {
78    let mut stems = BTreeSet::new();
79    for entry in std::fs::read_dir(dir).into_diagnostic()? {
80        let entry = entry.into_diagnostic()?;
81        let path = entry.path();
82        if path.extension().is_some_and(|ext| ext == "svg")
83            && let Some(stem) = path.file_stem()
84        {
85            stems.insert(stem.to_string_lossy().into_owned());
86        }
87    }
88    Ok(stems)
89}

95    /// Parses `$la-{name}: \fXXX;` and ignores non-icon variables.
96    #[test]
97    fn parses_icon_variables() {
98        let scss = "$la-font-path: \"../fonts\" !default;\n$la-500px: \\f26e;\n$la-address-book: \\f2b9;\n$la-version: 1.3.0 !default;\n";
99        let vars = parse_variables(scss).unwrap();
100        assert_eq!(vars.len(), 2);
101        assert_eq!(vars["500px"], 0xf26e);
102        assert_eq!(vars["address-book"], 0xf2b9);
103    }
104
105    /// Takes the last prefixed class of compound selectors.
106    #[test]
107    fn extracts_icon_classes() {
108        let css = ".la.la-address-book-o:before {\n  content: \"\\f2b9\"; }\n\n.la-fw {\n  width: 1em; }\n.fa-arrows-alt:before {\n  content: \"\\f0b2\"; }";
109        assert_eq!(icon_classes(css, "la-"), vec![("la-address-book-o".to_string(), 0xf2b9)]);
110        assert_eq!(icon_classes(css, "fa-"), vec![("fa-arrows-alt".to_string(), 0xf0b2)]);
111    }
