7/// A metadata provider: generates file content from the upstream
8/// checkout and/or external inputs, driven by `[[metadata]] with.*`
9/// parameters.
10pub trait Provider {
11    /// Registry name as used in `metadata.provider`.
12    fn name(&self) -> &'static str;
13
14    /// Generates the output content for one `[[metadata]]` entry.
15    fn generate(&self, context: &ProviderContext) -> miette::Result<ProviderOutput>;
16}
17
18/// Everything a provider needs for one `[[metadata]]` invocation.
19pub struct ProviderContext<'a> {
20    /// Root of the upstream sparse checkout; `None` when the manifest
21    /// has no git `[upstream]` (URL-only vendoring).
22    pub checkout: Option<&'a Path>,
23    /// The `with.*` parameter table.
24    pub with: &'a Table,
25    /// Shared per-sync HTTP fetch cache.
26    pub fetch: &'a FetchCache,
27}
28
29impl<'a> ProviderContext<'a> {
30    /// Fetches `url` via the shared cache (one request per URL per
31    /// sync, regardless of how many providers need it).
32    pub fn http_get(&self, url: &str) -> miette::Result<Vec<u8>> {
33        self.fetch.get(url)
34    }
35
36    /// The upstream checkout root. Fails for manifests without a git
37    /// `[upstream]` section - such providers must source purely via
38    /// `http_get`/`with` inputs.
39    pub fn checkout(&self) -> miette::Result<&'a Path> {
40        self.checkout.ok_or_else(|| miette!("provider requires a git [upstream] checkout"))
41    }
42}
43
44/// The result of a provider run: file content plus every external
45/// input fetched while producing it.
46pub struct ProviderOutput {
47    /// Generated file content, written to the declared output path.
48    pub content: Vec<u8>,
49    /// External non-git inputs fetched during generation (recorded in
50    /// the lockfile for provenance).
51    pub inputs: Vec<LockedInput>,
52}
53
54impl ProviderOutput {
55    /// Convenience constructor for outputs without external inputs.
56    pub fn file(content: Vec<u8>) -> Self {
57        Self { content, inputs: Vec::new() }
58    }
59}

61/// Reads a required string parameter from a `with` table.
62pub fn required_string(with: &Table, key: &str, provider: &str) -> miette::Result<String> {
63    with.get(key)
64        .and_then(|value| value.as_str())
65        .map(str::to_string)
66        .ok_or_else(|| miette!("Provider '{provider}' requires 'with.{key}' (string)"))
67}
68
69/// Reads a required string-list parameter from a `with` table.
70pub fn required_string_list(with: &Table, key: &str, provider: &str) -> miette::Result<Vec<String>> {
71    with.get(key)
72        .and_then(|value| value.as_array())
73        .map(|items| items.iter().filter_map(|item| item.as_str().map(str::to_string)).collect())
74        .ok_or_else(|| miette!("Provider '{provider}' requires 'with.{key}' (list of strings)"))
75}
