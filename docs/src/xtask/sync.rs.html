24/// Manifest file name looked up in each vendored crate.
25const MANIFEST_NAME: &str = "upstream.toml";

27impl Crate {
28    /// Runs `cargo xtask sync -p <package>`: resolves the manifest
29    /// ref, ensures the cached checkout, materializes artifacts,
30    /// runs metadata providers and writes `upstream.lock`. With
31    /// `update`, re-pins `ref` to the latest upstream state first
32    /// (newest version tag for tag pins, default HEAD for commit
33    /// pins; floating branch refs need no rewrite) and persists the
34    /// new pin in `upstream.toml`.
35    pub fn sync(&self, update: bool) -> miette::Result<()> {
36        let crate_dir = self.dir.clone();
37        let manifest_path = crate_dir.join(MANIFEST_NAME);
38        let mut manifest = Manifest::load(&manifest_path)?;
39
40        // Git upstream is optional: manifests made purely of `url`
41        // artifacts have no checkout at all.
42        let mut upstream_context = None;
43        if let Some(upstream) = &mut manifest.upstream {
44            let git = GitSource::new(&upstream.repo);
45            if update
46                && let Some((new_ref, _)) = git.update_target(&upstream.git_ref)?
47                && new_ref != upstream.git_ref
48            {
49                eprintln!("{}: {} -> {new_ref} (re-pinned)", upstream.repo, upstream.git_ref);
50                rewrite_manifest_ref(&manifest_path, &upstream.git_ref, &new_ref)?;
51                upstream.git_ref = new_ref;
52            }
53            let commit = git.resolve(&upstream.git_ref)?;
54            eprintln!("{}: {} -> {commit}", upstream.repo, upstream.git_ref);
55
56            let sparse = sparse_paths(
57                manifest.artifacts.iter().filter(|artifact| artifact.url.is_none()).filter_map(|artifact| {
58                    artifact
59                        .source
60                        .clone()
61                        .map(|source| (source, artifact.pattern.clone(), artifact.exclude.clone()))
62                }),
63                upstream
64                    .license
65                    .iter()
66                    .map(|license| license.source.clone())
67                    .chain(upstream.sparse.iter().cloned()),
68            );
69            let source = git.checkout(&commit, &sparse)?;
70            eprintln!("{}: {}", if source.cache_hit { "cache hit" } else { "fetched" }, source.checkout.display());
71            upstream_context = Some((upstream, commit, source));
72        }
73        let upstream = upstream_context;
74        let checkout_path = upstream.as_ref().map(|(_, _, source)| source.checkout.clone());
75
76        let fetch = FetchCache::default();
77        let mut artifacts = Vec::new();
78        let mut materialized: HashMap<String, (String, String)> = HashMap::new();
79
80        if let Some((upstream, _, source)) = &upstream
81            && let Some(license) = &upstream.license
82        {
83            copy_file(&source.checkout, &crate_dir, &license.source, &license.target, &mut artifacts, &mut materialized)?;
84        }
85        let empty_checkout = PathBuf::new();
86        for artifact in &manifest.artifacts {
87            materialize(
88                checkout_path.as_deref().unwrap_or(&empty_checkout),
89                &crate_dir,
90                artifact,
91                &fetch,
92                &mut artifacts,
93                &mut materialized,
94            )?;
95        }
96
97        let mut inputs: Vec<LockedInput> = Vec::new();
98        for spec in &manifest.metadata {
99            let provider = resolve_provider(&spec.provider)?;
100            let context = ProviderContext {
101                checkout: checkout_path.as_deref(),
102                with: &spec.with,
103                fetch: &fetch,
104            };
105            let output = provider
106                .generate(&context)
107                .map_err(|e| e.wrap_err(format!("Provider '{}' failed", spec.provider)))?;
108            let target = crate_dir.join(&spec.output);
109            write_file(&target, &output.content)?;
110            artifacts.push(LockedArtifact {
111                target: spec.output.clone(),
112                sha256: sha256_hex(&output.content),
113                source: format!("provider:{}", spec.provider),
114            });
115            inputs.extend(output.inputs);
116            eprintln!("provider {} -> {}", spec.provider, spec.output);
117        }
118
119        inputs.sort_by(|a, b| a.url.cmp(&b.url));
120        inputs.dedup_by(|a, b| a.url == b.url);
121
122        let lockfile = Lockfile {
123            upstream: upstream.map(|(upstream, commit, _)| LockedUpstream {
124                repo: upstream.repo.clone(),
125                git_ref: upstream.git_ref.clone(),
126                commit,
127            }),
128            artifacts,
129            inputs,
130        };
131
132        remove_stale(&crate_dir, &lockfile)?;
133        lockfile.save(&crate_dir.join(LOCKFILE_NAME))?;
134        eprintln!("wrote {} ({} artifacts, {} inputs)", LOCKFILE_NAME, lockfile.artifacts.len(), lockfile.inputs.len());
135        Ok(())
136    }
137}

139/// Rewrites `ref = "<old>"` to `ref = "<new>"` in `upstream.toml`,
140/// replacing the whole line (stale pin annotations are dropped - the
141/// bump is reviewed in the commit diff anyway).
142fn rewrite_manifest_ref(manifest_path: &Path, old_ref: &str, new_ref: &str) -> miette::Result<()> {
143    let text = std::fs::read_to_string(manifest_path).into_diagnostic()?;
144    let mut replaced = false;
145    let lines: Vec<String> = text
146        .lines()
147        .map(|line| {
148            if !replaced && line.trim_start().starts_with("ref =") && line.contains(&format!("\"{old_ref}\"")) {
149                replaced = true;
150                format!("ref = \"{new_ref}\"")
151            } else {
152                line.to_string()
153            }
154        })
155        .collect();
156    if !replaced {
157        return Err(miette!("ref = \"{old_ref}\" not found in {}", manifest_path.display()));
158    }
159    let mut body = lines.join("\n");
160    if text.ends_with('\n') {
161        body.push('\n');
162    }
163    std::fs::write(manifest_path, body)
164        .into_diagnostic()
165        .map_err(|e| e.wrap_err(format!("Failed to write {}", manifest_path.display())))
166}

168/// Copies `source` (checkout-relative file) to `target`
169/// (crate-relative file) and records it.
170fn copy_file(
171    checkout: &Path,
172    crate_dir: &Path,
173    source: &str,
174    target: &str,
175    artifacts: &mut Vec<LockedArtifact>,
176    materialized: &mut HashMap<String, (String, String)>,
177) -> miette::Result<()> {
178    let source_path = checkout.join(source);
179    if !source_path.is_file() {
180        return Err(miette!("Artifact source '{source}' does not exist in checkout"));
181    }
182    let sha = sha256_file(&source_path)?;
183    copy_dedup(&source_path, &crate_dir.join(target), source, target, &sha, materialized)?;
184    artifacts.push(LockedArtifact {
185        target: target.to_string(),
186        sha256: sha,
187        source: source.to_string(),
188    });
189    Ok(())
190}

192/// Materializes one `[[artifacts]]` entry: a 1:1 file copy, or a
193/// flattened glob copy of a directory.
194fn materialize(
195    checkout: &Path,
196    crate_dir: &Path,
197    artifact: &crate::manifest::Artifact,
198    fetch: &FetchCache,
199    artifacts: &mut Vec<LockedArtifact>,
200    materialized: &mut HashMap<String, (String, String)>,
201) -> miette::Result<()> {
202    if let Some(url) = &artifact.url {
203        return match &artifact.source {
204            None => {
205                if artifact.pattern.is_some() || !artifact.exclude.is_empty() {
206                    return Err(miette!("Artifact '{url}': 'pattern'/'exclude' need a 'source' path inside the archive"));
207                }
208                fetch_url(crate_dir, artifact, url, fetch, artifacts, materialized)
209            }
210            Some(source) => materialize_archive(crate_dir, artifact, url, source, fetch, artifacts, materialized),
211        };
212    }
213    let Some(source_path) = &artifact.source else {
214        return Err(miette!("Artifact '{}' needs 'source' or 'url'", artifact.target));
215    };
216    let source = checkout.join(source_path);
217    if source.is_file() {
218        return copy_file(checkout, crate_dir, source_path, &artifact.target, artifacts, materialized);
219    }
220    if !source.is_dir() {
221        return Err(miette!("Artifact source '{source_path}' does not exist in checkout"));
222    }
223
224    let pattern = artifact.pattern.as_deref().unwrap_or("**/*");
225    let mut builder = GlobSetBuilder::new();
226    builder.add(Glob::new(pattern).map_err(|e| miette!("Invalid pattern '{pattern}': {e}"))?);
227    let glob = builder.build().into_diagnostic()?;
228
229    let mut exclude_builder = GlobSetBuilder::new();
230    for exclude in &artifact.exclude {
231        exclude_builder.add(Glob::new(exclude).map_err(|e| miette!("Invalid exclude glob '{exclude}': {e}"))?);
232    }
233    let excludes = exclude_builder.build().into_diagnostic()?;
234
235    let mut files: Vec<(String, PathBuf)> = Vec::new();
236    collect_matching(&source, &source, &glob, &excludes, &mut files)?;
237    files.sort();
238    for (relative, source_path) in files {
239        let filename = source_path
240            .file_name()
241            .ok_or_else(|| miette!("Artifact '{}' produced a file without a name", source_path.display()))?;
242        let target = format!("{}/{}", artifact.target.trim_end_matches('/'), filename.to_string_lossy());
243        let sha = sha256_file(&source_path)?;
244        copy_dedup(&source_path, &crate_dir.join(&target), &relative, &target, &sha, materialized)?;
245        artifacts.push(LockedArtifact {
246            target,
247            sha256: sha,
248            source: relative,
249        });
250    }
251    Ok(())
252}

254/// Recursively collects `(checkout-relative path, absolute path)`
255/// pairs matching `glob` but not `excludes` below `dir`.
256fn collect_matching(root: &Path, dir: &Path, glob: &globset::GlobSet, excludes: &globset::GlobSet, files: &mut Vec<(String, PathBuf)>) -> miette::Result<()> {
257    for entry in std::fs::read_dir(dir).into_diagnostic()? {
258        let entry = entry.into_diagnostic()?;
259        let path = entry.path();
260        if path.is_dir() {
261            collect_matching(root, &path, glob, excludes, files)?;
262        } else {
263            let relative = path.strip_prefix(root).unwrap_or(&path).to_string_lossy().into_owned();
264            if glob.is_match(&relative) && !excludes.is_match(&relative) {
265                files.push((relative, path));
266            }
267        }
268    }
269    Ok(())
270}

272/// Materializes a `url`+`source` artifact: fetches an archive
273/// (`.tgz`/`.tar.gz`/`.zip`) and extracts `source` - a member file
274/// for 1:1 copies, or a member directory globbed with
275/// `pattern`/`exclude` and flattened into `target` like a checkout
276/// directory. Provenance is `url#member`.
277fn materialize_archive(
278    crate_dir: &Path,
279    artifact: &crate::manifest::Artifact,
280    url: &str,
281    source: &str,
282    fetch: &FetchCache,
283    artifacts: &mut Vec<LockedArtifact>,
284    materialized: &mut HashMap<String, (String, String)>,
285) -> miette::Result<()> {
286    let body = fetch.get(url)?;
287
288    let pattern = artifact.pattern.as_deref().unwrap_or("**/*");
289    let mut builder = GlobSetBuilder::new();
290    builder.add(Glob::new(pattern).map_err(|e| miette!("Invalid pattern '{pattern}': {e}"))?);
291    let glob = builder.build().into_diagnostic()?;
292    let mut exclude_builder = GlobSetBuilder::new();
293    for exclude in &artifact.exclude {
294        exclude_builder.add(Glob::new(exclude).map_err(|e| miette!("Invalid exclude glob '{exclude}': {e}"))?);
295    }
296    let excludes = exclude_builder.build().into_diagnostic()?;
297    let prefix = format!("{}/", source.trim_end_matches('/'));
298
299    let mut emit = |member: &str, content: &[u8]| -> miette::Result<bool> {
300        let target = if member == source {
301            artifact.target.clone()
302        } else if let Some(relative) = member.strip_prefix(&prefix) {
303            if !glob.is_match(relative) || excludes.is_match(relative) {
304                return Ok(false);
305            }
306            let Some(basename) = std::path::Path::new(relative).file_name() else {
307                return Ok(false);
308            };
309            format!("{}/{}", artifact.target.trim_end_matches('/'), basename.to_string_lossy())
310        } else {
311            return Ok(false);
312        };
313        let provenance = format!("{url}#{member}");
314        let sha = sha256_hex(content);
315        if dedup_target(materialized, &provenance, &target, &sha)? {
316            write_file(&crate_dir.join(&target), content)?;
317        }
318        artifacts.push(LockedArtifact {
319            target,
320            sha256: sha,
321            source: provenance,
322        });
323        Ok(true)
324    };
325
326    let mut matched = 0usize;
327    if url.ends_with(".tgz") || url.ends_with(".tar.gz") {
328        let mut archive = tar::Archive::new(flate2::read::GzDecoder::new(body.as_slice()));
329        for entry in archive.entries().into_diagnostic()? {
330            let mut entry = entry.into_diagnostic()?;
331            if !entry.header().entry_type().is_file() {
332                continue;
333            }
334            let member = entry.path().into_diagnostic()?.to_string_lossy().into_owned();
335            let mut content = Vec::new();
336            std::io::Read::read_to_end(&mut entry, &mut content).into_diagnostic()?;
337            if emit(&member, &content)? {
338                matched += 1;
339            }
340        }
341    } else if url.ends_with(".zip") {
342        let mut archive = zip::ZipArchive::new(std::io::Cursor::new(body)).into_diagnostic()?;
343        for index in 0..archive.len() {
344            let mut entry = archive.by_index(index).into_diagnostic()?;
345            if entry.is_dir() {
346                continue;
347            }
348            let member = entry.name().into_diagnostic()?.into_owned();
349            let mut content = Vec::new();
350            std::io::Read::read_to_end(&mut entry, &mut content).into_diagnostic()?;
351            if emit(&member, &content)? {
352                matched += 1;
353            }
354        }
355    } else {
356        return Err(miette!("Artifact '{url}': only .tgz/.tar.gz/.zip archives are supported"));
357    }
358    if matched == 0 {
359        return Err(miette!("Artifact '{url}#{source}': no archive members matched"));
360    }
361    Ok(())
362}

364/// Materializes a `url` artifact: fetches the URL via the shared
365/// fetch cache and writes it verbatim to `target` (a single file).
366/// The URL itself is recorded as the artifact `source`.
367fn fetch_url(
368    crate_dir: &Path,
369    artifact: &crate::manifest::Artifact,
370    url: &str,
371    fetch: &FetchCache,
372    artifacts: &mut Vec<LockedArtifact>,
373    materialized: &mut HashMap<String, (String, String)>,
374) -> miette::Result<()> {
375    let body = fetch.get(url)?;
376    let sha = sha256_hex(&body);
377    if dedup_target(materialized, url, &artifact.target, &sha)? {
378        write_file(&crate_dir.join(&artifact.target), &body)?;
379    }
380    artifacts.push(LockedArtifact {
381        target: artifact.target.clone(),
382        sha256: sha,
383        source: url.to_string(),
384    });
385    Ok(())
386}

388/// Copies `source_path` to `target_path` unless another source file
389/// with identical content was already materialized there (flattened
390/// directory dedup). Conflicting contents for the same target are an
391/// error.
392fn copy_dedup(
393    source_path: &Path,
394    target_path: &Path,
395    source: &str,
396    target: &str,
397    sha: &str,
398    materialized: &mut HashMap<String, (String, String)>,
399) -> miette::Result<()> {
400    if dedup_target(materialized, source, target, sha)? {
401        write_file(target_path, &std::fs::read(source_path).into_diagnostic()?)?;
402    }
403    Ok(())
404}

406/// Claims `target` for `source`/`sha` in the dedup map. Returns
407/// `true` when the caller must write the file, `false` when an
408/// identical artifact already materialized there; conflicting
409/// content for the same target is an error.
410fn dedup_target(materialized: &mut HashMap<String, (String, String)>, source: &str, target: &str, sha: &str) -> miette::Result<bool> {
411    if let Some((first_source, first_sha)) = materialized.get(target) {
412        if first_sha != sha {
413            return Err(miette!("Artifacts '{first_source}' and '{source}' both map to '{target}' with different content"));
414        }
415        return Ok(false);
416    }
417    materialized.insert(target.to_string(), (source.to_string(), sha.to_string()));
418    Ok(true)
419}

421/// Writes `content` to `path`, creating parent directories.
422fn write_file(path: &Path, content: &[u8]) -> miette::Result<()> {
423    if let Some(parent) = path.parent() {
424        std::fs::create_dir_all(parent).into_diagnostic()?;
425    }
426    std::fs::write(path, content)
427        .into_diagnostic()
428        .map_err(|e| e.wrap_err(format!("Failed to write {}", path.display())))
429}

431/// Deletes files recorded in the previous lockfile that are no longer
432/// produced, so removed upstream artifacts do not linger.
433fn remove_stale(crate_dir: &Path, lockfile: &Lockfile) -> miette::Result<()> {
434    let previous = match Lockfile::load(&crate_dir.join(LOCKFILE_NAME)) {
435        Ok(Some(previous)) => previous,
436        Ok(None) => return Ok(()),
437        Err(error) => {
438            eprintln!("warning: ignoring unparseable previous {LOCKFILE_NAME} ({error}); stale artifacts may linger");
439            return Ok(());
440        }
441    };
442    let current: HashSet<&str> = lockfile.artifacts.iter().map(|artifact| artifact.target.as_str()).collect();
443    for artifact in &previous.artifacts {
444        if current.contains(artifact.target.as_str()) {
445            continue;
446        }
447        let path = crate_dir.join(&artifact.target);
448        if path.is_file() {
449            eprintln!("removing stale artifact {}", artifact.target);
450            std::fs::remove_file(&path).into_diagnostic()?;
451        }
452    }
453    Ok(())
454}

461        let dir = std::env::temp_dir().join("xtask-test-rewrite-ref");
462        std::fs::create_dir_all(&dir).unwrap();
463        let manifest = dir.join("upstream.toml");

468        assert!(!text.contains("vendored 2020-01-01"), "stale comment kept: {text}");

485    /// Flattened copies dedup identical content and reject conflicts.
486    #[test]
487    fn copy_dedup_detects_conflicts() {
488        let dir = std::env::temp_dir().join(format!("xtask-test-{}", std::process::id()));
489        let _ = std::fs::remove_dir_all(&dir);
490        std::fs::create_dir_all(&dir).unwrap();
491        let a = dir.join("a.txt");
492        let b = dir.join("b.txt");
493        std::fs::write(&a, b"same").unwrap();
494        std::fs::write(&b, b"different").unwrap();
495        let target = dir.join("out.txt");
496
497        let mut materialized = HashMap::new();
498        let sha_a = sha256_file(&a).unwrap();
499        copy_dedup(&a, &target, "a", "out.txt", &sha_a, &mut materialized).unwrap();
500        // Same content from another source: dedup, no error.
501        copy_dedup(&a, &target, "a2", "out.txt", &sha_a, &mut materialized).unwrap();
502        // Different content at the same target: conflict.
503        let sha_b = sha256_file(&b).unwrap();
504        assert!(copy_dedup(&b, &target, "b", "out.txt", &sha_b, &mut materialized).is_err());
505        let _ = std::fs::remove_dir_all(&dir);
506    }
