9impl CheckRow {
10    /// Re-fetches live URL artifacts and provider inputs and reports
11    /// content drift on the row. Immutable URLs (commit pins,
12    /// version-pinned registries, content-addressed CDNs, release
13    /// assets) are skipped - they cannot drift by construction.
14    /// Fetch failures land in `fetch_errors`: an unfetchable source
15    /// must never masquerade as `up-to-date`.
16    pub fn check_urls(&mut self, manifest: &Manifest, lockfile: Option<&Lockfile>, fetch: &FetchCache) {
17        let Some(lockfile) = lockfile else { return };
18
19        for artifact in &manifest.artifacts {
20            let Some(url) = &artifact.url else { continue };
21            if artifact.source.is_some() || is_immutable_url(url) {
22                continue;
23            }
24            let Some(locked) = lockfile.artifacts.iter().find(|locked| locked.source == *url) else {
25                continue;
26            };
27            self.compare_url(url, &locked.sha256, "artifact", fetch);
28        }
29        for input in &lockfile.inputs {
30            if is_immutable_url(&input.url) {
31                continue;
32            }
33            self.compare_url(&input.url, &input.sha256, "input", fetch);
34        }
35    }
36
37    /// Fetches `url` and compares the body hash to `locked_sha256`:
38    /// drift goes to `changed`, fetch failures to `fetch_errors`.
39    fn compare_url(&mut self, url: &str, locked_sha256: &str, kind: &'static str, fetch: &FetchCache) {
40        let body = match fetch.get(url) {
41            Ok(body) => body,
42            Err(error) => {
43                self.fetch_errors.push(format!("{url}: {error:?}"));
44                return;
45            }
46        };
47        let current = sha256_hex(&body);
48        if current != locked_sha256 {
49            self.changed.push(ChangedUrl {
50                kind,
51                url: url.to_string(),
52                locked_sha256: locked_sha256.to_string(),
53                current_sha256: current,
54            });
55        }
56    }
57}
58
59/// Whether a URL pins immutable content and can never drift:
60/// commit-SHA-pinned raw files, version-pinned registries,
61/// content-addressed CDNs and GitHub release assets.
62fn is_immutable_url(url: &str) -> bool {
63    url.contains("/releases/download/")
64        || url.contains("fonts.gstatic.com/")
65        || (url.contains("unpkg.com/") && url.contains('@'))
66        || url.split('/').any(is_commit_sha)
67}
