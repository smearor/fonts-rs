12/// Git upstream backend: resolves refs via `ls-remote` and maintains
13/// commit-keyed sparse checkouts under the cache root.
14pub struct GitSource {
15    /// Repository spec as declared in the manifest.
16    repo: GitRepository,
17    /// Remote URL derived from `repo`.
18    url: Url,
19}

21impl GitSource {
22    /// Creates a backend for `repo` (`owner/name` shorthand, URL or
23    /// SSH remote); the remote URL is derived from the repo spec.
24    pub fn new(repo: &GitRepository) -> Self {
25        Self {

27            url: repo.remote_url(),

31    /// Resolves `git_ref` (tag, branch, `HEAD` or a full commit SHA)
32    /// to a commit SHA via `git ls-remote`.
33    pub fn resolve(&self, git_ref: &GitRef) -> miette::Result<String> {
34        if git_ref.is_commit_sha() {
35            return Ok(git_ref.to_string());
36        }
37        let output = git(&["ls-remote", &self.url, &format!("{git_ref}^{{}}"), git_ref])?;
38        let mut tag_peeled = None;
39        let mut tag = None;
40        let mut branch = None;
41        for line in output.lines() {
42            let (sha, name) = line.split_once('\t').ok_or_else(|| miette!("Malformed ls-remote line: {line}"))?;
43            match name {
44                n if n == format!("refs/tags/{git_ref}^{{}}") => tag_peeled = Some(sha),
45                n if n == format!("refs/tags/{git_ref}") => tag = Some(sha),
46                n if n == format!("refs/heads/{git_ref}") => branch = Some(sha),
47                _ => {}
48            }
49        }
50        tag_peeled
51            .or(tag)
52            .or(branch)
53            .map(str::to_string)
54            .ok_or_else(|| miette!("Ref '{git_ref}' not found on {}", self.url))
55    }
56
57    /// Resolves the remote's default HEAD via `git ls-remote <url> HEAD`.
58    pub fn default_head(&self) -> miette::Result<String> {
59        let output = git(&["ls-remote", &self.url, "HEAD"])?;
60        output
61            .lines()
62            .next()
63            .and_then(|line| line.split('\t').next())
64            .map(str::to_string)
65            .ok_or_else(|| miette!("No HEAD advertised by {}", self.url))
66    }

68    /// Whether `git_ref` names a tag on the remote.
69    pub fn is_tag(&self, git_ref: &GitRef) -> miette::Result<bool> {
70        Ok(!git(&["ls-remote", &self.url, &format!("refs/tags/{git_ref}")])?.trim().is_empty())
71    }

73    /// The newest version-like tag (`v?\d+.\d+...`) on the remote,
74    /// compared by numeric components. Non-version tags (pre-releases,
75    /// named tags) are ignored; `^{}` deref lines are skipped.
76    pub fn latest_tag(&self) -> miette::Result<GitRef> {
77        let output = git(&["ls-remote", "--tags", &self.url])?;
78        output
79            .lines()
80            .filter_map(|line| line.split('\t').nth(1))
81            .filter_map(|name| name.strip_prefix("refs/tags/"))
82            .filter(|name| !name.ends_with("^{}"))
83            .filter_map(|name| version_key(name).map(|key| (key, name)))
84            .max_by(|(a, _), (b, _)| a.cmp(b))
85            .map(|(_, name)| GitRef::from(name))
86            .ok_or_else(|| miette!("No version tags found on {}", self.url))
87    }

89    /// The re-pin target for `git_ref`: tag pins track the newest
90    /// version tag, commit pins track the default HEAD; floating
91    /// branch refs have no update target. Returns `(new_ref, commit)`.
92    pub fn update_target(&self, git_ref: &GitRef) -> miette::Result<Option<(GitRef, String)>> {
93        if !git_ref.is_commit_sha() && self.is_tag(git_ref)? {
94            let tag = self.latest_tag()?;
95            let commit = self.resolve(&tag)?;
96            return Ok(Some((tag, commit)));
97        }
98        if git_ref.is_commit_sha() {
99            let head = self.default_head()?;
100            return Ok(Some((GitRef::from(head.clone()), head)));
101        }
102        Ok(None)
103    }

105    /// Ensures a sparse checkout of `commit` exists in the cache and
106    /// returns it. `sparse_paths` are gitignore-style patterns matched
107    /// against the repository root (`dir` covers the whole tree). The
108    /// cache key includes a digest of the pattern set so manifests
109    /// with different file selections never share a checkout.
110    pub fn checkout(&self, commit: &str, sparse_paths: &[String]) -> miette::Result<ResolvedSource> {
111        let sparse_key = crate::sha256::sha256_hex(sparse_paths.join("\n").as_bytes());
112        let dir = cache_root()?.join(repo_key(&self.repo)).join(format!("{commit}-{}", &sparse_key[..12]));
113        if dir.is_dir() {
114            return Ok(ResolvedSource {
115                checkout: dir,
116                cache_hit: true,
117            });
118        }
119
120        let tmp = dir.with_file_name(format!(".{}.tmp", dir.file_name().unwrap_or_default().to_string_lossy()));
121        if tmp.exists() {
122            std::fs::remove_dir_all(&tmp).into_diagnostic()?;
123        }
124        if let Some(parent) = tmp.parent() {
125            std::fs::create_dir_all(parent).into_diagnostic()?;
126        }
127        std::fs::create_dir_all(&tmp).into_diagnostic()?;
128
129        let result = (|| -> miette::Result<()> {
130            git_in(&tmp, &["init", "--quiet"])?;
131            git_in(&tmp, &["remote", "add", "origin", &self.url])?;
132            git_in(&tmp, &["sparse-checkout", "init", "--no-cone"])?;
133            let mut set_args = vec!["sparse-checkout", "set", "--no-cone"];
134            set_args.extend(sparse_paths.iter().map(String::as_str));
135            git_in(&tmp, &set_args)?;
136            git_in(&tmp, &["fetch", "--quiet", "--depth", "1", "--filter=blob:none", "origin", commit])?;
137            git_in(&tmp, &["checkout", "--quiet", "--detach", "FETCH_HEAD"])?;
138            Ok(())
139        })();
140
141        match result {
142            Ok(()) => {
143                std::fs::rename(&tmp, &dir).into_diagnostic()?;
144                Ok(ResolvedSource {
145                    checkout: dir,
146                    cache_hit: false,
147                })
148            }
149            Err(e) => {
150                let _ = std::fs::remove_dir_all(&tmp);
151                Err(e.wrap_err(format!("Failed to fetch {}@{commit}", self.repo)))
152            }
153        }
154    }
155}

157/// Whether `value` looks like a full 40-hex commit SHA.
158pub(crate) fn is_commit_sha(value: &str) -> bool {
159    value.len() == 40 && value.chars().all(|c| c.is_ascii_hexdigit())
160}

162/// Version key for tag sorting: `vX.Y.Z...` → `[X, Y, Z]`; `None`
163/// for non-numeric tags (pre-releases, named tags). Requires at
164/// least a major.minor pair so `v1` does not outrank real releases.
165fn version_key(tag: &str) -> Option<Vec<u64>> {
166    let tag = tag.strip_prefix('v').unwrap_or(tag);
167    let parts: Option<Vec<u64>> = tag.split('.').map(|part| part.parse().ok()).collect();
168    parts.filter(|parts| parts.len() >= 2)
169}

171/// Runs `git` with `args` outside a repository and returns stdout.
172fn git(args: &[&str]) -> miette::Result<String> {
173    run_git(None, args)
174}
175
176/// Runs `git` inside `dir` and returns stdout.
177fn git_in(dir: &Path, args: &[&str]) -> miette::Result<String> {
178    run_git(Some(dir), args)
179}
180
181/// Shared git subprocess wrapper: fails with stderr on non-zero exit.
182fn run_git(dir: Option<&Path>, args: &[&str]) -> miette::Result<String> {
183    let mut command = Command::new("git");
184    command.env("GIT_TERMINAL_PROMPT", "0").args(args);
185    if let Some(dir) = dir {
186        command.current_dir(dir);
187    }
188    let output = command.output().into_diagnostic().map_err(|e| e.wrap_err("Failed to run git"))?;
189    if !output.status.success() {
190        return Err(miette!("git {} failed: {}", args.join(" "), String::from_utf8_lossy(&output.stderr).trim()));
191    }
192    String::from_utf8(output.stdout).into_diagnostic()
193}

195/// Returns the sorted, deduplicated non-cone sparse-checkout pattern
196/// set for a manifest. Artifact sources with a glob `pattern` yield
197/// `source/pattern` (so e.g. `assets/**/ic_fluent_*_24_regular.svg`
198/// checks out only the 24px slices of a giant tree); plain sources
199/// match the file itself or, for directories, the whole subtree.
200/// `extra` covers license sources and explicit `sparse` entries.
201/// Artifact `exclude` globs become `!`-negated patterns so excluded
202/// blobs are not checked out either.
203pub fn sparse_paths(artifacts: impl Iterator<Item = (String, Option<String>, Vec<String>)>, extra: impl Iterator<Item = String>) -> Vec<String> {
204    let mut paths: Vec<String> = artifacts
205        .flat_map(|(source, pattern, excludes)| {
206            let source = source.trim_end_matches('/').to_string();
207            let include = match pattern {
208                Some(pattern) => format!("{source}/{pattern}"),
209                None => source.clone(),
210            };
211            std::iter::once(include).chain(excludes.into_iter().map(move |exclude| format!("!{source}/{exclude}")))
212        })
213        .chain(extra)
214        .collect();
215    paths.sort();
216    paths.dedup();
217    paths
218}

224    /// Parses dotted version tags; rejects pre-releases and majors.
225    #[test]
226    fn version_key_parsing() {
227        assert_eq!(version_key("v3.48.0"), Some(vec![3, 48, 0]));
228        assert_eq!(version_key("5.15.4"), Some(vec![5, 15, 4]));
229        assert_eq!(version_key("v0.50beta1"), None);
230        assert_eq!(version_key("release-2024"), None);
231        assert_eq!(version_key("v1"), None);
232    }

236    fn detects_commit_sha() {
237        assert!(is_commit_sha("a563cf9166f4f91aa617557ed272612b7f0a2f72"));
238        assert!(!is_commit_sha("v2.2.0"));
239        assert!(!is_commit_sha("a563cf9166f4f91aa617557ed272612b7f0a2f7")); // 39 chars
240    }

242    /// Glob patterns extend the source path; excludes become `!`
243    /// negations; plain sources stay whole.
244    #[test]
245    fn sparse_paths_combine_source_pattern_and_excludes() {
246        let paths = sparse_paths(
247            [
248                ("optimized/24/outline".to_string(), Some("*.svg".to_string()), Vec::new()),

261            [

265                "optimized/24/outline/*.svg",
