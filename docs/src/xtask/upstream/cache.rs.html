1use miette::miette;
2use std::path::PathBuf;
3
4/// Returns the root of the upstream checkout cache:
5/// `$XDG_CACHE_HOME/fonts-rs/upstream` (fallback `~/.cache/...`).
6///
7/// Checkouts are keyed by `<repo-key>/<commit>` so repeated syncs of
8/// the same resolved commit are free.
9pub fn cache_root() -> miette::Result<PathBuf> {
10    if let Some(dir) = std::env::var_os("XDG_CACHE_HOME") {
11        return Ok(PathBuf::from(dir).join("fonts-rs/upstream"));
12    }
13    let home = std::env::var_os("HOME").ok_or_else(|| miette!("Neither XDG_CACHE_HOME nor HOME is set"))?;
14    Ok(PathBuf::from(home).join(".cache/fonts-rs/upstream"))
15}
16
17/// Derives a filesystem-safe cache key for a repository spec
18/// (`owner/name` or URL): alphanumerics plus `-`, `.`, `_`.
19pub fn repo_key(repo: &str) -> String {
20    repo.chars()
21        .map(|c| if c.is_ascii_alphanumeric() || matches!(c, '-' | '.' | '_') { c } else { '_' })
22        .collect()
23}
